    17→/* 登录方式切换：点击后真实切换，并通过总线联动其它组件的显隐           */
    18→/* ------------------------------------------------------------------ */
    19→export function LoginTabsInteractive({ props }: InteractiveCtx) {
    20→  const channel = String(props.channel || 'loginMode');
    21→  const value = useInteractionBus((s) => s.values[channel]);
    22→  const setBus = useInteractionBus((s) => s.set);
    23→  const active = value ?? String(props.active ?? 'left');
    24→  /* 频道默认值：保证初始「选中项」与密码/验证码框的显隐联动同步（修复初始密码框消失） */
    25→  useChannelDefault(channel, String(props.active ?? 'left'));
    26→
    27→  return (
    28→    <div className="w-chip flex rounded-full p-1" style={{ borderRadius: '999px' }}>
    29→      <div className="relative flex flex-1">
    30→        <motion.span
    31→          className="absolute inset-y-0 rounded-full"
    32→          style={{
    33→            width: '50%',
    34→            background: 'var(--p)',
    35→            boxShadow: '0 2px 8px color-mix(in srgb, var(--p) 45%, transparent)',
    36→          }}
    37→          animate={{ x: active === 'right' ? '100%' : '0%' }}
    38→          transition={{ type: 'spring', stiffness: 520, damping: 38 }}
    39→        />
    40→        {(['left', 'right'] as const).map((k) => {
    41→          const isActive = active === k;
    42→          return (
    43→            <button
    44→              key={k}
    45→              type="button"
    46→              onClick={(e) => {
    47→                e.stopPropagation();
    48→                setBus(channel, k);
    49→              }}
    50→              className="relative z-10 flex-1 py-2 text-center text-[13px] font-semibold transition-colors"
    51→              style={isActive ? { color: 'var(--pf)' } : { opacity: 0.55 }}
    52→            >
    53→              {k === 'left' ? props.left : props.right}
    54→            </button>
    55→          );
    56→        })}
    57→      </div>
    58→    </div>
    59→  );
    60→}

   120→/* ------------------------------------------------------------------ */
   121→export function SmsInputInteractive({ props }: InteractiveCtx) {
   122→  const [code, setCode] = useState('');
   123→  const [left, setLeft] = useState(0);
   124→  const timers = useRef<ReturnType<typeof setTimeout>[]>([]);
   125→
   126→  const counting = left > 0;
   127→  useEffect(() => {
   128→    if (!counting) return;
   129→    const iv = setInterval(() => setLeft((v) => Math.max(0, v - 1)), 1000);
   130→    return () => clearInterval(iv);
   131→  }, [counting]);
   132→
   133→  useEffect(() => () => timers.current.forEach(clearTimeout), []);
   134→
   135→  const send = () => {
   136→    if (left > 0) return;
   137→    setLeft(60);
   138→    timers.current.push(
   139→      setTimeout(() => {
   140→        // 模拟收到短信，自动填充验证码
   141→        setCode('284616');
   142→      }, 1100)
   143→    );
   144→  };
   145→
   146→  return (
   147→    <div className="w-input flex h-11 items-center gap-2 pl-3 pr-1" style={{ borderRadius: 'var(--pr)' }}>
   148→      <ShieldCheck className="size-4 opacity-45" />
   149→      <input
   150→        value={code}
   151→        onChange={(e) => setCode(e.target.value.replace(/\D/g, '').slice(0, 6))}
   152→        placeholder={props.placeholder}
   153→        inputMode="numeric"
   154→        className="flex-1 bg-transparent text-sm outline-none placeholder:opacity-40"
   155→      />
   156→      <button
   157→        type="button"
   158→        onClick={(e) => {
   159→          e.stopPropagation();
   160→          send();
   161→        }}
   162→        className="flex h-8 shrink-0 items-center px-3 text-xs font-semibold disabled:opacity-60"
   163→        style={{
   164→          borderRadius: 'calc(var(--pr) - 4px)',
   165→          background: left > 0 ? 'transparent' : 'var(--p)',
   166→          color: left > 0 ? 'var(--p)' : 'var(--pf)',
   167→          border: left > 0 ? '1px solid var(--p)' : 'none',
   168→        }}
   169→      >
   170→        {left > 0 ? `${left}s 后重发` : props.btnText}
   171→      </button>
   172→    </div>
   173→  );

   178→/* ------------------------------------------------------------------ */
   179→export function PrimaryBtnInteractive({ props, onTap }: InteractiveCtx) {
   180→  const [st, setSt] = useState<'idle' | 'loading' | 'done'>('idle');
   181→  const timers = useRef<ReturnType<typeof setTimeout>[]>([]);
   182→  useEffect(() => () => timers.current.forEach(clearTimeout), []);
   183→
   184→  const click = (e: React.MouseEvent) => {
   185→    e.stopPropagation();
   186→    if (st !== 'idle') return;
   187→    setSt('loading');
   188→    timers.current.push(
   189→      setTimeout(() => {
   190→        setSt('done');
   191→        timers.current.push(
   192→          setTimeout(() => {
   193→            setSt('idle');
   194→            onTap?.();
   195→          }, 650)
   196→        );
   197→      }, 900)
   198→    );
   199→  };
   200→
   201→  return (
   202→    <button
   203→      type="button"
   204→      onClick={click}
   205→      className="flex h-12 w-full items-center justify-center gap-2 text-[15px] font-bold shadow-md transition-transform active:scale-[0.98]"
   206→      style={{ borderRadius: 'var(--pr)', background: 'var(--p)', color: 'var(--pf)' }}
   207→    >
   208→      {st === 'loading' ? (
   209→        <><LoaderCircle className="size-4 animate-spin" /> 正在登录…</>
   210→      ) : st === 'done' ? (
   211→        <><Check className="size-4" /> 登录成功</>
   212→      ) : (
   213→        <><Lock className="size-4" /> {props.text}{props.sub ? <span className="text-xs font-normal opacity-70">{props.sub}</span> : null}</>
   214→      )}
   215→    </button>
   216→  );
   217→}

   274→type Phase = 'idle' | 'splash' | 'auth' | 'granting' | 'success' | 'denied';
   275→
   276→export function SocialRowInteractive({ props, onTap }: InteractiveCtx) {
   277→  const [phase, setPhase] = useState<Phase>('idle');
   278→  const [brand, setBrand] = useState<BrandKey | null>(null);
   279→  const timers = useRef<ReturnType<typeof setTimeout>[]>([]);
   280→
   281→  useEffect(() => {
   282→    return () => timers.current.forEach(clearTimeout);
   283→  }, []);
   284→
   285→  const later = (fn: () => void, ms: number) => timers.current.push(setTimeout(fn, ms));
   286→
   287→  const launch = (key: BrandKey) => {
   288→    if (phase !== 'idle') return;
   289→    setBrand(key);
   290→    setPhase('splash');
   291→    later(() => setPhase('auth'), 850);
   292→  };
   293→
   294→  const agree = () => {
   295→    setPhase('granting');
   296→    later(() => {
   297→      setPhase('success');
   298→      later(() => {
   299→        setPhase('idle');
   300→        setBrand(null);
   301→        onTap?.();
   302→      }, 950);
   303→    }, 850);
   304→  };
   305→
   306→  const deny = () => {
   307→    setPhase('denied');
   308→    later(() => {
   309→      setPhase('idle');
   310→      setBrand(null);
   311→    }, 750);
   312→  };
   313→
   314→  const items = ([
   315→    ['showWechat', 'wechat'],
   316→    ['showApple', 'apple'],
   317→    ['showWeb', 'browser'],
   318→    ['showMail', 'mail'],
   319→    ['showPhone', 'phone'],
   320→  ] as const)
   321→    .filter(([k]) => props[k] !== false)
   322→    .map(([, key]) => key);
   323→
   324→  const row = (
   325→    <div className="flex items-center justify-center gap-4 py-1.5">
   326→      {items.map((key) => {
   327→        const Icon = BRAND_ICONS[key];
   328→        const b = BRANDS[key];
   329→        return (
   330→          <button
   331→            key={key}
   332→            type="button"
   333→            onClick={(e) => {
   334→              e.stopPropagation();
   335→              launch(key);
   336→            }}
   337→            className="flex touch-manipulation flex-col items-center gap-1.5 transition-transform active:scale-90"
   338→          >
   339→            <span
   340→              className="flex size-11 items-center justify-center rounded-full border w-line shadow-sm transition-colors"
   341→              style={{ background: `${b.color}14`, borderColor: `${b.color}30` }}
   342→            >
   343→              <Icon className="size-5" style={{ color: b.color }} />
   344→            </span>
   345→            <span className="text-[10px] opacity-45">
   346→              {key === 'phone' ? '本机号码' : b.label}
   347→            </span>
   348→          </button>
   349→        );
   350→      })}
   351→    </div>
   352→  );
   353→
   354→  if (phase === 'idle' || !brand) return row;
   355→
   356→  const b = BRANDS[brand];
   357→  const appName = String(props.appName || '星云 App');
   358→
   359→  /* 唤起 / 授权 全屏覆盖层（挂载到手机屏幕容器） */
   360→  const overlay = (
   361→    <AnimatePresence>
   362→      <motion.div
   363→        key={phase}
   364→        initial={{ opacity: 0 }}
   365→        animate={{ opacity: 1 }}
   366→        exit={{ opacity: 0 }}
   367→        transition={{ duration: 0.22 }}
   368→        className="absolute inset-0 z-[80] flex flex-col overflow-hidden"
   369→        style={{ background: phase === 'splash' ? '#000' : b.dark ? '#0a0a0c' : '#f4f5f7' }}
   370→        onClick={(e) => e.stopPropagation()}
   371→      >
   372→        {/* 仿系统状态栏 */}
   373→        <div
   374→          className="flex h-12 shrink-0 items-center justify-between px-7 pt-2 text-[13px] font-semibold"
   375→          style={{ color: phase === 'splash' || b.dark ? '#fff' : '#111' }}
   376→        >
   377→          <span>9:41</span>
   378→          <div className="flex items-center gap-1.5">
   379→            <Signal className="size-3.5" />
   380→            <Wifi className="size-3.5" />
   381→            <BatteryFull className="size-4" />
   382→          </div>
   383→        </div>
   384→
   385→        {phase === 'splash' && (
   386→          <div className="flex flex-1 flex-col items-center justify-center gap-4 text-white">
   387→            <motion.div
   388→              initial={{ scale: 0.6, opacity: 0 }}
   389→              animate={{ scale: 1, opacity: 1 }}
   390→              transition={{ type: 'spring', stiffness: 300, damping: 22 }}
   391→              className="flex size-20 items-center justify-center rounded-[22px] shadow-2xl"
   392→              style={{ background: b.color }}
   393→            >
   394→              {(() => {
   395→                const Icon = BRAND_ICONS[brand];
   396→                return <Icon className="size-10 text-white" />;
   397→              })()}
   398→            </motion.div>
   399→            <p className="text-lg font-bold">{BRANDS[brand].label}</p>

   429→            <p className={`text-base font-bold ${b.dark ? 'text-white' : 'text-zinc-900'}`}>
   430→              授权成功
   431→            </p>
   432→            <p className={`text-xs ${b.dark ? 'text-white/50' : 'text-zinc-500'}`}>正在返回 {appName}…</p>
   433→          </div>
   434→        )}
   435→
   436→        {phase === 'denied' && (
   437→          <div className="flex flex-1 flex-col items-center justify-center gap-3">
   438→            <motion.div initial={{ scale: 0.6 }} animate={{ scale: 1 }}>

   441→            <p className={`text-sm font-semibold ${b.dark ? 'text-white/80' : 'text-zinc-600'}`}>已取消授权</p>

   443→        )}
   444→
   445→        {/* 仿 Home 条 */}
   446→        <div className="flex h-6 shrink-0 items-center justify-center">

   452→
   453→  const target = typeof document !== 'undefined' ? document.getElementById('phone-screen') : null;

   456→  return (

   458→      {row}
   459→      {overlayNode}
   460→    </>

   462→}

   895→/* 主按钮（fn.big-button）：点击 → 加载 → 成功 → 触发绑定跳转          */
   896→/* ------------------------------------------------------------------ */
   897→
   898→export function BigButtonInteractive({ props, onTap }: InteractiveCtx) {
   899→  const [st, setSt] = useState<'idle' | 'loading' | 'done'>('idle');
   900→  const timers = useRef<ReturnType<typeof setTimeout>[]>([]);
   901→  useEffect(() => () => timers.current.forEach(clearTimeout), []);
   902→
   903→  const variant = props.style === 'outline' || props.style === 'ghost' ? props.style : 'primary';
   904→  const br = props.radius === 'normal' ? 'var(--pr)' : '999px';
   905→  const skin = variant === 'primary'
   906→    ? { background: 'var(--p)', color: 'var(--pf)' }
   907→    : variant === 'outline'
   908→      ? { border: '1.5px solid var(--p)', color: 'var(--p)' }
   909→      : { color: 'var(--p)' };
   910→
   911→  const click = (e: React.MouseEvent) => {
   912→    e.stopPropagation();
   913→    if (st !== 'idle') return;
   914→    setSt('loading');
   915→    timers.current.push(
   916→      setTimeout(() => {
   917→        setSt('done');
   918→        timers.current.push(
   919→          setTimeout(() => {
   920→            setSt('idle');
   921→            onTap?.();
   922→          }, 620)
   923→        );
   924→      }, 700)
   925→    );
   926→  };
   927→
   928→  return (
   929→    <button
   930→      type="button"
   931→      onClick={click}
   932→      className={`flex h-12 w-full items-center justify-center gap-2 text-[15px] font-bold transition-transform active:scale-[0.98] ${variant === 'primary' ? 'shadow-md' : variant === 'ghost' ? 'w-chip' : ''}`}
   933→      style={{ borderRadius: br, ...skin }}
   934→    >
   935→      {st === 'loading' ? (
   936→        <><LoaderCircle className="size-4 animate-spin" /> 请稍候…</>
   937→      ) : st === 'done' ? (
   938→        <><Check className="size-4" /> 操作成功</>
   939→      ) : (
   940→        props.text
   941→      )}
   942→    </button>
   943→  );
   944→}
