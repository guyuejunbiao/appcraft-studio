   437→function Label({ children }: { children: React.ReactNode }) {

   441→/** 自由布局数值输入（X/Y/W/H） */
   442→function NumField({
   443→  label, value, min, max, onChange,
   444→}: {
   445→  label: string;
   446→  value: number;
   447→  min: number;
   448→  max: number;
   449→  onChange: (v: number) => void;
   450→}) {
   451→  return (
   452→    <div>
   453→      <Label>{label}</Label>
   454→      <Input
   455→        type="number"
   456→        className="h-8 text-xs tabular-nums"
   457→        value={Math.round(value)}
   458→        min={min}
   459→        max={max}
   460→        onChange={(e) => {
   461→          const v = Number(e.target.value);
   462→          if (!Number.isNaN(v)) onChange(Math.max(min, Math.min(max, v)));
   463→        }}
   464→      />
   465→    </div>
   466→  );
   467→}

   473→      <p className="text-[10px] text-zinc-400">{label}</p>
   474→    </div>
   475→  );
   476→}

   478→/** 属性字段控件 */
   479→function FieldControl({
   480→  field, value, onChange,
   481→}: {
   482→  field: PropField;
   483→  value: any;
   484→  onChange: (v: any) => void;
   485→}) {
   486→  if (field.type === 'switch') {
   487→    return (
   488→      <div className="flex items-center justify-between">
   489→        <Label>{field.label}</Label>
   490→        <Switch checked={value !== false} onCheckedChange={onChange} />
   491→      </div>
   492→    );
   493→  }
   494→  if (field.type === 'textarea') {
   495→    return (
   496→      <div>
   497→        <Label>{field.label}</Label>
   498→        <Textarea
   499→          className="min-h-16 text-xs"
   500→          value={String(value ?? '')}
   501→          onChange={(e) => onChange(e.target.value)}
   502→          placeholder={field.placeholder}
   503→        />
   504→      </div>
   505→    );
   506→  }
   507→  if (field.type === 'select') {
   508→    return (
   509→      <div>
   510→        <Label>{field.label}</Label>
   511→        <Select value={String(value ?? '')} onValueChange={onChange}>
   512→          <SelectTrigger className="h-8 text-xs"><SelectValue /></SelectTrigger>
   513→          <SelectContent>
   514→            {(field.options ?? []).map((o) => (
   515→              <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>
   516→            ))}
   517→          </SelectContent>
   518→        </Select>
   519→      </div>
   520→    );
   521→  }
   522→  if (field.type === 'color') {
   523→    return (
   524→      <div>
   525→        <Label>{field.label}</Label>
   526→        <div className="flex items-center gap-2">
   527→          <input
   528→            type="color"
   529→            value={String(value ?? '#000000')}
   530→            onChange={(e) => onChange(e.target.value)}
   531→            className="size-8 cursor-pointer rounded-lg border"
   532→          />
   533→          <Input className="h-8 flex-1 font-mono text-xs" value={String(value ?? '')} onChange={(e) => onChange(e.target.value)} />
   534→        </div>
   535→      </div>
   536→    );
   537→  }
   538→  if (field.type === 'number') {
   539→    return (
   540→      <div>
   541→        <div className="mb-1.5 flex items-center justify-between">
   542→          <Label>{field.label}</Label>
   543→          <span className="text-xs font-semibold tabular-nums text-zinc-500">{Number(value ?? 0)}</span>
   544→        </div>
   545→        <Slider
   546→          value={[Number(value ?? 0)]}
   547→          min={field.min ?? 0}
   548→          max={field.max ?? 100}
   549→          step={field.step ?? 1}
   550→          onValueChange={([v]) => onChange(v)}
   551→        />
   552→      </div>
   553→    );
   554→  }
   555→  return (
   556→    <div>
   557→      <Label>{field.label}</Label>
   558→      <Input
   559→        className="h-8 text-xs"
   560→        value={String(value ?? '')}
   561→        onChange={(e) => onChange(e.target.value)}
   562→        placeholder={field.placeholder}
   563→      />
   564→    </div>
   565→  );
   566→}

   568→/** 交互（连接）编辑器 */
   569→function ConnectionEditor({
   570→  pageId, widgetId, widgetName, fromConnections, slots, onAdd, onRemove,
   571→}: {
   572→  pageId: string;
   573→  widgetId: string;
   574→  widgetName: string;
   575→  fromConnections: { id: string; toPageId: string; animation: string; slot?: string }[];
   576→  slots: { key: string; label: string }[];
   577→  onAdd: (c: { fromPageId: string; fromWidgetId: string; toPageId: string; animation: 'slide' | 'fade' | 'push' | 'none'; slot?: string }) => void;
   578→  onRemove: (id: string) => void;
   579→}) {
   580→  const pages = useBuilder((s) => s.pages);
   581→  const [toPage, setToPage] = useState('');
   582→  const [anim, setAnim] = useState<'slide' | 'fade' | 'push' | 'none'>('slide');
   583→  const [slot, setSlot] = useState('widget');
   584→
   585→  const targets = useMemo(() => pages.filter((p) => p.id !== pageId), [pages, pageId]);
   586→  const slotLabel = (key?: string) =>
   587→    key ? (slots.find((s) => s.key === key)?.label ?? `槽位 ${key}`) : null;
   588→
   589→  return (
   590→    <div className="space-y-3">
   591→      <div className="flex items-center gap-1.5 text-xs text-zinc-500">
   592→        <Link2 className="size-3.5" />
   593→        「{widgetName}」被点击后跳转到
   594→      </div>
   595→
   596→      {fromConnections.length > 0 && (
   597→        <div className="space-y-1.5">
   598→          {fromConnections.map((c) => {
   599→            const target = pages.find((p) => p.id === c.toPageId);
   600→            const animLabel = ANIM_OPTIONS.find((a) => a.value === c.animation)?.label ?? c.animation;
   601→            const sLabel = slotLabel(c.slot);
   602→            return (
   603→              <div key={c.id} className="flex items-center gap-2 rounded-xl border bg-zinc-50 px-2.5 py-2">
   604→                <ArrowRight className="size-3.5 shrink-0 text-emerald-500" />
   605→                <span className="min-w-0 flex-1 truncate text-xs font-semibold">
   606→                  {sLabel && <span className="mr-1 rounded bg-emerald-100 px-1 py-0.5 text-[9px] font-semibold text-emerald-600">{sLabel}</span>}
   607→                  {target?.name ?? '未知页面'}
   608→                  <span className="ml-1.5 rounded bg-zinc-200 px-1 py-0.5 text-[9px] font-normal text-zinc-500">{animLabel}</span>
   609→                </span>
   610→                <button
   611→                  className="text-zinc-400 hover:text-rose-500"
   612→                  onClick={() => onRemove(c.id)}
   613→                  title="删除连接"
   614→                >
   615→                  <X className="size-3.5" />
   616→                </button>
   617→              </div>
   618→            );
   619→          })}
   620→        </div>
   621→      )}
   622→
   623→      <div className="space-y-2 rounded-xl border border-dashed p-3">
   624→        {slots.length > 0 && (
   625→          <div>
   626→            <Label>绑定目标</Label>
   627→            <Select value={slot} onValueChange={setSlot}>
   628→              <SelectTrigger className="h-8 text-xs"><SelectValue /></SelectTrigger>
   629→              <SelectContent>
   630→                <SelectItem value="widget">整个组件（任意点击）</SelectItem>
   631→                {slots.map((s) => (
   632→                  <SelectItem key={s.key} value={s.key}>{s.label}</SelectItem>
   633→                ))}
   634→              </SelectContent>
   635→            </Select>
   636→            {slot !== 'widget' && (
   637→              <p className="mt-1 text-[10px] leading-3.5 text-zinc-400">
   638→                预览中点击该标签即跳转，适合底部导航切换页面
   639→              </p>
   640→            )}
   641→          </div>
   642→        )}
   643→        <Select value={toPage} onValueChange={setToPage}>
   644→          <SelectTrigger className="h-8 text-xs"><SelectValue placeholder="选择目标页面" /></SelectTrigger>
   645→          <SelectContent>
   646→            {targets.map((p) => (
   647→              <SelectItem key={p.id} value={p.id}>
   648→                {p.name}
   649→                {p.isHome ? '（主页）' : ''}
   650→              </SelectItem>
   651→            ))}
   652→          </SelectContent>
   653→        </Select>
   654→        <Select value={anim} onValueChange={(v) => setAnim(v as never)}>
   655→          <SelectTrigger className="h-8 text-xs"><SelectValue /></SelectTrigger>
   656→          <SelectContent>
   657→            {ANIM_OPTIONS.map((a) => (
   658→              <SelectItem key={a.value} value={a.value}>{a.label}</SelectItem>
   659→            ))}
   660→          </SelectContent>
   661→        </Select>
   662→        <Button
   663→          size="sm"
   664→          className="w-full"
   665→          disabled={!toPage}
   666→          onClick={() => {
   667→            if (!toPage) return;
   668→            onAdd({
   669→              fromPageId: pageId,
   670→              fromWidgetId: widgetId,
   671→              toPageId: toPage,
   672→              animation: anim,
   673→              slot: slot === 'widget' ? undefined : slot,
   674→            });
   675→            setToPage('');
   676→          }}
   677→        >
   678→          <Plus className="mr-1 size-3.5" /> 添加跳转连接
   679→        </Button>
   680→      </div>
   681→
   682→      {targets.length === 0 && (
   683→        <p className="text-center text-[11px] text-zinc-400">只有一页，先添加新页面才能建立连接</p>
   684→      )}
   685→    </div>
   686→  );
   687→}
