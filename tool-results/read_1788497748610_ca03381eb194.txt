     4→     4→import {

     9→     9→import { useDnd } from '@/lib/dnd-store';

    15→    15→
    16→    16→const SNAP = 4;
    17→    17→const MIN_W = 48;
    18→    18→const MIN_H = 32;
    19→    19→const SCREEN_W = 375;
    20→    20→const EMPTY_GUIDES: { v: number[]; h: number[] } = { v: [], h: [] };
    21→    21→const HANDLE_DIRS = ['n', 's', 'e', 'w', 'ne', 'nw', 'se', 'sw'] as const;
    22→    22→const HANDLE_CLASS: Record<string, string> = {
    23→    23→  n: 'left-1/2 -top-1.5 -translate-x-1/2 cursor-ns-resize',
    24→    24→  s: 'left-1/2 -bottom-1.5 -translate-x-1/2 cursor-ns-resize',
    25→    25→  e: '-right-1.5 top-1/2 -translate-y-1/2 cursor-ew-resize',
    26→    26→  w: '-left-1.5 top-1/2 -translate-y-1/2 cursor-ew-resize',
    27→    27→  ne: '-right-1.5 -top-1.5 cursor-nesw-resize',
    28→    28→  sw: '-left-1.5 -bottom-1.5 cursor-nesw-resize',
    29→    29→  nw: '-left-1.5 -top-1.5 cursor-nwse-resize',
    30→    30→  se: '-right-1.5 -bottom-1.5 cursor-nwse-resize',
    31→    31→};
    32→    32→
    33→    33→/**
    34→    34→ * 编辑画布（指针拖拽系统）：
    35→    35→ * - 从组件仓库拖入（Pointer Events，鼠标/触屏通吃，替代不可靠的 HTML5 DnD）
    36→    36→ * - 流式布局：拖拽排序 + 插入指示线
    37→    37→ * - 自由布局：任意摆放位置 + 8 向手柄调整大小 + 对齐吸附线 + 4px 网格吸附
    38→    38→ */
    39→    39→export function Canvas() {
    40→    40→  const pages = useBuilder((s) => s.pages);
    41→    41→  const currentPageId = useBuilder((s) => s.currentPageId);
    42→    42→  const selectedWidgetId = useBuilder((s) => s.selectedWidgetId);
    43→    43→  const theme = useBuilder((s) => s.project?.theme);
    44→    44→  const select = useBuilder((s) => s.select);
    45→    45→  const removeWidget = useBuilder((s) => s.removeWidget);
    46→    46→  const duplicateWidget = useBuilder((s) => s.duplicateWidget);
    47→    47→
    48→    48→  const [zoom, setZoom] = useState(0.85);
    49→    49→  const [indicator, setIndicator] = useState<{ index: number; y: number } | null>(null);
    50→    50→  const [freePreview, setFreePreview] = useState<{ x: number; y: number; w: number } | null>(null);
    51→    51→  const [guides, setGuides] = useState<{ v: number[]; h: number[] }>(EMPTY_GUIDES);
    52→    52→  const [sizeBadge, setSizeBadge] = useState<string | null>(null);
    53→    53→  const [freeHeight, setFreeHeight] = useState(700);
    54→    54→
    55→    55→  const itemRefs = useRef<Map<string, HTMLDivElement>>(new Map());
    56→    56→  const contentRef = useRef<HTMLDivElement>(null);
    57→    57→  const suppressClickRef = useRef(false);
    58→    58→  const dragMetaRef = useRef<{ offX: number; offY: number; batch: boolean }>({ offX: 0, offY: 0, batch: false });
    59→    59→  const resizeRef = useRef<null | {
    60→    60→    id: string; dir: string; startX: number; startY: number;
    61→    61→    rect: { x: number; y: number; w: number; h: number };
    62→    62→  }>(null);
    63→    63→  const freePreviewRef = useRef<{ x: number; y: number; w: number } | null>(null);
    64→    64→  const zoomRef = useRef(zoom);
    65→    65→  const isFreeRef = useRef(false);
    66→    66→
    67→    67→  useEffect(() => { zoomRef.current = zoom; }, [zoom]);
    68→    68→  useEffect(() => { isFreeRef.current = pages.find((p) => p.id === currentPageId)?.layout === 'free'; }, [pages, currentPageId]);
    69→    69→  useEffect(() => () => { useDnd.getState().end(); }, []);
    70→    70→
    71→    71→  const page = pages.find((p) => p.id === currentPageId);
    72→    72→  const isFree = page?.layout === 'free';
    73→    73→  const t = theme ?? { primary: '#f97316', radius: 'md' as const, dark: false };
    74→    74→  const dndActive = useDnd((s) => s.active);
    75→    75→
    76→    76→  /* ---------- 坐标与落点计算 ---------- */
    77→    77→
    78→    78→  const toLocal = useCallback((clientX: number, clientY: number) => {
    79→    79→    const el = contentRef.current;
    80→    80→    const rect = el?.getBoundingClientRect();
    81→    81→    if (!el || !rect) return { x: 0, y: 0 };
    82→    82→    return {
    83→    83→      x: (clientX - rect.left) / zoomRef.current,
    84→    84→      y: (clientY - rect.top) / zoomRef.current + el.scrollTop,
    85→    85→    };
    86→    86→  }, []);
    87→    87→
    88→    88→  const insideContent = useCallback((clientX: number, clientY: number) => {
    89→    89→    const rect = contentRef.current?.getBoundingClientRect();
    90→    90→    if (!rect) return false;
    91→    91→    const pad = 8;
    92→    92→    return (
    93→    93→      clientX >= rect.left - pad && clientX <= rect.right + pad &&
    94→    94→      clientY >= rect.top - pad && clientY <= rect.bottom + pad
    95→    95→    );
    96→    96→  }, []);
    97→    97→
    98→    98→  const computeDrop = useCallback((clientY: number) => {
    99→    99→    const b = useBuilder.getState();
   100→   100→    const list = b.pages.find((p) => p.id === b.currentPageId)?.components ?? [];
   101→   101→    const origin = contentRef.current?.getBoundingClientRect();
   102→   102→    const scrollTop = contentRef.current?.scrollTop ?? 0;
   103→   103→    for (let i = 0; i < list.length; i++) {
   104→   104→      const rect = itemRefs.current.get(list[i].id)?.getBoundingClientRect();
   105→   105→      if (!rect) continue;
   106→   106→      if (clientY < rect.top + rect.height / 2) {
   107→   107→        return { index: i, y: origin ? (rect.top - origin.top) / zoomRef.current + scrollTop : 0 };
   108→   108→      }
   109→   109→    }
   110→   110→    const last = list.length > 0 ? itemRefs.current.get(list[list.length - 1].id)?.getBoundingClientRect() : null;
   111→   111→    return { index: list.length, y: origin && last ? (last.bottom - origin.top) / zoomRef.current + scrollTop : 0 };
   112→   112→  }, []);
   113→   113→
   114→   114→  /** 自由布局吸附：其它组件边缘/中心 + 屏幕边缘/水平中心 + 4px 网格 */
   115→   115→  const snapPos = useCallback((x: number, y: number, movingId: string | null, w: number, h: number) => {
   116→   116→    const b = useBuilder.getState();
   117→   117→    const comps = b.pages.find((p) => p.id === b.currentPageId)?.components ?? [];
   118→   118→    const T = 6;
   119→   119→    const vCands: { p: number; line: number }[] = [
   120→   120→      { p: 0, line: 0 },
   121→   121→      { p: (SCREEN_W - w) / 2, line: SCREEN_W / 2 },
   122→   122→      { p: SCREEN_W - w, line: SCREEN_W },
   123→   123→    ];
   124→   124→    const hCands: { p: number; line: number }[] = [{ p: 0, line: 0 }];
   125→   125→    comps.forEach((c) => {
   126→   126→      if (c.id === movingId) return;
   127→   127→      const cx = c.x ?? 0, cy = c.y ?? 0, cw = c.w ?? 355, ch = c.h ?? 56;
   128→   128→      vCands.push({ p: cx, line: cx }, { p: cx + cw / 2 - w / 2, line: cx + cw / 2 }, { p: cx + cw - w, line: cx + cw });
   129→   129→      hCands.push({ p: cy, line: cy }, { p: cy + ch / 2 - h / 2, line: cy + ch / 2 }, { p: cy + ch - h, line: cy + ch }, { p: cy + ch, line: cy + ch });
   130→   130→    });
   131→   131→    let nx = x, ny = y;
   132→   132→    const g: { v: number[]; h: number[] } = { v: [], h: [] };
   133→   133→    for (const c of vCands) {
   134→   134→      if (Math.abs(x - c.p) <= T) { nx = Math.max(0, Math.min(c.p, SCREEN_W - w)); g.v.push(c.line); break; }
   135→   135→    }
   136→   136→    for (const c of hCands) {
   137→   137→      if (Math.abs(y - c.p) <= T) { ny = Math.max(0, c.p); g.h.push(c.line); break; }
   138→   138→    }
   139→   139→    if (g.v.length === 0) nx = Math.round(x / SNAP) * SNAP;
   140→   140→    if (g.h.length === 0) ny = Math.round(y / SNAP) * SNAP;
   141→   141→    return { x: Math.max(0, Math.min(nx, SCREEN_W - w)), y: Math.max(0, ny), guides: g };
   142→   142→  }, []);
   143→   143→
   144→   144→  /* ---------- 拖拽过程 ---------- */
   145→   145→
   146→   146→  const handleDragMove = useCallback((clientX: number, clientY: number) => {
   147→   147→    const st = useDnd.getState();
   148→   148→
   149→   149→    /* 缩放手柄拖拽中 */
   150→   150→    if (resizeRef.current) {
   151→   151→      const r = resizeRef.current;
   152→   152→      const dx = (clientX - r.startX) / zoomRef.current;
   153→   153→      const dy = (clientY - r.startY) / zoomRef.current;
   154→   154→      let { x, y, w, h } = r.rect;
   155→   155→      if (r.dir.includes('e')) w = r.rect.w + dx;
   156→   156→      if (r.dir.includes('s')) h = r.rect.h + dy;
   157→   157→      if (r.dir.includes('w')) { w = r.rect.w - dx; x = r.rect.x + dx; }
   158→   158→      if (r.dir.includes('n')) { h = r.rect.h - dy; y = r.rect.y + dy; }
   159→   159→      if (dirClampWest(r.dir) && x < 0) { w = r.rect.w + r.rect.x; x = 0; }
   160→   160→      if (dirClampNorth(r.dir) && y < 0) { h = r.rect.h + r.rect.y; y = 0; }
   161→   161→      w = Math.max(MIN_W, w);
   162→   162→      h = Math.max(MIN_H, h);
   163→   163→      if (x + w > SCREEN_W) w = SCREEN_W - x;
   164→   164→      x = Math.round(x / SNAP) * SNAP;
   165→   165→      y = Math.round(Math.max(0, y) / SNAP) * SNAP;
   166→   166→      w = Math.round(w / SNAP) * SNAP;
   167→   167→      h = Math.round(h / SNAP) * SNAP;
   168→   168→      useBuilder.getState().setWidgetRect(r.id, { x, y, w, h }, true);
   169→   169→      setSizeBadge(`${w} × ${h}`);
   170→   170→      return;
   171→   171→    }
   172→   172→
   173→   173→    if (!insideContent(clientX, clientY)) {
   174→   174→      setIndicator(null);
   175→   175→      setFreePreview(null);
   176→   176→      freePreviewRef.current = null;
   177→   177→      setGuides(EMPTY_GUIDES);
   178→   178→      return;
   179→   179→    }
   180→   180→
   181→   181→    if (isFreeRef.current) {
   182→   182→      setIndicator(null);
   183→   183→      if (st.kind === 'new') {
   184→   184→        const def = getWidget(st.widgetType ?? '');
   185→   185→        const w0 = def?.fullBleed ? SCREEN_W : 355;
   186→   186→        const { x, y } = toLocal(clientX, clientY);
   187→   187→        const snapped = snapPos(x - w0 / 2, y - 24, null, w0, 56);
   188→   188→        const preview = { x: snapped.x, y: snapped.y, w: w0 };
   189→   189→        freePreviewRef.current = preview;
   190→   190→        setFreePreview(preview);
   191→   191→        setGuides(snapped.guides);
   192→   192→      } else if (st.kind === 'move') {
   193→   193→        const id = st.moveId!;
   194→   194→        const b = useBuilder.getState();
   195→   195→        const comp = b.pages
   196→   196→          .find((p) => p.id === b.currentPageId)
   197→   197→          ?.components.find((c) => c.id === id);
   198→   198→        if (!comp) return;
   199→   199→        if (!dragMetaRef.current.batch) {
   200→   200→          b.beginBatch();
   201→   201→          dragMetaRef.current.batch = true;
   202→   202→        }
   203→   203→        const cw = comp.w ?? 355;
   204→   204→        const ch = comp.h ?? 56;
   205→   205→        const { x, y } = toLocal(clientX, clientY);
   206→   206→        const snapped = snapPos(x - dragMetaRef.current.offX, y - dragMetaRef.current.offY, id, cw, ch);
   207→   207→        b.setWidgetRect(id, { x: snapped.x, y: snapped.y }, true);
   208→   208→        setGuides(snapped.guides);
   209→   209→        setFreePreview(null);
   210→   210→        freePreviewRef.current = null;
   211→   211→      }
   212→   212→    } else {
   213→   213→      setFreePreview(null);
   214→   214→      freePreviewRef.current = null;
   215→   215→      setGuides(EMPTY_GUIDES);
   216→   216→      setIndicator(computeDrop(clientY));
   217→   217→    }
   218→   218→  }, [computeDrop, insideContent, snapPos, toLocal]);
   219→   219→
   220→   220→  const handleUp = useCallback((clientX: number, clientY: number) => {
   221→   221→    const st = useDnd.getState();
   222→   222→    const b = useBuilder.getState();
   223→   223→    const pageNow = b.pages.find((p) => p.id === b.currentPageId);
   224→   224→    const freeNow = pageNow?.layout === 'free';
   225→   225→
   226→   226→    /* 缩放结束：合并历史 */
   227→   227→    if (resizeRef.current) {
   228→   228→      if (dragMetaRef.current.batch) b.endBatch();
   229→   229→      resizeRef.current = null;
   230→   230→      setSizeBadge(null);
   231→   231→      dragMetaRef.current = { offX: 0, offY: 0, batch: false };
   232→   232→      useDnd.getState().end();
   233→   233→      return;
   234→   234→    }
   235→   235→
   236→   236→    if (st.started) {
   237→   237→      suppressClickRef.current = true;
   238→   238→      useDnd.setState({ suppressNextClick: true });
   239→   239→      setTimeout(() => { suppressClickRef.current = false; }, 90);
   240→   240→    }
   241→   241→
   242→   242→    if (st.started && st.kind === 'new' && st.widgetType) {
   243→   243→      if (insideContent(clientX, clientY)) {
   244→   244→        if (freeNow) {
   245→   245→          const { x, y } = toLocal(clientX, clientY);
   246→   246→          const def = getWidget(st.widgetType);
   247→   247→          const w0 = def?.fullBleed ? SCREEN_W : 355;
   248→   248→          const fp = freePreviewRef.current;
   249→   249→          const rx = fp?.x ?? Math.max(0, Math.round((x - w0 / 2) / SNAP) * SNAP);
   250→   250→          const ry = fp?.y ?? Math.max(0, Math.round((y - 24) / SNAP) * SNAP);
   251→   251→          b.addWidget(st.widgetType, undefined, { x: rx, y: ry, w: w0 });
   252→   252→        } else {
   253→   253→          b.addWidget(st.widgetType, computeDrop(clientY).index);
   254→   254→        }
   255→   255→      }
   256→   256→    } else if (st.started && st.kind === 'move' && st.moveId) {
   257→   257→      if (freeNow) {
   258→   258→        if (dragMetaRef.current.batch) b.endBatch();
   259→   259→      } else {
   260→   260→        b.moveWidget(st.moveId, computeDrop(clientY).index);
   261→   261→      }
   262→   262→    }
   263→   263→
   264→   264→    dragMetaRef.current = { offX: 0, offY: 0, batch: false };
   265→   265→    setIndicator(null);
   266→   266→    setFreePreview(null);
   267→   267→    freePreviewRef.current = null;
   268→   268→    setGuides(EMPTY_GUIDES);
   269→   269→    setSizeBadge(null);
   270→   270→    useDnd.getState().end();
   271→   271→  }, [computeDrop, insideContent, toLocal]);
   272→   272→
   273→   273→  /* 拖拽期间挂全局指针监听 */
   274→   274→  useEffect(() => {
   275→   275→    if (!dndActive) return;
   276→   276→    const onMove = (e: PointerEvent) => {
   277→   277→      useDnd.getState().move(e.clientX, e.clientY);
   278→   278→      if (!useDnd.getState().started) return;
   279→   279→      handleDragMove(e.clientX, e.clientY);
   280→   280→    };
   281→   281→    const onUp = (e: PointerEvent) => handleUp(e.clientX, e.clientY);
   282→   282→    window.addEventListener('pointermove', onMove);
   283→   283→    window.addEventListener('pointerup', onUp, true);
   284→   284→    window.addEventListener('pointercancel', onUp, true);
   285→   285→    return () => {
   286→   286→      window.removeEventListener('pointermove', onMove);
   287→   287→      window.removeEventListener('pointerup', onUp, true);
   288→   288→      window.removeEventListener('pointercancel', onUp, true);
   289→   289→    };
   290→   290→  }, [dndActive, handleDragMove, handleUp]);
   291→   291→
   292→   292→  /* ---------- 布局切换（流式 → 自由：按当前渲染高度垂直堆叠迁移） ---------- */
   293→   293→
   294→   294→  const toggleLayout = () => {
   295→   295→    const b = useBuilder.getState();
   296→   296→    const pageNow = b.pages.find((p) => p.id === b.currentPageId);
   297→   297→    if (!pageNow) return;
   298→   298→    if (pageNow.layout === 'free') {
   299→   299→      b.updatePage(pageNow.id, { layout: 'flow' });
   300→   300→    } else {
   301→   301→      const rects: Record<string, { x: number; y: number; w: number }> = {};
   302→   302→      let y = 12;
   303→   303→      pageNow.components.forEach((c) => {
   304→   304→        const el = itemRefs.current.get(c.id);
   305→   305→        const h = el ? el.offsetHeight : 64;
   306→   306→        const full = getWidget(c.type)?.fullBleed;
   307→   307→        rects[c.id] = { x: full ? 0 : 10, y, w: full ? SCREEN_W : 355 };
   308→   308→        y += h + 12;
   309→   309→      });
   310→   310→      b.convertToFreeLayout(rects);
   311→   311→    }
   312→   312→  };
   313→   313→
   314→   314→  /* 自由布局内容总高（含未被 h 固定的自适应组件）— ResizeObserver 订阅式测量 */
   315→   315→  useEffect(() => {
   316→   316→    if (!isFree) return;
   317→   317→    const measure = () => {
   318→   318→      let max = 700;
   319→   319→      const b = useBuilder.getState();
   320→   320→      const comps = b.pages.find((p) => p.id === b.currentPageId)?.components ?? [];
   321→   321→      comps.forEach((c) => {
   322→   322→        const el = itemRefs.current.get(c.id);
   323→   323→        if (el) max = Math.max(max, (c.y ?? 0) + el.offsetHeight + 24);
   324→   324→      });
   325→   325→      setFreeHeight((prev) => (Math.abs(prev - max) > 2 ? Math.max(700, max) : prev));
   326→   326→    };
   327→   327→    const ro = new ResizeObserver(measure);
   328→   328→    itemRefs.current.forEach((el) => ro.observe(el));
   329→   329→    return () => ro.disconnect();
   330→   330→  }, [isFree, page?.components, zoom]);
   331→   331→
   332→   332→  /* ---------- 渲染 ---------- */
   333→   333→
   334→   334→  if (!page) {
   335→   335→    return (
   336→   336→      <div className="flex flex-1 items-center justify-center text-sm text-zinc-400">
   337→   337→        正在加载画布…
   338→   338→      </div>
   339→   339→    );
   340→   340→  }
   341→   341→
   342→   342→  return (
   343→   343→    <div className="relative flex min-w-0 flex-1 flex-col items-center overflow-auto bg-dot bg-[#eef0f4] py-8 thin-scroll">
   344→   344→      <DragGhost />
   345→   345→
   346→   346→      {/* 画布信息 */}
   347→   347→      <div className="mb-4 flex flex-wrap items-center justify-center gap-2 text-xs text-zinc-500">
   348→   348→        <button
   349→   349→          onClick={toggleLayout}
   350→   350→          className="flex items-center gap-1.5 rounded-full bg-zinc-900 px-3 py-1 font-semibold text-white shadow-sm transition-colors hover:bg-zinc-700"
   351→   351→          title="切换布局模式"
   352→   352→        >
   353→   353→          {isFree ? <Rows3 className="size-3.5" /> : <Move3d className="size-3.5" />}
   354→   354→          {isFree ? '切换为流式布局' : '切换为自由布局'}
   355→   355→        </button>
   356→   356→        <span className="rounded-full bg-white px-3 py-1 shadow-sm">
   357→   357→          {isFree ? '自由布局 · 拖动摆放 · 8 向手柄调大小' : '拖入左侧组件 · 点击选中 · 拖动排序'}
   358→   358→        </span>
   359→   359→        <span className="rounded-full bg-white px-3 py-1 shadow-sm">
   360→   360→          {page.name} · {page.components.length} 个组件
   361→   361→        </span>
   362→   362→      </div>
   363→   363→
   364→   364→      <div style={{ width: 395 * zoom, height: 832 * zoom }}>
   365→   365→        <div style={{ transform: `scale(${zoom})`, transformOrigin: 'top left', width: 395, height: 832 }}>
   366→   366→          <PhoneFrame theme={t} pageBg={page.background}>
   367→   367→            <div
