    22→interface BuilderState {

    36→  selectedIds: string[];

    38→  saving: boolean;
    39→  lastSavedAt: string | null;
    40→  /* 拖拽/批量编辑期间的临时快照（合并为一条历史） */
    41→  batchSnap: HistorySnap | null;

    59→  moveWidget: (id: string, toIndex: number) => void;
    60→  removeWidget: (id: string) => void;
    61→  /** 批量删除（一条历史；联动清理相关连接） */
    62→  removeWidgets: (ids: string[]) => void;
    63→  duplicateWidget: (id: string) => void;
    64→  /** 批量复制：返回新 id 列表（一条历史，整体错开 16px） */
    65→  duplicateWidgets: (ids: string[]) => string[];
    66→  /** 复制组件到剪贴板（Ctrl+C） */
    67→  copyWidget: (id: string) => void;
    68→  /** 粘贴剪贴板组件（Ctrl+V）；自由布局错开 12px，流式追加到末尾；返回是否真的粘贴了 */
    69→  pasteWidget: () => boolean;
    70→  /** 图层置顶/置底/上移一层/下移一层（数组序即层叠序，流式下置顶置底等于移到末尾/开头） */
    71→  reorderWidget: (id: string, pos: 'front' | 'back' | 'up' | 'down') => void;
    72→  updateWidget: (id: string, patch: Partial<WidgetInstance>) => void;
    73→  updateWidgetProps: (id: string, props: Record<string, any>) => void;
    74→  /** 自由布局：更新位置/大小（transient=true 时不记录历史，用于拖拽过程） */
    75→  setWidgetRect: (id: string, rect: Partial<Pick<WidgetInstance, 'x' | 'y' | 'w' | 'h'>>, transient?: boolean) => void;
    76→  /** 多选批量更新矩形（组拖拽用，transient 同上） */
    77→  setWidgetsRects: (map: Record<string, Partial<Pick<WidgetInstance, 'x' | 'y' | 'w' | 'h'>>>, transient?: boolean) => void;
    78→  /** 多选对齐（选择集包围盒）：left/hcenter/right/top/vcenter/bottom */
    79→  alignWidgets: (mode: 'left' | 'hcenter' | 'right' | 'top' | 'vcenter' | 'bottom') => void;
    80→  /** 多选等间距分布（需要画布测量尺寸，sizes 缺省用存储 h/w） */
    81→  distributeWidgets: (axis: 'h' | 'v', sizes?: Record<string, { w?: number; h?: number }>) => void;
    82→  /** 多选统一宽度/高度：以主选中组件（面板当前显示的）为基准；measured 提供实测尺寸（自适应高度组件用） */
    83→  unifyWidgetsSize: (dim: 'w' | 'h', measured?: Record<string, { w?: number; h?: number }>) => void;
    84→  /** 批量编辑：拖拽/缩放开始时调用，结束时 endBatch 合并为一条历史 */
    85→  beginBatch: () => void;
    86→  endBatch: () => void;
    87→  /** 流式 → 自由布局迁移（rects 由画布测量后传入，记录历史） */
    88→  convertToFreeLayout: (rects: Record<string, { x: number; y: number; w: number }>) => void;
    89→  /** 旧数据自动升级为自由布局（不记历史，用户无感知迁移） */
    90→  migrateToFree: (pageId: string, rects?: Record<string, { x: number; y: number; w: number }>) => void;
    91→  select: (id: string | null, additive?: boolean) => void;
    92→  /** 框选/批量设置选中集合（同时更新 primary） */
    93→  selectMany: (ids: string[]) => void;
    94→  /** 清空多选（保留 primary 不变或一并清空） */
    95→  clearMulti: () => void;
    96→  /** 编组：把多选组件绑为一组（返回组 id；单选/含锁定成员时为空操作） */
    97→  groupWidgets: (ids: string[]) => string | null;
    98→  /** 解组：清除这些组件的编组标记（组内仅剩 1 个时自动清掉孤组） */
    99→  ungroupWidgets: (ids: string[]) => void;
   100→  /** 选择整组：若 id 属于某组，返回全组成员 id（含自己） */
   101→  groupMembersOf: (id: string) => string[] | null;
   102→  setCurrentPage: (id: string) => void;
   103→  addPage: () => void;
   104→  duplicatePage: (id: string) => void;
   105→  /** 页面拖拽排序：把页面移到目标下标（数组序即标签序） */
   106→  movePage: (id: string, toIndex: number) => void;
   107→  /** 设为主页：转移 isHome 标记（原主页自动让位） */
   108→  setHomePage: (id: string) => void;
   109→  updatePage: (id: string, patch: Partial<PageData>) => void;
   110→  removePage: (id: string) => void;
   111→  setFlowPos: (id: string, x: number, y: number) => void;
   112→  addConnection: (c: Omit<ConnectionData, 'id' | 'action'>) => void;
   113→  removeConnection: (id: string) => void;
   114→  updateTheme: (patch: Partial<ThemeConfig>) => void;
   115→  undo: () => void;
   116→  redo: () => void;
   117→  save: () => Promise<boolean>;
   118→  setPublishOpen: (b: boolean) => void;
   119→  publish: (name: string, description: string) => Promise<number | null>;
   120→  openPublishPreview: (publishId: string) => Promise<void>;
   121→  exitPublishPreview: () => void;
   122→  exportProject: () => void;
   123→  importProject: (payload: unknown) => Promise<boolean>;
   124→  /** 拉取组件市场列表 */
   125→  loadPresets: () => Promise<void>;
   126→  /** 把选中的组件存为可复用组合；返回保存结果（失败为 null） */
   127→  savePreset: (name: string, icon: string, ids: string[]) => Promise<PresetData | null>;
   128→  deletePreset: (id: string) => Promise<void>;
   129→  /** 插入组合到当前页面（自由布局还原相对位置 / 流式逐个追加）；anchor 给定时以释放点为组合原点 */
   130→  insertPreset: (preset: PresetData, anchor?: { x: number; y: number }) => void;
   131→  /** 把整页存为页面模板（保留绝对坐标 + 背景 + 布局） */
   132→  savePagePreset: (name: string, icon: string, pageId: string) => Promise<PresetData | null>;
   133→  /** 从页面模板创建新页面（重生成组件 id，自动切换到新页） */
   134→  addPageFromTemplate: (preset: PresetData) => void;
   135→}

   137→const MAX_HISTORY = 60;
   138→
   139→export const useBuilder = create<BuilderState>((set, get) => {
   140→  /** 提交一次变更：记录历史 + 标脏 */
   141→  const commit = (fn: (s: { pages: PageData[]; connections: ConnectionData[] }) => Partial<BuilderState>) => {
   142→    const { pages, connections, past } = get();
   143→    const snap: HistorySnap = { pages: deepClone(pages), connections: deepClone(connections) };
   144→    set({
   145→      past: [...past.slice(-MAX_HISTORY), snap],
   146→      future: [],
   147→      dirty: true,
   148→      ...fn({ pages, connections }),
   149→    });
   150→  };
   151→
   152→  const currentPage = () => get().pages.find((p) => p.id === get().currentPageId);
   153→
   154→  return {
   155→    view: 'home',
   156→    projects: [],
   157→    publishes: [],
   158→    homeLoading: false,
   159→    homeLoaded: false,
   160→    project: null,
   161→    pages: [],
   162→    connections: [],
   163→    currentPageId: null,
   164→    selectedWidgetId: null,
   165→    selectedIds: [],
   166→    dirty: false,
   167→    saving: false,
   168→    lastSavedAt: null,
   169→    batchSnap: null,
   170→    publishOpen: false,
   171→    previewSnapshot: null,
   172→    presets: [],
   173→    presetsLoaded: false,
   174→    past: [],
   175→    future: [],
   176→
   177→    async loadHome() {
   178→      set({ homeLoading: true });
   179→      try {
   180→        const [projects, publishes] = await Promise.all([
   181→          fetch('/api/projects').then((r) => r.json()),
   182→          fetch('/api/publishes').then((r) => r.json()),
   183→        ]);
   184→        set({ projects, publishes, homeLoaded: true });
   185→      } finally {
   186→        set({ homeLoading: false });
   187→      }
   188→    },
   189→
   190→    async createProject(name, description, templateId) {
   191→      const res = await fetch('/api/projects', {
   192→        method: 'POST',
   193→        headers: { 'Content-Type': 'application/json' },
   194→        body: JSON.stringify({ name, description, templateId }),
   195→      });
   196→      const { id } = await res.json();
   197→      await get().loadHome();
   198→      await get().openProject(id);
   199→    },
   200→
   201→    async deleteProject(id) {
   202→      await fetch(`/api/projects/${id}`, { method: 'DELETE' });
   203→      if (get().project?.id === id) set({ project: null, pages: [], connections: [], currentPageId: null });
   204→      await get().loadHome();
   205→    },
   206→
   207→    async renameProject(id, name) {
   208→      const { projects } = get();
   209→      set({ projects: projects.map((p) => (p.id === id ? { ...p, name } : p)) });
   210→      if (get().project?.id === id) {
   211→        set({ project: { ...get().project!, name } });
   212→        get().save();
   213→      }
   214→    },
   215→
   216→    async openProject(id) {
   217→      const res = await fetch(`/api/projects/${id}`);
   218→      if (!res.ok) return;
   219→      const data: ProjectData = await res.json();
   220→      const home = data.pages.find((p) => p.isHome);
   221→      set({
   222→        project: data,
   223→        pages: data.pages,
   224→        connections: data.connections,
   225→        currentPageId: home?.id || data.pages[0]?.id || null,
   226→        selectedWidgetId: null,
   227→        selectedIds: [],
   228→        dirty: false,
   229→        saving: false,
   230→        lastSavedAt: null,
   231→        past: [],
   232→        future: [],
   233→        previewSnapshot: null,
   234→        view: 'editor',
   235→      });
   236→    },
   237→
   238→    setView(v) {
   239→      set({ view: v });
   240→      if (v === 'home') {
   241→        get().loadHome();
   242→      }
   243→    },
   244→
   245→    addWidget(type, index, rect) {
   246→      const page = currentPage();
   247→      if (!page) return;
   248→      const def = getWidget(type);
   249→      if (!def) return;
   250→      const w: WidgetInstance = {
   251→        id: uid(),
   252→        type,
   253→        props: deepClone(def.defaultProps),
   254→        width: 'full',
   255→        align: 'left',
   256→        mt: 0,
   257→        mb: 8,
   258→      };
   259→      /* 自由布局：按落点/智能位置放置（不钳制，可摆到首屏以下，画布自动增高） */
   260→      if (page.layout === 'free') {
   261→        const fullW = def.fullBleed ? 375 : 355;
   262→        const maxBottom = page.components.reduce(
   263→          (m, c) => Math.max(m, (c.y ?? 0) + (c.h ?? 64)),
   264→          0
   265→        );
   266→        w.x = rect?.x ?? (def.fullBleed ? 0 : 10);
   267→        w.y = rect?.y ?? maxBottom + 12;
   268→        w.w = rect?.w ?? fullW;
   269→        w.x = Math.max(0, Math.min(w.x ?? 0, 375 - (w.w ?? fullW)));
   270→      }
   271→      commit(({ pages }) => ({
   272→        pages: pages.map((p) =>
   273→          p.id === page.id
   274→            ? {
   275→                ...p,
   276→                components: index === undefined
   277→                  ? [...p.components, w]
   278→                  : [...p.components.slice(0, index), w, ...p.components.slice(index)],
   279→              }
   280→            : p
   281→        ),
   282→      }));
   283→      set({ selectedWidgetId: w.id, selectedIds: [w.id] });
   284→    },
   285→
   286→    moveWidget(id, toIndex) {
   287→      const page = currentPage();
   288→      if (!page) return;
   289→      commit(({ pages }) => ({
   290→        pages: pages.map((p) => {
   291→          if (p.id !== page.id) return p;
   292→          const from = p.components.findIndex((c) => c.id === id);
   293→          if (from < 0) return p;
   294→          const list = [...p.components];
   295→          const [item] = list.splice(from, 1);
   296→          const target = from < toIndex ? toIndex - 1 : toIndex;
   297→          list.splice(Math.max(0, Math.min(target, list.length)), 0, item);
   298→          return { ...p, components: list };
   299→        }),
   300→      }));
   301→    },
   302→
   303→    removeWidget(id) {
   304→      const page = currentPage();
   305→      if (!page) return;
   306→      commit(({ pages, connections }) => ({
   307→        pages: pages.map((p) => (p.id === page.id ? { ...p, components: p.components.filter((c) => c.id !== id) } : p)),
   308→        connections: connections.filter((c) => !(c.fromPageId === page.id && c.fromWidgetId === id)),
   309→      }));
   310→      if (get().selectedWidgetId === id) set({ selectedWidgetId: null });
   311→      set({ selectedIds: get().selectedIds.filter((x) => x !== id) });
   312→    },
   313→
   314→    removeWidgets(ids) {
   315→      const page = currentPage();
   316→      if (!page) return;
   317→      const set2 = new Set(ids);
   318→      commit(({ pages, connections }) => ({
   319→        pages: pages.map((p) => (p.id === page.id ? { ...p, components: p.components.filter((c) => !set2.has(c.id) || c.locked) } : p)),
   320→        connections: connections.filter((c) => !(c.fromPageId === page.id && set2.has(c.fromWidgetId))),
   321→      }));
   322→      const remaining = get().pages.find((p) => p.id === page.id)?.components.map((c) => c.id) ?? [];
   323→      set({
   324→        selectedIds: get().selectedIds.filter((x) => remaining.includes(x)),
   325→        selectedWidgetId: remaining.includes(get().selectedWidgetId ?? '') ? get().selectedWidgetId : null,
   326→      });
   327→    },
   328→
   329→    duplicateWidget(id) {
   330→      const [copyId] = get().duplicateWidgets([id]);
   331→      if (copyId) set({ selectedWidgetId: copyId, selectedIds: [copyId] });
   332→    },
   333→
   334→    duplicateWidgets(ids) {
   335→      const page = currentPage();
   336→      if (!page || ids.length === 0) return [];
   337→      const OFFSET = 16;
   338→      const srcList = page.components;
   339→      const pairs = ids
   340→        .map((id) => srcList.find((c) => c.id === id))
   341→        .filter((c): c is WidgetInstance => !!c && !c.locked)
   342→        .map((src) => {
   343→          const copy: WidgetInstance = { ...deepClone(src), id: uid() };
   344→          if (page.layout === 'free') {
   345→            const cw = copy.w ?? 355;
   346→            copy.x = Math.max(0, Math.min((copy.x ?? 0) + OFFSET, Math.max(0, 375 - cw)));
   347→            copy.y = (copy.y ?? 0) + OFFSET;
   348→          }
   349→          return { srcId: src.id, clone: copy };
   350→        });
   351→      if (pairs.length === 0) return [];
   352→      const created = pairs.map((p) => p.clone.id);
   353→      commit(({ pages }) => ({
   354→        pages: pages.map((p) => {
   355→          if (p.id !== page.id) return p;
   356→          const list = [...p.components];
   357→          /* 倒序插入：插在各自源组件后面，保持图层相对顺序 */
   358→          for (let i = pairs.length - 1; i >= 0; i--) {
   359→            const idx = list.findIndex((c) => c.id === pairs[i].srcId);
   360→            list.splice(idx >= 0 ? idx + 1 : list.length, 0, pairs[i].clone);
   361→          }
   362→          return { ...p, components: list };
   363→        }),
   364→      }));
   365→      set({ selectedWidgetId: created[created.length - 1], selectedIds: created });
   366→      return created;
   367→    },
   368→
   369→    copyWidget(id) {
   370→      const page = currentPage();
   371→      if (!page) return;
   372→      const src = page.components.find((c) => c.id === id);
   373→      if (!src) return;
   374→      widgetClipboard = { type: src.type, props: deepClone(src.props), width: src.width, align: src.align };
   375→    },
   376→
   377→    pasteWidget() {
   378→      if (!widgetClipboard) return false;
   379→      const page = currentPage();
   380→      if (!page) return false;
   381→      const def = getWidget(widgetClipboard.type);
   382→      if (!def) return false;
   383→      const w: WidgetInstance = {
   384→        id: uid(),
   385→        type: widgetClipboard.type,
   386→        props: deepClone(widgetClipboard.props),
   387→        width: widgetClipboard.width ?? 'full',
   388→        align: widgetClipboard.align ?? 'left',
   389→        mt: 0,
   390→        mb: 8,
   391→      };
   392→      if (page.layout === 'free') {
   393→        const fullW = def.fullBleed ? 375 : 355;
   394→        const maxBottom = page.components.reduce(
   395→          (m, c) => Math.max(m, (c.y ?? 0) + (c.h ?? 64)),
   396→          0
   397→        );
   398→        w.x = def.fullBleed ? 0 : 10;
   399→        w.y = maxBottom + 12;
   400→        w.w = fullW;
   401→      }
   402→      commit(({ pages }) => ({
   403→        pages: pages.map((p) => (p.id === page.id ? { ...p, components: [...p.components, w] } : p)),
   404→      }));
   405→      set({ selectedWidgetId: w.id, selectedIds: [w.id] });
   406→      return true;
   407→    },
   408→
   409→    reorderWidget(id, pos) {
   410→      const page = currentPage();
   411→      if (!page) return;
   412→      commit(({ pages }) => ({
   413→        pages: pages.map((p) => {
   414→          if (p.id !== page.id) return p;
   415→          const from = p.components.findIndex((c) => c.id === id);
   416→          if (from < 0) return p;
   417→          const list = [...p.components];
   418→          const [item] = list.splice(from, 1);
   419→          if (pos === 'front') list.push(item);
   420→          else if (pos === 'back') list.unshift(item);
   421→          else if (pos === 'up') list.splice(Math.min(from + 1, list.length), 0, item);
   422→          else list.splice(Math.max(from - 1, 0), 0, item);
   423→          return { ...p, components: list };
   424→        }),
   425→      }));
   426→    },
   427→
   428→    updateWidget(id, patch) {
   429→      const page = currentPage();
   430→      if (!page) return;
   431→      commit(({ pages }) => ({
   432→        pages: pages.map((p) =>
   433→          p.id === page.id
   434→            ? { ...p, components: p.components.map((c) => (c.id === id ? { ...c, ...patch } : c)) }
   435→            : p
   436→        ),
   437→      }));
   438→    },
   439→
   440→    updateWidgetProps(id, props) {
   441→      const page = currentPage();
   442→      if (!page) return;
   443→      commit(({ pages }) => ({
   444→        pages: pages.map((p) =>
   445→          p.id === page.id
   446→            ? { ...p, components: p.components.map((c) => (c.id === id ? { ...c, props: { ...c.props, ...props } } : c)) }
   447→            : p
   448→        ),
   449→      }));
   450→    },
   451→
   452→    setWidgetRect(id, rect, transient = false) {
   453→      const page = currentPage();
   454→      if (!page) return;
   455→      const patchPages = (pages: PageData[]) =>
   456→        pages.map((p) =>
   457→          p.id === page.id
   458→            ? { ...p, components: p.components.map((c) => (c.id === id ? { ...c, ...rect } : c)) }
   459→            : p
   460→        );
   461→      if (transient) {
   462→        set({ pages: patchPages(get().pages), dirty: true });
   463→      } else {
   464→        commit(({ pages }) => ({ pages: patchPages(pages) }));
   465→      }
   466→    },
   467→
   468→    setWidgetsRects(map, transient = false) {
   469→      const page = currentPage();
   470→      if (!page) return;
   471→      const patchPages = (pages: PageData[]) =>
   472→        pages.map((p) =>
   473→          p.id === page.id
   474→            ? { ...p, components: p.components.map((c) => (map[c.id] ? { ...c, ...map[c.id] } : c)) }
   475→            : p

   484→    alignWidgets(mode) {
   485→      const page = currentPage();
   486→      if (!page || page.layout !== 'free') return;
   487→      const { selectedIds } = get();

   502→      const map: Record<string, Partial<Pick<WidgetInstance, 'x' | 'y'>>> = {};

   504→        if (mode === 'left') map[r.id] = { x: minX };
   505→        else if (mode === 'right') map[r.id] = { x: maxX - r.w };
   506→        else if (mode === 'hcenter') map[r.id] = { x: Math.round((minX + maxX) / 2 - r.w / 2) };
   507→        else if (mode === 'top') map[r.id] = { y: minY };
   508→        else if (mode === 'bottom') map[r.id] = { y: maxY - r.h };
   509→        else if (mode === 'vcenter') map[r.id] = { y: Math.round((minY + maxY) / 2 - r.h / 2) };

   514→    distributeWidgets(axis, sizes) {
   515→      const page = currentPage();
   516→      if (!page || page.layout !== 'free') return;
   517→      const { selectedIds } = get();
   518→      if (selectedIds.length < 3) return;
   519→      const comps = page.components.filter((c) => selectedIds.includes(c.id) && !c.locked);
   520→      if (comps.length < 3) return;
   521→      const items = comps
   522→        .map((c) => ({
   523→          id: c.id,
   524→          x: c.x ?? 0,
   525→          y: c.y ?? 0,
   526→          w: sizes?.[c.id]?.w ?? c.w ?? 355,
   527→          h: sizes?.[c.id]?.h ?? c.h ?? 56,
   528→        }))
   529→        .sort((a, b) => (axis === 'h' ? a.x - b.x : a.y - b.y));
   530→      const first = items[0];
   531→      const last = items[items.length - 1];
   532→      if (axis === 'h') {
   533→        const span = last.x + last.w - first.x;
   534→        const totalW = items.reduce((s, r) => s + r.w, 0);
   535→        const gap = (span - totalW) / (items.length - 1);
   536→        if (!isFinite(gap) || gap < 0) return;
   537→        const map: Record<string, Partial<Pick<WidgetInstance, 'x' | 'y'>>> = {};
   538→        let cursor = first.x;
   539→        for (const r of items) {
   540→          if (r.id !== first.id && r.id !== last.id) map[r.id] = { x: Math.round(cursor) };
   541→          cursor += r.w + gap;
   542→        }
   543→        get().setWidgetsRects(map);
   544→      } else {
   545→        const span = last.y + last.h - first.y;
   546→        const totalH = items.reduce((s, r) => s + r.h, 0);
   547→        const gap = (span - totalH) / (items.length - 1);
   548→        if (!isFinite(gap) || gap < 0) return;
   549→        const map: Record<string, Partial<Pick<WidgetInstance, 'x' | 'y'>>> = {};
   550→        let cursor = first.y;
   551→        for (const r of items) {
   552→          if (r.id !== first.id && r.id !== last.id) map[r.id] = { y: Math.round(cursor) };
   553→          cursor += r.h + gap;
   554→        }
   555→        get().setWidgetsRects(map);
   556→      }

  1139→/** 自动保存：脏状态 1.2s 后保存 */
  1140→export function bindAutosave() {
  1141→  let timer: ReturnType<typeof setTimeout> | null = null;
  1142→  let stopped = false;
  1143→  const unsub = useBuilder.subscribe((state) => {
  1144→    if (stopped) return;
  1145→    if (state.dirty && state.view !== 'home') {
  1146→      if (timer) clearTimeout(timer);
  1147→      timer = setTimeout(() => {
  1148→        useBuilder.getState().save();
  1149→      }, 1200);
  1150→    }
  1151→  });
  1152→  return () => {
  1153→    stopped = true;
  1154→    if (timer) clearTimeout(timer);
  1155→    unsub();
  1156→  };
  1157→}
