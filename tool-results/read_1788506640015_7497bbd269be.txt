   104→  /* ---------- 坐标与落点计算 ---------- */
   105→
   106→  const toLocal = useCallback((clientX: number, clientY: number) => {
   107→    const el = contentRef.current;
   108→    const rect = el?.getBoundingClientRect();
   109→    if (!el || !rect) return { x: 0, y: 0 };
   110→    return {
   111→      x: (clientX - rect.left) / zoomRef.current,
   112→      y: (clientY - rect.top) / zoomRef.current + el.scrollTop,
   113→    };
   114→  }, []);
   115→
   116→  const insideContent = useCallback((clientX: number, clientY: number) => {
   117→    const rect = contentRef.current?.getBoundingClientRect();
   118→    if (!rect) return false;
   119→    const pad = 8;
   120→    return (
   121→      clientX >= rect.left - pad && clientX <= rect.right + pad &&
   122→      clientY >= rect.top - pad && clientY <= rect.bottom + pad
   123→    );
   124→  }, []);
   125→
   126→  const computeDrop = useCallback((clientY: number) => {
   127→    const b = useBuilder.getState();
   128→    const list = b.pages.find((p) => p.id === b.currentPageId)?.components ?? [];
   129→    const origin = contentRef.current?.getBoundingClientRect();
   130→    const scrollTop = contentRef.current?.scrollTop ?? 0;
   131→    for (let i = 0; i < list.length; i++) {
   132→      const rect = itemRefs.current.get(list[i].id)?.getBoundingClientRect();
   133→      if (!rect) continue;
   134→      if (clientY < rect.top + rect.height / 2) {
   135→        return { index: i, y: origin ? (rect.top - origin.top) / zoomRef.current + scrollTop : 0 };
   136→      }
   137→    }
   138→    const last = list.length > 0 ? itemRefs.current.get(list[list.length - 1].id)?.getBoundingClientRect() : null;
   139→    return { index: list.length, y: origin && last ? (last.bottom - origin.top) / zoomRef.current + scrollTop : 0 };
   140→  }, []);
   141→
   142→  /** 自由布局吸附：其它组件边缘/中心 + 屏幕边缘/水平中心 + 4px 网格 */
   143→  const snapPos = useCallback((x: number, y: number, movingId: string | null, w: number, h: number) => {
   144→    const b = useBuilder.getState();
   145→    const comps = b.pages.find((p) => p.id === b.currentPageId)?.components ?? [];
   146→    const T = 6;
   147→    const vCands: { p: number; line: number }[] = [
   148→      { p: 0, line: 0 },
   149→      { p: (SCREEN_W - w) / 2, line: SCREEN_W / 2 },
   150→      { p: SCREEN_W - w, line: SCREEN_W },
   151→    ];
   152→    const hCands: { p: number; line: number }[] = [{ p: 0, line: 0 }];
   153→    comps.forEach((c) => {
   154→      if (c.id === movingId) return;
   155→      const cx = c.x ?? 0, cy = c.y ?? 0, cw = c.w ?? 355, ch = c.h ?? 56;
   156→      vCands.push({ p: cx, line: cx }, { p: cx + cw / 2 - w / 2, line: cx + cw / 2 }, { p: cx + cw - w, line: cx + cw });
   157→      hCands.push({ p: cy, line: cy }, { p: cy + ch / 2 - h / 2, line: cy + ch / 2 }, { p: cy + ch - h, line: cy + ch }, { p: cy + ch, line: cy + ch });
   158→    });
   159→    let nx = x, ny = y;
   160→    const g: { v: number[]; h: number[] } = { v: [], h: [] };
   161→    for (const c of vCands) {
   162→      if (Math.abs(x - c.p) <= T) { nx = Math.max(0, Math.min(c.p, SCREEN_W - w)); g.v.push(c.line); break; }
   163→    }
   164→    for (const c of hCands) {
   165→      if (Math.abs(y - c.p) <= T) { ny = Math.max(0, c.p); g.h.push(c.line); break; }
   166→    }
   167→    if (g.v.length === 0) nx = Math.round(x / SNAP) * SNAP;
   168→    if (g.h.length === 0) ny = Math.round(y / SNAP) * SNAP;
   169→    return { x: Math.max(0, Math.min(nx, SCREEN_W - w)), y: Math.max(0, ny), guides: g };
   170→  }, []);
   171→
   172→  /* ---------- 拖拽过程 ---------- */
   173→
   174→  const handleDragMove = useCallback((clientX: number, clientY: number) => {
   175→    const st = useDnd.getState();
   176→
   177→    /* 缩放手柄拖拽中 */
   178→    if (resizeRef.current) {
   179→      const r = resizeRef.current;
   180→      setPosBadge(null);
   181→      const dx = (clientX - r.startX) / zoomRef.current;
   182→      const dy = (clientY - r.startY) / zoomRef.current;
   183→      let { x, y, w, h } = r.rect;
   184→      if (r.dir.includes('e')) w = r.rect.w + dx;
   185→      if (r.dir.includes('s')) h = r.rect.h + dy;
   186→      if (r.dir.includes('w')) { w = r.rect.w - dx; x = r.rect.x + dx; }
   187→      if (r.dir.includes('n')) { h = r.rect.h - dy; y = r.rect.y + dy; }
   188→      if (dirClampWest(r.dir) && x < 0) { w = r.rect.w + r.rect.x; x = 0; }
   189→      if (dirClampNorth(r.dir) && y < 0) { h = r.rect.h + r.rect.y; y = 0; }
   190→      w = Math.max(MIN_W, w);
   191→      h = Math.max(MIN_H, h);
   192→      if (x + w > SCREEN_W) w = SCREEN_W - x;
   193→      x = Math.round(x / SNAP) * SNAP;
   194→      y = Math.round(Math.max(0, y) / SNAP) * SNAP;
   195→      w = Math.round(w / SNAP) * SNAP;
   196→      h = Math.round(h / SNAP) * SNAP;
   197→      useBuilder.getState().setWidgetRect(r.id, { x, y, w, h }, true);
   198→      setSizeBadge(`${w} × ${h}`);
   199→      return;
   200→    }
   201→
   202→    if (!insideContent(clientX, clientY)) {
   203→      setIndicator(null);
   204→      setFreePreview(null);
   205→      freePreviewRef.current = null;
   206→      setGuides(EMPTY_GUIDES);
   207→      setPosBadge(null);
   208→      return;
   209→    }
   210→
   211→    if (isFreeRef.current) {
   212→      setIndicator(null);
   213→      setPosBadge(null);
   214→      if (st.kind === 'marquee') {
   215→        /* 框选：更新矩形 + 实时命中 */
   216→        const { x, y } = toLocal(clientX, clientY);
   217→        const rect = { x0: marqueeStartRef.current.x, y0: marqueeStartRef.current.y, x1: x, y1: y };
   218→        setMarquee(rect);
   219→        const bx0 = Math.min(rect.x0, rect.x1), bx1 = Math.max(rect.x0, rect.x1);
   220→        const by0 = Math.min(rect.y0, rect.y1), by1 = Math.max(rect.y0, rect.y1);
   221→        const b = useBuilder.getState();
   222→        const hits: string[] = [];
   223→        b.pages
   224→          .find((p) => p.id === b.currentPageId)
   225→          ?.components.forEach((c) => {
   226→            if (c.hidden || c.locked) return;
   227→            const el = itemRefs.current.get(c.id);
   228→            const cw = c.w ?? el?.offsetWidth ?? 355;
   229→            const ch = c.h ?? el?.offsetHeight ?? 56;
   230→            const cx = c.x ?? 0, cy = c.y ?? 0;
   231→            if (cx < bx1 && cx + cw > bx0 && cy < by1 && cy + ch > by0) hits.push(c.id);
   232→          });
   233→        marqueeHitRef.current = hits;
   234→        if (hits.length > 0) b.selectMany(hits);
   235→        else b.select(null);
   236→        return;
   237→      }
   238→      if (st.kind === 'new') {
   239→        const def = getWidget(st.widgetType ?? '');
   240→        const w0 = def?.fullBleed ? SCREEN_W : 355;
   241→        const { x, y } = toLocal(clientX, clientY);
   242→        const snapped = snapPos(x - w0 / 2, y - 24, null, w0, 56);
   243→        const preview = { x: snapped.x, y: snapped.y, w: w0 };
   244→        freePreviewRef.current = preview;
   245→        setFreePreview(preview);
   246→        setGuides(snapped.guides);
   247→      } else if (st.kind === 'move') {
   248→        const id = st.moveId!;
   249→        const b = useBuilder.getState();
   250→        const comp = b.pages
   251→          .find((p) => p.id === b.currentPageId)
   252→          ?.components.find((c) => c.id === id);
   253→        if (!comp) return;
   254→        if (!dragMetaRef.current.batch) {
   255→          b.beginBatch();
   256→          dragMetaRef.current.batch = true;
   257→        }
   258→        const { x, y } = toLocal(clientX, clientY);
   259→        const gs = groupStartRef.current;
   260→        if (gs && Object.keys(gs.map).length > 1) {
   261→          /* 组拖拽：所有选中组件整体平移（4px 网格吸附，不做边缘吸附） */
   262→          const dx = x - gs.ox;
   263→          const dy = y - gs.oy;
   264→          const map: Record<string, { x: number; y: number }> = {};
   265→          for (const [wid, s0] of Object.entries(gs.map)) {
   266→            const cw = useBuilder
   267→              .getState()
   268→              .pages.find((p) => p.id === b.currentPageId)
   269→              ?.components.find((c) => c.id === wid)?.w ?? 355;
   270→            map[wid] = {
   271→              x: Math.max(0, Math.min(Math.round((s0.x + dx) / SNAP) * SNAP, SCREEN_W - cw)),
   272→              y: Math.max(0, Math.round((s0.y + dy) / SNAP) * SNAP),
   273→            };
   274→          }
   275→          b.setWidgetsRects(map, true);
   276→          setPosBadge(`${Math.round(dx)} · ${Math.round(dy)}`);
   277→          setGuides(EMPTY_GUIDES);
   278→          setFreePreview(null);
   279→          freePreviewRef.current = null;
   280→          return;
   281→        }
   282→        const cw = comp.w ?? 355;
   283→        const ch = comp.h ?? 56;
   284→        const snapped = snapPos(x - dragMetaRef.current.offX, y - dragMetaRef.current.offY, id, cw, ch);
   285→        b.setWidgetRect(id, { x: snapped.x, y: snapped.y }, true);
   286→        setPosBadge(`${Math.round(snapped.x)} · ${Math.round(snapped.y)}`);
   287→        setGuides(snapped.guides);
   288→        setFreePreview(null);
   289→        freePreviewRef.current = null;
   290→      }
   291→    } else {
   292→      setFreePreview(null);
   293→      freePreviewRef.current = null;
   294→      setGuides(EMPTY_GUIDES);
   295→      setPosBadge(null);
   296→      setIndicator(computeDrop(clientY));
   297→      /* 流式拖动：浮出提示（结束后再保留 6 秒，方便点击切换） */
   298→      if (st.started && st.kind === 'move') {
   299→        if (flowHintTimer.current) clearTimeout(flowHintTimer.current);
   300→        setFlowDragHint(true);
   301→      }
   302→    }
   303→  }, [computeDrop, insideContent, snapPos, toLocal]);
   304→
   305→  const handleUp = useCallback((clientX: number, clientY: number) => {
   306→    const st = useDnd.getState();
   307→    const b = useBuilder.getState();
   308→    const pageNow = b.pages.find((p) => p.id === b.currentPageId);
   309→    const freeNow = pageNow?.layout === 'free';
   310→
   311→    /* 框选结束：命中集已在移动中实时写入 selectedIds；全部落空则清空。
   312→       同时抑制紧随的 click，避免容器 onClick 把框选结果清空 */
   313→    if (st.kind === 'marquee') {
   314→      if (st.started) {
   315→        suppressClickRef.current = true;
   316→        useDnd.setState({ suppressNextClick: true });
   317→        setTimeout(() => { suppressClickRef.current = false; }, 90);
   318→      }
   319→      if (marqueeHitRef.current.length === 0) b.select(null);
   320→      setMarquee(null);
   321→      marqueeHitRef.current = [];
   322→      useDnd.getState().end();
   323→      return;
   324→    }
   325→
   326→    /* 缩放结束：合并历史 */
   327→    if (resizeRef.current) {
   328→      if (dragMetaRef.current.batch) b.endBatch();
   329→      resizeRef.current = null;
   330→      setSizeBadge(null);
   331→      dragMetaRef.current = { offX: 0, offY: 0, batch: false };
   332→      useDnd.getState().end();
   333→      return;
   334→    }
   335→
   336→    if (st.started) {
   337→      suppressClickRef.current = true;
   338→      useDnd.setState({ suppressNextClick: true });
   339→      setTimeout(() => { suppressClickRef.current = false; }, 90);
   340→    }
   341→
   342→    if (st.started && st.kind === 'new' && st.widgetType) {
   343→      if (insideContent(clientX, clientY)) {
   344→        if (freeNow) {
   345→          const { x, y } = toLocal(clientX, clientY);
   346→          const def = getWidget(st.widgetType);
   347→          const w0 = def?.fullBleed ? SCREEN_W : 355;
   348→          const fp = freePreviewRef.current;
   349→          const rx = fp?.x ?? Math.max(0, Math.round((x - w0 / 2) / SNAP) * SNAP);
   350→          const ry = fp?.y ?? Math.max(0, Math.round((y - 24) / SNAP) * SNAP);
   351→          b.addWidget(st.widgetType, undefined, { x: rx, y: ry, w: w0 });
   352→        } else {
   353→          b.addWidget(st.widgetType, computeDrop(clientY).index);
   354→        }
   355→      }
   356→    } else if (st.started && st.kind === 'move' && st.moveId) {
   357→      if (freeNow) {
   358→        if (dragMetaRef.current.batch) b.endBatch();
   359→      } else if (insideContent(clientX, clientY)) {
   360→        /* 仅画布内松手才排序；拖出画布松手视为取消 */
   361→        b.moveWidget(st.moveId, computeDrop(clientY).index);
   362→      }
   363→    }
   364→
   365→    dragMetaRef.current = { offX: 0, offY: 0, batch: false };
   366→    groupStartRef.current = null;
   367→    setIndicator(null);
   368→    setFreePreview(null);
   369→    freePreviewRef.current = null;
   370→    setGuides(EMPTY_GUIDES);
   371→    setSizeBadge(null);
   372→    setPosBadge(null);
   373→    /* 流式拖动结束后：提示再保留 6 秒，之后自动隐藏 */
   374→    if (flowHintTimer.current) clearTimeout(flowHintTimer.current);
   375→    flowHintTimer.current = setTimeout(() => setFlowDragHint(false), 6000);
   376→    useDnd.getState().end();
   377→  }, [computeDrop, insideContent, toLocal]);
   378→
   379→  /* 拖拽期间挂全局指针监听 */
   380→  useEffect(() => {
   381→    if (!dndActive) return;
   382→    const onMove = (e: PointerEvent) => {
   383→      useDnd.getState().move(e.clientX, e.clientY);
   384→      if (!useDnd.getState().started) return;
   385→      handleDragMove(e.clientX, e.clientY);
   386→    };
   387→    const onUp = (e: PointerEvent) => handleUp(e.clientX, e.clientY);
   388→    window.addEventListener('pointermove', onMove);
   389→    window.addEventListener('pointerup', onUp, true);
   390→    window.addEventListener('pointercancel', onUp, true);
   391→    return () => {
   392→      window.removeEventListener('pointermove', onMove);
   393→      window.removeEventListener('pointerup', onUp, true);
   394→      window.removeEventListener('pointercancel', onUp, true);
   395→    };
   396→  }, [dndActive, handleDragMove, handleUp]);
   397→
   398→  /* ---------- 布局切换 ---------- */
   399→
   400→  /** 启用自由布局：已有坐标直接切；缺坐标（旧数据）先按当前渲染堆叠生成坐标 */
   401→  const enableFree = useCallback(() => {
   402→    const b = useBuilder.getState();
   403→    const pg = b.pages.find((p) => p.id === b.currentPageId);
   404→    if (!pg || pg.layout === 'free') return;
   405→    if (!missingFreeCoords(pg.components)) {
   406→      b.updatePage(pg.id, { layout: 'free' });
   407→      setFlowDragHint(false);
   408→      return;
   409→    }
   410→    const rects: Record<string, { x: number; y: number; w: number }> = {};

   864→                                  resizeRef.current = {
   865→                                    id: w.id,
   866→                                    dir,
   867→                                    startX: e.clientX,
   868→                                    startY: e.clientY,

   876→                                  dragMetaRef.current.batch = false;
   877→                                  useBuilder.getState().beginBatch();
   878→                                  dragMetaRef.current.batch = true;
   879→                                  useDnd.getState().begin({ kind: 'move', moveId: w.id, px: e.clientX, py: e.clientY });

   885→                              />
   886→                            ))}
   887→                          </>
   888→                        )}
   889→                      </div>
   890→                    );
   891→                  })}
   892→                </div>
   893→              ) : (
   894→                /* ---------- 流式布局：垂直排列 + 拖拽排序 ---------- */
   895→                <>
   896→                  {page.components.filter((w: WidgetInstance) => !w.hidden).map((w: WidgetInstance, idx: number) => {
   897→                    const def = getWidget(w.type);
   898→                    const selected = selectedWidgetId === w.id;
   899→                    return (
   900→                      <div
   901→                        key={w.id}
   902→                        ref={(el) => {
   903→                          if (el) itemRefs.current.set(w.id, el);
   904→                          else itemRefs.current.delete(w.id);
   905→                        }}
   906→                        onPointerDown={(e) => {
   907→                          if (e.button !== 0) return;
   908→                          e.stopPropagation();
   909→                          if (w.locked) { select(w.id); return; } /* 锁定：只选中，不启动排序拖拽 */
   910→                          dragMetaRef.current = { offX: 0, offY: 0, batch: false };
   911→                          useDnd.getState().begin({ kind: 'move', moveId: w.id, px: e.clientX, py: e.clientY });
   912→                        }}
   913→                        onClick={(e) => {
   914→                          e.stopPropagation();
   915→                          if (suppressClickRef.current) return;
   916→                          select(w.id);
   917→                        }}
   918→                        className={`group relative cursor-grab touch-none rounded-xl active:cursor-grabbing ${
   919→                          selected ? 'ring-2' : 'hover:ring-1'
   920→                        }`}
   921→                        style={{
   922→                          '--tw-ring-color': selected ? t.primary : 'rgba(0,0,0,0.12)',
   923→                        } as React.CSSProperties}
   924→                      >
   925→                        <WidgetRenderer w={w} />
   926→
   927→                        {/* 悬浮操作条 */}
   928→                        {!w.locked && (
   929→                        <div
   930→                          className={`absolute -top-2.5 right-2 z-30 flex items-center gap-0.5 rounded-lg bg-zinc-900/90 p-0.5 shadow-lg transition-opacity ${
   931→                            selected ? 'opacity-100' : 'opacity-0 group-hover:opacity-100'
   932→                          }`}
   933→                          onClick={(e) => e.stopPropagation()}
   934→                          onPointerDown={(e) => e.stopPropagation()}
   935→                        >
   936→                          <ItemBtn
   937→                            title="上移"
   938→                            disabled={idx === 0}
   939→                            onClick={() => useBuilder.getState().moveWidget(w.id, idx - 1)}
   940→                          >
   941→                            <ChevronUp className="size-3.5" />
   942→                          </ItemBtn>
   943→                          <ItemBtn
   944→                            title="下移"
   945→                            disabled={idx === page.components.length - 1}
   946→                            onClick={() => useBuilder.getState().moveWidget(w.id, idx + 2)}
   947→                          >
   948→                            <ChevronDown className="size-3.5" />
   949→                          </ItemBtn>
   950→                          <ItemBtn title="复制" onClick={() => duplicateWidget(w.id)}>
   951→                            <Copy className="size-3.5" />
   952→                          </ItemBtn>
   953→                          <ItemBtn title="删除" danger onClick={() => removeWidget(w.id)}>
   954→                            <Trash2 className="size-3.5" />
   955→                          </ItemBtn>
   956→                        </div>
   957→                        )}
   958→                        {w.locked && selected && (
   959→                          <span className="absolute -right-1 -top-6 z-30 flex items-center gap-1 rounded-md bg-zinc-900 px-1.5 py-0.5 text-[10px] font-semibold text-white shadow">
   960→                            <Lock className="size-2.5" /> 已锁定
   961→                          </span>
   962→                        )}
   963→
   964→                        {/* 类型徽标 */}
   965→                        {selected && def && (
   966→                          <span
   967→                            className="pointer-events-none absolute -left-1 -top-6 z-30 rounded-md px-1.5 py-0.5 text-[10px] font-semibold text-white shadow"
   968→                            style={{ background: t.primary }}
   969→                          >
   970→                            {def.name}
   971→                          </span>
   972→                        )}
   973→                      </div>
   974→                    );
   975→                  })}
   976→                </>
   977→              )}
   978→
   979→              {/* 底部留白便于拖入 */}
   980→              <div className="h-16" />
   981→            </div>
   982→          </PhoneFrame>
   983→        </div>
   984→      </div>
   985→
   986→      {/* 多选批量操作工具条（自由布局） */}
   987→      {isFreeMode && selectedIds.length > 1 && (
   988→        <div className="fixed left-1/2 top-16 z-50 -translate-x-1/2">
   989→          <div className="flex items-center gap-0.5 rounded-2xl border border-zinc-200 bg-white/95 p-1.5 shadow-2xl backdrop-blur animate-in fade-in slide-in-from-top-2 duration-200">
   990→            <span
   991→              className="mr-1 flex items-center gap-1 rounded-xl px-2.5 py-1.5 text-xs font-bold text-white"
   992→              style={{ background: t.primary }}
   993→            >
   994→              <Scan className="size-3.5" />
   995→              已选 {selectedIds.length}
   996→            </span>
   997→            <ToolbarBtn title="左对齐" onClick={() => useBuilder.getState().alignWidgets('left')}>
   998→              <AlignStartVertical className="size-4" />
   999→            </ToolbarBtn>
  1000→            <ToolbarBtn title="水平居中" onClick={() => useBuilder.getState().alignWidgets('hcenter')}>
  1001→              <AlignCenterVertical className="size-4" />
  1002→            </ToolbarBtn>
  1003→            <ToolbarBtn title="右对齐" onClick={() => useBuilder.getState().alignWidgets('right')}>
  1004→              <AlignEndVertical className="size-4" />
  1005→            </ToolbarBtn>
  1006→            <span className="mx-0.5 h-5 w-px bg-zinc-200" />
  1007→            <ToolbarBtn title="顶对齐" onClick={() => useBuilder.getState().alignWidgets('top')}>
  1008→              <AlignStartHorizontal className="size-4" />
  1009→            </ToolbarBtn>
  1010→            <ToolbarBtn title="垂直居中" onClick={() => useBuilder.getState().alignWidgets('vcenter')}>
  1011→              <AlignCenterHorizontal className="size-4" />

  1016→            <span className="mx-0.5 h-5 w-px bg-zinc-200" />
  1017→            <ToolbarBtn
  1018→              title="横向等间距分布"
  1019→              disabled={selectedIds.length < 3}
  1020→              onClick={() => {
  1021→                const sizes: Record<string, { w?: number; h?: number }> = {};
  1022→                selectedIds.forEach((id) => {
  1023→                  const el = itemRefs.current.get(id);
  1024→                  if (el) sizes[id] = { w: el.offsetWidth, h: el.offsetHeight };
  1025→                });
  1026→                useBuilder.getState().distributeWidgets('h', sizes);
  1027→              }}
  1028→            >
  1029→              <AlignHorizontalSpaceAround className="size-4" />
  1030→            </ToolbarBtn>
  1031→            <ToolbarBtn
  1032→              title="纵向等间距分布"
  1033→              disabled={selectedIds.length < 3}
  1034→              onClick={() => {
  1035→                const sizes: Record<string, { w?: number; h?: number }> = {};
  1036→                selectedIds.forEach((id) => {
  1037→                  const el = itemRefs.current.get(id);
  1038→                  if (el) sizes[id] = { w: el.offsetWidth, h: el.offsetHeight };
  1039→                });
  1040→                useBuilder.getState().distributeWidgets('v', sizes);
  1041→              }}
  1042→            >
  1043→              <AlignVerticalSpaceAround className="size-4" />
  1044→            </ToolbarBtn>
  1045→            <span className="mx-0.5 h-5 w-px bg-zinc-200" />
  1046→            <ToolbarBtn
  1047→              title="批量复制"
  1048→              onClick={() => {
  1049→                const n = useBuilder.getState().duplicateWidgets(selectedIds).length;
  1050→                if (n > 0) toast(`已复制 ${n} 个组件`);
  1051→              }}
  1052→            >
  1053→              <Copy className="size-4" />
  1054→            </ToolbarBtn>
  1055→            <ToolbarBtn
  1056→              title="批量删除"
  1057→              danger
  1058→              onClick={() => {
  1059→                const n = selectedIds.length;
  1060→                useBuilder.getState().removeWidgets(selectedIds);
  1061→                toast(`已删除 ${n} 个组件`);
  1062→              }}
  1063→            >
  1064→              <Trash2 className="size-4" />
  1065→            </ToolbarBtn>
  1066→          </div>
  1067→        </div>
  1068→      )}
  1069→
  1070→      {/* 缩放控制 */}
  1071→      <div className="fixed bottom-6 left-1/2 z-40 flex -translate-x-1/2 items-center gap-1 rounded-full border bg-white/95 px-2 py-1.5 shadow-lg backdrop-blur">
  1072→        <button
  1073→          className="flex size-7 items-center justify-center rounded-full hover:bg-zinc-100 disabled:opacity-30"
  1074→          onClick={() => setZoom((z) => Math.max(0.5, +(z - 0.05).toFixed(2)))}
  1075→          disabled={zoom <= 0.5}
  1076→          aria-label="缩小"
  1077→        >
  1078→          <ZoomOut className="size-4" />
  1079→        </button>
  1080→        <span className="w-12 text-center text-xs font-semibold tabular-nums">{Math.round(zoom * 100)}%</span>
  1081→        <button
  1082→          className="flex size-7 items-center justify-center rounded-full hover:bg-zinc-100 disabled:opacity-30"
  1083→          onClick={() => setZoom((z) => Math.min(1.2, +(z + 0.05).toFixed(2)))}
  1084→          disabled={zoom >= 1.2}
  1085→          aria-label="放大"
  1086→        >
  1087→          <ZoomIn className="size-4" />

  1089→        <span className="mx-1 h-4 w-px bg-zinc-200" />
  1090→        <span className="flex h-7 items-center gap-1 px-2 text-xs text-zinc-500">
  1091→          <Scan className="size-3.5" /> {page.components.filter((c) => !c.hidden).length} 组件
  1092→        </span>
  1093→      </div>
  1094→    </div>
  1095→  );
  1096→}
