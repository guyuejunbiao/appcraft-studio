    35→          animate={{ x: active === 'right' ? '100%' : '0%' }}
    36→          transition={{ type: 'spring', stiffness: 520, damping: 38 }}
    37→        />
    38→        {(['left', 'right'] as const).map((k) => {
    39→          const isActive = active === k;
    40→          return (
    41→            <button
    42→              key={k}
    43→              type="button"
    44→              onClick={(e) => {
    45→                e.stopPropagation();

    51→              {k === 'left' ? props.left : props.right}
    52→            </button>
    53→          );
    54→        })}
    55→      </div>

   118→/* ------------------------------------------------------------------ */
   119→export function SmsInputInteractive({ props }: InteractiveCtx) {
   120→  const [code, setCode] = useState('');
   121→  const [left, setLeft] = useState(0);
   122→  const timers = useRef<ReturnType<typeof setTimeout>[]>([]);
   123→
   124→  const counting = left > 0;
   125→  useEffect(() => {
   126→    if (!counting) return;
   127→    const iv = setInterval(() => setLeft((v) => Math.max(0, v - 1)), 1000);
   128→    return () => clearInterval(iv);
   129→  }, [counting]);
   130→
   131→  useEffect(() => () => timers.current.forEach(clearTimeout), []);
   132→
   133→  const send = () => {
   134→    if (left > 0) return;
   135→    setLeft(60);
   136→    timers.current.push(
   137→      setTimeout(() => {
   138→        // 模拟收到短信，自动填充验证码
   139→        setCode('284616');
   140→      }, 1100)
   141→    );
   142→  };
   143→
   144→  return (
   145→    <div className="w-input flex h-11 items-center gap-2 pl-3 pr-1" style={{ borderRadius: 'var(--pr)' }}>
   146→      <ShieldCheck className="size-4 opacity-45" />
   147→      <input
   148→        value={code}
   149→        onChange={(e) => setCode(e.target.value.replace(/\D/g, '').slice(0, 6))}
   150→        placeholder={props.placeholder}
   151→        inputMode="numeric"
   152→        className="flex-1 bg-transparent text-sm outline-none placeholder:opacity-40"
   153→      />
   154→      <button
   155→        type="button"
   156→        onClick={(e) => {
   157→          e.stopPropagation();
   158→          send();
   159→        }}
   160→        className="flex h-8 shrink-0 items-center px-3 text-xs font-semibold disabled:opacity-60"
   161→        style={{
   162→          borderRadius: 'calc(var(--pr) - 4px)',
   163→          background: left > 0 ? 'transparent' : 'var(--p)',
   164→          color: left > 0 ? 'var(--p)' : 'var(--pf)',
   165→          border: left > 0 ? '1px solid var(--p)' : 'none',
   166→        }}
   167→      >
   168→        {left > 0 ? `${left}s 后重发` : props.btnText}
   169→      </button>
   170→    </div>
   171→  );

   175→/* 登录按钮：点击后 加载 → 成功，再触发已绑定的页面跳转                */
   176→/* ------------------------------------------------------------------ */
   177→export function PrimaryBtnInteractive({ props, onTap }: InteractiveCtx) {
   178→  const [st, setSt] = useState<'idle' | 'loading' | 'done'>('idle');
   179→  const timers = useRef<ReturnType<typeof setTimeout>[]>([]);
   180→  useEffect(() => () => timers.current.forEach(clearTimeout), []);
   181→
   182→  const click = (e: React.MouseEvent) => {
   183→    e.stopPropagation();
   184→    if (st !== 'idle') return;
   185→    setSt('loading');
   186→    timers.current.push(
   187→      setTimeout(() => {
   188→        setSt('done');
   189→        timers.current.push(
   190→          setTimeout(() => {
   191→            setSt('idle');
   192→            onTap?.();
   193→          }, 650)
   194→        );
   195→      }, 900)
   196→    );
   197→  };
   198→
   199→  return (
   200→    <button
   201→      type="button"
   202→      onClick={click}
   203→      className="flex h-12 w-full items-center justify-center gap-2 text-[15px] font-bold shadow-md transition-transform active:scale-[0.98]"
   204→      style={{ borderRadius: 'var(--pr)', background: 'var(--p)', color: 'var(--pf)' }}
   205→    >
   206→      {st === 'loading' ? (
   207→        <><LoaderCircle className="size-4 animate-spin" /> 正在登录…</>
   208→      ) : st === 'done' ? (
   209→        <><Check className="size-4" /> 登录成功</>
   210→      ) : (
   211→        <><Lock className="size-4" /> {props.text}{props.sub ? <span className="text-xs font-normal opacity-70">{props.sub}</span> : null}</>
   212→      )}
   213→    </button>
   214→  );
   215→}

   254→type BrandKey = 'wechat' | 'apple' | 'browser' | 'mail' | 'phone';

   256→const BRANDS: Record<BrandKey, { label: string; color: string; dark?: boolean }> = {

   261→  phone: { label: '一键登录', color: '#0f9d58' },

   264→const BRAND_ICONS: Record<BrandKey, typeof MessageCircle> = {

   272→type Phase = 'idle' | 'splash' | 'auth' | 'granting' | 'success' | 'denied';
   273→
   274→export function SocialRowInteractive({ props, onTap }: InteractiveCtx) {
   275→  const [phase, setPhase] = useState<Phase>('idle');
   276→  const [brand, setBrand] = useState<BrandKey | null>(null);
   277→  const timers = useRef<ReturnType<typeof setTimeout>[]>([]);
   278→
   279→  useEffect(() => {
   280→    return () => timers.current.forEach(clearTimeout);
   281→  }, []);
   282→
   283→  const later = (fn: () => void, ms: number) => timers.current.push(setTimeout(fn, ms));
   284→
   285→  const launch = (key: BrandKey) => {
   286→    if (phase !== 'idle') return;
   287→    setBrand(key);
   288→    setPhase('splash');
   289→    later(() => setPhase('auth'), 850);
   290→  };
   291→
   292→  const agree = () => {
   293→    setPhase('granting');
   294→    later(() => {
   295→      setPhase('success');
   296→      later(() => {
   297→        setPhase('idle');
   298→        setBrand(null);
   299→        onTap?.();
   300→      }, 950);
   301→    }, 850);
   302→  };
   303→
   304→  const deny = () => {
   305→    setPhase('denied');
   306→    later(() => {
   307→      setPhase('idle');
   308→      setBrand(null);
   309→    }, 750);
   310→  };
   311→
   312→  const items = ([
   313→    ['showWechat', 'wechat'],
   314→    ['showApple', 'apple'],
   315→    ['showWeb', 'browser'],
   316→    ['showMail', 'mail'],
   317→    ['showPhone', 'phone'],
   318→  ] as const)
   319→    .filter(([k]) => props[k] !== false)
   320→    .map(([, key]) => key);
   321→
   322→  const row = (
   323→    <div className="flex items-center justify-center gap-4 py-1.5">
   324→      {items.map((key) => {
   325→        const Icon = BRAND_ICONS[key];
   326→        const b = BRANDS[key];
   327→        return (
   328→          <button
   329→            key={key}
   330→            type="button"
   331→            onClick={(e) => {
   332→              e.stopPropagation();
   333→              launch(key);
   334→            }}
   335→            className="flex touch-manipulation flex-col items-center gap-1.5 transition-transform active:scale-90"
   336→          >
   337→            <span
   338→              className="flex size-11 items-center justify-center rounded-full border w-line shadow-sm transition-colors"
   339→              style={{ background: `${b.color}14`, borderColor: `${b.color}30` }}
   340→            >
   341→              <Icon className="size-5" style={{ color: b.color }} />
   342→            </span>
   343→            <span className="text-[10px] opacity-45">
   344→              {key === 'phone' ? '本机号码' : b.label}
   345→            </span>
   346→          </button>
   347→        );
   348→      })}
   349→    </div>
   350→  );
   351→
   352→  if (phase === 'idle' || !brand) return row;
   353→
   354→  const b = BRANDS[brand];
   355→  const appName = String(props.appName || '星云 App');
   356→
   357→  /* 唤起 / 授权 全屏覆盖层（挂载到手机屏幕容器） */
   358→  const overlay = (
   359→    <AnimatePresence>
   360→      <motion.div
   361→        key={phase}
   362→        initial={{ opacity: 0 }}
   363→        animate={{ opacity: 1 }}
   364→        exit={{ opacity: 0 }}
   365→        transition={{ duration: 0.22 }}
   366→        className="absolute inset-0 z-[80] flex flex-col overflow-hidden"
   367→        style={{ background: phase === 'splash' ? '#000' : b.dark ? '#0a0a0c' : '#f4f5f7' }}
   368→        onClick={(e) => e.stopPropagation()}
   369→      >
   370→        {/* 仿系统状态栏 */}
   371→        <div
   372→          className="flex h-12 shrink-0 items-center justify-between px-7 pt-2 text-[13px] font-semibold"
   373→          style={{ color: phase === 'splash' || b.dark ? '#fff' : '#111' }}
   374→        >
   375→          <span>9:41</span>
   376→          <div className="flex items-center gap-1.5">
   377→            <Signal className="size-3.5" />
   378→            <Wifi className="size-3.5" />
   379→            <BatteryFull className="size-4" />
   380→          </div>
   381→        </div>
   382→
   383→        {phase === 'splash' && (
   384→          <div className="flex flex-1 flex-col items-center justify-center gap-4 text-white">
   385→            <motion.div
   386→              initial={{ scale: 0.6, opacity: 0 }}
   387→              animate={{ scale: 1, opacity: 1 }}
   388→              transition={{ type: 'spring', stiffness: 300, damping: 22 }}
   389→              className="flex size-20 items-center justify-center rounded-[22px] shadow-2xl"
   390→              style={{ background: b.color }}
   391→            >
   392→              {(() => {
   393→                const Icon = BRAND_ICONS[brand];
   394→                return <Icon className="size-10 text-white" />;
   395→              })()}
   396→            </motion.div>
   397→            <p className="text-lg font-bold">{BRANDS[brand].label}</p>
   398→            <div className="flex items-center gap-1.5 text-xs text-white/50">
   399→              <LoaderCircle className="size-3.5 animate-spin" />
   400→              正在打开「{BRANDS[brand].label}」…
   401→            </div>
   402→            <p className="absolute bottom-10 text-[10px] text-white/35">模拟真机唤起 · 演示环境</p>
   403→          </div>
   404→        )}
   405→
   406→        {(phase === 'auth' || phase === 'granting') && (
   407→          <AuthSheet
   408→            brand={brand}
   409→            color={b.color}
   410→            dark={!!b.dark}
   411→            appName={appName}
   412→            busy={phase === 'granting'}
   413→            onAgree={agree}
   414→            onDeny={deny}
   415→          />
   416→        )}
   417→
   418→        {phase === 'success' && (
   419→          <div className="flex flex-1 flex-col items-center justify-center gap-3">
   420→            <motion.div
   421→              initial={{ scale: 0.5, opacity: 0 }}
   422→              animate={{ scale: 1, opacity: 1 }}
   423→              transition={{ type: 'spring', stiffness: 320, damping: 20 }}
   424→            >
   425→              <CircleCheck className="size-16" style={{ color: b.color }} />
   426→            </motion.div>
   427→            <p className={`text-base font-bold ${b.dark ? 'text-white' : 'text-zinc-900'}`}>
   428→              授权成功
   429→            </p>
   430→            <p className={`text-xs ${b.dark ? 'text-white/50' : 'text-zinc-500'}`}>正在返回 {appName}…</p>
   431→          </div>
   432→        )}
   433→
   434→        {phase === 'denied' && (
   435→          <div className="flex flex-1 flex-col items-center justify-center gap-3">
   436→            <motion.div initial={{ scale: 0.6 }} animate={{ scale: 1 }}>
   437→              <CircleX className="size-14 text-zinc-400" />
   438→            </motion.div>
   439→            <p className={`text-sm font-semibold ${b.dark ? 'text-white/80' : 'text-zinc-600'}`}>已取消授权</p>
   440→          </div>
   441→        )}
   442→
   443→        {/* 仿 Home 条 */}
   444→        <div className="flex h-6 shrink-0 items-center justify-center">
   445→          <div className={`h-1 w-32 rounded-full ${b.dark || phase === 'splash' ? 'bg-white/35' : 'bg-black/25'}`} />
   446→        </div>
   447→      </motion.div>
   448→    </AnimatePresence>
   449→  );
   450→
   451→  const target = typeof document !== 'undefined' ? document.getElementById('phone-screen') : null;
   452→  const overlayNode = target ? createPortal(overlay, target) : null;
   453→
   454→  return (
   455→    <>
   456→      {row}
   457→      {overlayNode}
   458→    </>
   459→  );
   460→}

   882→/* 主按钮（fn.big-button）：点击 → 加载 → 成功 → 触发绑定跳转          */
   883→/* ------------------------------------------------------------------ */
   884→
   885→export function BigButtonInteractive({ props, onTap }: InteractiveCtx) {
   886→  const [st, setSt] = useState<'idle' | 'loading' | 'done'>('idle');
   887→  const timers = useRef<ReturnType<typeof setTimeout>[]>([]);
   888→  useEffect(() => () => timers.current.forEach(clearTimeout), []);
   889→
   890→  const variant = props.style === 'outline' || props.style === 'ghost' ? props.style : 'primary';
   891→  const br = props.radius === 'normal' ? 'var(--pr)' : '999px';
   892→  const skin = variant === 'primary'
   893→    ? { background: 'var(--p)', color: 'var(--pf)' }
   894→    : variant === 'outline'
   895→      ? { border: '1.5px solid var(--p)', color: 'var(--p)' }
   896→      : { color: 'var(--p)' };
   897→
   898→  const click = (e: React.MouseEvent) => {
   899→    e.stopPropagation();
   900→    if (st !== 'idle') return;
   901→    setSt('loading');
   902→    timers.current.push(
   903→      setTimeout(() => {
   904→        setSt('done');
   905→        timers.current.push(
   906→          setTimeout(() => {
   907→            setSt('idle');
   908→            onTap?.();
   909→          }, 620)
   910→        );
   911→      }, 700)
   912→    );
   913→  };
   914→
   915→  return (
   916→    <button
   917→      type="button"
   918→      onClick={click}
   919→      className={`flex h-12 w-full items-center justify-center gap-2 text-[15px] font-bold transition-transform active:scale-[0.98] ${variant === 'primary' ? 'shadow-md' : variant === 'ghost' ? 'w-chip' : ''}`}
   920→      style={{ borderRadius: br, ...skin }}
   921→    >
   922→      {st === 'loading' ? (
   923→        <><LoaderCircle className="size-4 animate-spin" /> 请稍候…</>
   924→      ) : st === 'done' ? (
   925→        <><Check className="size-4" /> 操作成功</>
   926→      ) : (
   927→        props.text
   928→      )}
   929→    </button>
   930→  );
   931→}
