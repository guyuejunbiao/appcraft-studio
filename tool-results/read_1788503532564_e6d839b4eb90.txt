   402→function Label({ children }: { children: React.ReactNode }) {

   406→/** 自由布局数值输入（X/Y/W/H） */
   407→function NumField({
   408→  label, value, min, max, onChange,
   409→}: {
   410→  label: string;
   411→  value: number;
   412→  min: number;
   413→  max: number;
   414→  onChange: (v: number) => void;
   415→}) {
   416→  return (
   417→    <div>
   418→      <Label>{label}</Label>
   419→      <Input
   420→        type="number"
   421→        className="h-8 text-xs tabular-nums"
   422→        value={Math.round(value)}
   423→        min={min}
   424→        max={max}
   425→        onChange={(e) => {
   426→          const v = Number(e.target.value);
   427→          if (!Number.isNaN(v)) onChange(Math.max(min, Math.min(max, v)));
   428→        }}
   429→      />
   430→    </div>
   431→  );
   432→}

   434→function Stat({ label, value }: { label: string; value: number }) {

   437→      <p className="text-lg font-extrabold tabular-nums">{value}</p>
   438→      <p className="text-[10px] text-zinc-400">{label}</p>
   439→    </div>
   440→  );
   441→}

   443→/** 属性字段控件 */
   444→function FieldControl({
   445→  field, value, onChange,
   446→}: {
   447→  field: PropField;
   448→  value: any;
   449→  onChange: (v: any) => void;
   450→}) {
   451→  if (field.type === 'switch') {
   452→    return (
   453→      <div className="flex items-center justify-between">
   454→        <Label>{field.label}</Label>
   455→        <Switch checked={value !== false} onCheckedChange={onChange} />
   456→      </div>
   457→    );
   458→  }
   459→  if (field.type === 'textarea') {
   460→    return (
   461→      <div>
   462→        <Label>{field.label}</Label>
   463→        <Textarea
   464→          className="min-h-16 text-xs"
   465→          value={String(value ?? '')}
   466→          onChange={(e) => onChange(e.target.value)}
   467→          placeholder={field.placeholder}
   468→        />
   469→      </div>
   470→    );
   471→  }
   472→  if (field.type === 'select') {
   473→    return (
   474→      <div>
   475→        <Label>{field.label}</Label>
   476→        <Select value={String(value ?? '')} onValueChange={onChange}>
   477→          <SelectTrigger className="h-8 text-xs"><SelectValue /></SelectTrigger>
   478→          <SelectContent>
   479→            {(field.options ?? []).map((o) => (
   480→              <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>
   481→            ))}
   482→          </SelectContent>
   483→        </Select>
   484→      </div>
   485→    );
   486→  }
   487→  if (field.type === 'color') {
   488→    return (
   489→      <div>
   490→        <Label>{field.label}</Label>
   491→        <div className="flex items-center gap-2">
   492→          <input
   493→            type="color"
   494→            value={String(value ?? '#000000')}
   495→            onChange={(e) => onChange(e.target.value)}
   496→            className="size-8 cursor-pointer rounded-lg border"
   497→          />
   498→          <Input className="h-8 flex-1 font-mono text-xs" value={String(value ?? '')} onChange={(e) => onChange(e.target.value)} />
   499→        </div>
   500→      </div>
   501→    );
   502→  }
   503→  if (field.type === 'number') {
   504→    return (
   505→      <div>
   506→        <div className="mb-1.5 flex items-center justify-between">
   507→          <Label>{field.label}</Label>
   508→          <span className="text-xs font-semibold tabular-nums text-zinc-500">{Number(value ?? 0)}</span>
   509→        </div>
   510→        <Slider
   511→          value={[Number(value ?? 0)]}
   512→          min={field.min ?? 0}
   513→          max={field.max ?? 100}
   514→          step={field.step ?? 1}
   515→          onValueChange={([v]) => onChange(v)}
   516→        />
   517→      </div>
   518→    );
   519→  }
   520→  return (
   521→    <div>
   522→      <Label>{field.label}</Label>
   523→      <Input
   524→        className="h-8 text-xs"
   525→        value={String(value ?? '')}
   526→        onChange={(e) => onChange(e.target.value)}
   527→        placeholder={field.placeholder}
   528→      />
   529→    </div>
   530→  );
   531→}

   533→/** 交互（连接）编辑器 */
   534→function ConnectionEditor({
   535→  pageId, widgetId, widgetName, fromConnections, slots, onAdd, onRemove,
   536→}: {
   537→  pageId: string;
   538→  widgetId: string;
   539→  widgetName: string;
   540→  fromConnections: { id: string; toPageId: string; animation: string; slot?: string }[];
   541→  slots: { key: string; label: string }[];
   542→  onAdd: (c: { fromPageId: string; fromWidgetId: string; toPageId: string; animation: 'slide' | 'fade' | 'push' | 'none'; slot?: string }) => void;
   543→  onRemove: (id: string) => void;
   544→}) {
   545→  const pages = useBuilder((s) => s.pages);
   546→  const [toPage, setToPage] = useState('');
   547→  const [anim, setAnim] = useState<'slide' | 'fade' | 'push' | 'none'>('slide');
   548→  const [slot, setSlot] = useState('widget');
   549→
   550→  const targets = useMemo(() => pages.filter((p) => p.id !== pageId), [pages, pageId]);
   551→  const slotLabel = (key?: string) =>
   552→    key ? (slots.find((s) => s.key === key)?.label ?? `槽位 ${key}`) : null;
   553→
   554→  return (
   555→    <div className="space-y-3">
   556→      <div className="flex items-center gap-1.5 text-xs text-zinc-500">
   557→        <Link2 className="size-3.5" />
   558→        「{widgetName}」被点击后跳转到
   559→      </div>
   560→
   561→      {fromConnections.length > 0 && (
   562→        <div className="space-y-1.5">
   563→          {fromConnections.map((c) => {
   564→            const target = pages.find((p) => p.id === c.toPageId);
   565→            const animLabel = ANIM_OPTIONS.find((a) => a.value === c.animation)?.label ?? c.animation;
   566→            const sLabel = slotLabel(c.slot);
   567→            return (
   568→              <div key={c.id} className="flex items-center gap-2 rounded-xl border bg-zinc-50 px-2.5 py-2">
   569→                <ArrowRight className="size-3.5 shrink-0 text-emerald-500" />
   570→                <span className="min-w-0 flex-1 truncate text-xs font-semibold">
   571→                  {sLabel && <span className="mr-1 rounded bg-emerald-100 px-1 py-0.5 text-[9px] font-semibold text-emerald-600">{sLabel}</span>}
   572→                  {target?.name ?? '未知页面'}
   573→                  <span className="ml-1.5 rounded bg-zinc-200 px-1 py-0.5 text-[9px] font-normal text-zinc-500">{animLabel}</span>
   574→                </span>
   575→                <button
   576→                  className="text-zinc-400 hover:text-rose-500"
   577→                  onClick={() => onRemove(c.id)}
   578→                  title="删除连接"
   579→                >
   580→                  <X className="size-3.5" />
   581→                </button>
   582→              </div>
   583→            );
   584→          })}
   585→        </div>
   586→      )}
   587→
   588→      <div className="space-y-2 rounded-xl border border-dashed p-3">
   589→        {slots.length > 0 && (
   590→          <div>
   591→            <Label>绑定目标</Label>
   592→            <Select value={slot} onValueChange={setSlot}>
   593→              <SelectTrigger className="h-8 text-xs"><SelectValue /></SelectTrigger>
   594→              <SelectContent>
   595→                <SelectItem value="widget">整个组件（任意点击）</SelectItem>
   596→                {slots.map((s) => (
   597→                  <SelectItem key={s.key} value={s.key}>{s.label}</SelectItem>
   598→                ))}
   599→              </SelectContent>
   600→            </Select>
   601→            {slot !== 'widget' && (
   602→              <p className="mt-1 text-[10px] leading-3.5 text-zinc-400">
   603→                预览中点击该标签即跳转，适合底部导航切换页面
   604→              </p>
   605→            )}
   606→          </div>
   607→        )}
   608→        <Select value={toPage} onValueChange={setToPage}>
   609→          <SelectTrigger className="h-8 text-xs"><SelectValue placeholder="选择目标页面" /></SelectTrigger>
   610→          <SelectContent>
   611→            {targets.map((p) => (
   612→              <SelectItem key={p.id} value={p.id}>
   613→                {p.name}
   614→                {p.isHome ? '（主页）' : ''}
   615→              </SelectItem>
   616→            ))}
   617→          </SelectContent>
   618→        </Select>
   619→        <Select value={anim} onValueChange={(v) => setAnim(v as never)}>
   620→          <SelectTrigger className="h-8 text-xs"><SelectValue /></SelectTrigger>
   621→          <SelectContent>
   622→            {ANIM_OPTIONS.map((a) => (
   623→              <SelectItem key={a.value} value={a.value}>{a.label}</SelectItem>
   624→            ))}
   625→          </SelectContent>
   626→        </Select>
   627→        <Button
   628→          size="sm"
   629→          className="w-full"
   630→          disabled={!toPage}
   631→          onClick={() => {
   632→            if (!toPage) return;
   633→            onAdd({
   634→              fromPageId: pageId,
   635→              fromWidgetId: widgetId,
   636→              toPageId: toPage,
   637→              animation: anim,
   638→              slot: slot === 'widget' ? undefined : slot,
   639→            });
   640→            setToPage('');
   641→          }}
   642→        >
   643→          <Plus className="mr-1 size-3.5" /> 添加跳转连接
   644→        </Button>
   645→      </div>
   646→
   647→      {targets.length === 0 && (
   648→        <p className="text-center text-[11px] text-zinc-400">只有一页，先添加新页面才能建立连接</p>
   649→      )}
   650→    </div>
   651→  );
   652→}
