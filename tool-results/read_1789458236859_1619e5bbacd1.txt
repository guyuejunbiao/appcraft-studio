    15→ */
    16→
    17→/** 逗号（中英文）分隔 → 字符串数组（去空白项） */
    18→const splitList = (raw: unknown): string[] =>
    19→  String(raw ?? '').split(/[,,]/).map((s) => s.trim()).filter(Boolean);
    20→
    21→/** 逗号分隔 → 数值数组（非法项按 0 处理） */
    22→const numList = (raw: unknown): number[] =>
    23→  splitList(raw).map((s) => (Number.isFinite(Number(s)) ? Number(s) : 0));
    24→
    25→/** 数字钳制 */
    26→const clamp = (v: number, min: number, max: number) => Math.min(max, Math.max(min, v));
    27→
    28→/** 宽松取数：容忍千分位逗号 / 单位文字（如 '8,426' / '62.5kg'） */
    29→const looseNum = (raw: unknown): number =>
    30→  Number(String(raw ?? '').replace(/[^\d.-]/g, '')) || 0;
    31→
