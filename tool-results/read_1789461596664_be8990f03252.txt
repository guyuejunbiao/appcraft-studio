    45→/* ==================== 组件选择器（画板「+」弹层） ==================== */
    46→function WidgetPickerPopover({ pageId, children }: { pageId: string; children: React.ReactNode }) {
    47→  const [open, setOpen] = useState(false);
    48→  const [q, setQ] = useState('');
    49→  const [cat, setCat] = useState<string>('all');
    50→  const addWidgetToPage = useBuilder((s) => s.addWidgetToPage);
    51→
    52→  const list = useMemo(() => {
    53→    const kw = q.trim().toLowerCase();
    54→    return allWidgets.filter((w) => {
    55→      if (cat !== 'all' && w.category !== cat) return false;
    56→      if (!kw) return true;
    57→      return w.name.toLowerCase().includes(kw) || w.desc.toLowerCase().includes(kw) || w.type.includes(kw);
    58→    });
    59→  }, [q, cat]);
    60→
    61→  return (
    62→    <Popover open={open} onOpenChange={(b) => { setOpen(b); if (!b) setQ(''); }}>
    63→      <PopoverTrigger asChild>{children}</PopoverTrigger>
    64→      <PopoverContent side="bottom" align="end" className="w-80 p-2.5">
    65→        <div className="relative mb-2">
    66→          <Search className="pointer-events-none absolute left-2.5 top-1/2 size-3.5 -translate-y-1/2 text-zinc-400" />
    67→          <Input
    68→            value={q}
    69→            onChange={(e) => setQ(e.target.value)}
    70→            placeholder="搜索组件…"
    71→            className="h-8 pl-8 text-xs"
    72→            autoFocus
    73→          />
    74→        </div>
    75→        <div className="mb-2 flex gap-1 overflow-x-auto thin-scroll pb-1">
    76→          <button
    77→            onClick={() => setCat('all')}
    78→            className={`shrink-0 rounded-full px-2.5 py-1 text-[11px] font-semibold transition-colors ${
    79→              cat === 'all' ? 'bg-violet-500 text-white' : 'bg-zinc-100 text-zinc-500 hover:bg-zinc-200'
    80→            }`}
    81→          >
    82→            全部
    83→          </button>
    84→          {categories.map((c) => (
    85→            <button
    86→              key={c.id}
    87→              onClick={() => setCat(c.id)}
    88→              className={`shrink-0 rounded-full px-2.5 py-1 text-[11px] font-semibold transition-colors ${
    89→                cat === c.id ? 'bg-violet-500 text-white' : 'bg-zinc-100 text-zinc-500 hover:bg-zinc-200'
    90→              }`}
    91→            >
    92→              {c.name}
    93→            </button>
    94→          ))}
    95→        </div>
    96→        <div className="grid max-h-64 grid-cols-3 gap-1 overflow-y-auto thin-scroll">
    97→          {list.map((w) => (
    98→            <button
    99→              key={w.type}
   100→              onClick={() => {
   101→                addWidgetToPage(pageId, w.type);
   102→                toast.success(`已添加「${w.name}」到「${useBuilder.getState().pages.find((p) => p.id === pageId)?.name ?? '页面'}」`);
   103→              }}
   104→              title={w.desc}
   105→              className="flex flex-col items-center gap-1 rounded-lg border border-transparent p-2 text-center transition-all hover:border-violet-200 hover:bg-violet-50/50 active:scale-95"
   106→            >
   107→              <span className="flex size-7 items-center justify-center rounded-md bg-violet-50 text-violet-500">
   108→                <w.icon className="size-4" />
   109→              </span>
   110→              <span className="w-full truncate text-[10px] font-semibold text-zinc-600">{w.name}</span>
   111→            </button>
   112→          ))}
   113→          {list.length === 0 && (
   114→            <p className="col-span-3 py-6 text-center text-xs text-zinc-400">没有匹配的组件</p>
   115→          )}
   116→        </div>
   117→      </PopoverContent>
   118→    </Popover>
   119→  );
   120→}

   122→/* ==================== 选中组件快捷编辑（文字就地 / 间距 / 布局） ==================== */
   123→function QuickEditor({ widget, onDone }: { widget: WidgetInstance; onDone: () => void }) {
   124→  const def = getWidget(widget.type);
   125→  const updateWidgetProps = useBuilder((s) => s.updateWidgetProps);
   126→  const updateWidget = useBuilder((s) => s.updateWidget);
   127→  if (!def) return null;
   128→  const merged = { ...def.defaultProps, ...widget.props };
   129→  const textFields = def.fields.filter((f) => f.type === 'text' || f.type === 'textarea');
   130→
   131→  return (
   132→    <div className="w-64 space-y-3 p-3">
   133→      <div className="flex items-center gap-1.5">
   134→        <span className="flex size-6 items-center justify-center rounded-md bg-violet-50 text-violet-500">
   135→          <def.icon className="size-3.5" />
   136→        </span>
   137→        <span className="text-xs font-bold">{def.name}</span>
   138→        <span className="ml-auto rounded bg-violet-50 px-1.5 py-0.5 text-[9px] font-bold text-violet-500">就地编辑</span>
   139→      </div>
   140→
   141→      {/* 文字内容就地编辑（实时生效，无需跳回编辑器） */}
   142→      {textFields.slice(0, 3).map((f) => (
   143→        <div key={f.key}>
   144→          <p className="mb-1 flex items-center gap-1 text-[11px] font-semibold text-zinc-500">
   145→            <Type className="size-3" /> {f.label}
   146→          </p>
   147→          {f.type === 'textarea' ? (
   148→            <Textarea
   149→              value={String(merged[f.key] ?? '')}
   150→              onChange={(e) => updateWidgetProps(widget.id, { [f.key]: e.target.value })}
   151→              placeholder={f.placeholder}
   152→              className="min-h-[52px] text-xs"
   153→              rows={2}
   154→              autoFocus={f === textFields[0]}
   155→            />
   156→          ) : (
   157→            <Input
   158→              value={String(merged[f.key] ?? '')}
   159→              onChange={(e) => updateWidgetProps(widget.id, { [f.key]: e.target.value })}
   160→              placeholder={f.placeholder}
   161→              className="h-8 text-xs"
   162→              autoFocus={f === textFields[0]}
   163→            />
   164→          )}
   165→        </div>
   166→      ))}
   167→      {textFields.length === 0 && (
   168→        <p className="rounded-lg bg-zinc-50 px-2.5 py-2 text-[11px] text-zinc-400">
   169→          该组件没有可编辑文字，可在编辑器属性面板配置
   170→        </p>
   171→      )}
   172→
   173→      {/* 间距（上/下边距）+ 宽度 + 对齐 */}
   174→      <div className="space-y-2 rounded-lg border bg-zinc-50/70 p-2">
   175→        <p className="flex items-center gap-1 text-[11px] font-semibold text-zinc-500">
   176→          <ArrowUpDown className="size-3" /> 间距与布局
   177→        </p>
   178→        {([
   179→          ['mt', '上间距'],
   180→          ['mb', '下间距'],
   181→        ] as const).map(([key, label]) => (
   182→          <div key={key} className="flex items-center gap-2">
   183→            <span className="w-11 shrink-0 text-[11px] text-zinc-500">{label}</span>
   184→            <input
   185→              type="range"
   186→              min={0}
   187→              max={48}
   188→              step={2}
   189→              value={widget[key] ?? 0}
   190→              onChange={(e) => updateWidget(widget.id, { [key]: Number(e.target.value) })}
   191→              className="h-1 flex-1 accent-violet-500"
   192→              aria-label={label}
   193→            />
   194→            <span className="w-6 text-right text-[10px] font-semibold tabular-nums text-zinc-500">
   195→              {widget[key] ?? 0}
   196→            </span>
   197→          </div>
   198→        ))}
   199→        <div className="flex items-center gap-2">
   200→          <Select
   201→            value={widget.width ?? 'full'}
   202→            onValueChange={(v) => updateWidget(widget.id, { width: v as WidgetInstance['width'] })}
   203→          >
   204→            <SelectTrigger className="h-7 flex-1 text-[11px]"><SelectValue /></SelectTrigger>
   205→            <SelectContent>
   206→              <SelectItem value="full">通栏</SelectItem>
   207→              <SelectItem value="three-quarter">3/4 宽</SelectItem>
   208→              <SelectItem value="half">半宽</SelectItem>
   209→              <SelectItem value="third">1/3 宽</SelectItem>
   210→            </SelectContent>
   211→          </Select>
   212→          <Select
   213→            value={widget.align ?? 'left'}
   214→            onValueChange={(v) => updateWidget(widget.id, { align: v as WidgetInstance['align'] })}
   215→          >
   216→            <SelectTrigger className="h-7 flex-1 text-[11px]"><SelectValue /></SelectTrigger>
   217→            <SelectContent>
   218→              <SelectItem value="left">靠左</SelectItem>
   219→              <SelectItem value="center">居中</SelectItem>
   220→              <SelectItem value="right">靠右</SelectItem>
   221→            </SelectContent>
   222→          </Select>
   223→        </div>
   224→      </div>
   225→
   226→      <Button size="sm" variant="outline" className="w-full" onClick={onDone}>完成</Button>
   227→    </div>
   228→  );
   229→}

   231→/* ==================== 单个画板 ==================== */
   232→function Artboard({
   233→  page,
   234→  index,
   235→  zoom,
   236→  theme,
   237→  focused,
   238→  selWidgetId,
   239→  linkDragging,
   240→  onWidgetPointerDown,
   241→  onBodyPointerDown,
   242→  onOpenEditor,
   243→  onLinkStart,
   244→}: {
   245→  page: PageData;
   246→  index: number;
   247→  zoom: number;
   248→  theme: { primary: string; radius: 'none' | 'sm' | 'md' | 'lg' | 'full'; dark: boolean; icon?: string; iconBG?: string };
   249→  focused: boolean;
   250→  selWidgetId: string | null;
   251→  linkDragging: boolean;
   252→  /** 点击组件：选中（pageId, widgetId）；widgetId 为空 = 清除选择 */
   253→  onWidgetPointerDown: (pageId: string, widgetId: string) => void;
   254→  onBodyPointerDown: (pageId: string) => void;
   255→  onOpenEditor: (pageId: string) => void;
   256→  onLinkStart: (pageId: string, e: React.PointerEvent) => void;
   257→}) {
   258→  const tabs = useBuilder((s) => s.tabs);
   259→  const setFlowPos = useBuilder((s) => s.setFlowPos);
   260→  const updatePage = useBuilder((s) => s.updatePage);
   261→  const boardRef = useRef<HTMLDivElement>(null);
   262→  const [dragging, setDragging] = useState<{ dx: number; dy: number } | null>(null);
   263→  const [renaming, setRenaming] = useState(false);
   264→  const [nameDraft, setNameDraft] = useState(page.name);
   265→
   266→  const pos = {
   267→    x: page.flowX ?? defaultPos(index).x,
   268→    y: page.flowY ?? defaultPos(index).y,
   269→  };
   270→  const isFree = page.layout === 'free' && page.components.every((c) => typeof c.x === 'number');
   271→  const visible = page.components.filter((w) => !w.hidden);
   272→
   273→  /* 拖动画板位置（按住 chrome 条；坐标按 zoom 还原） */
   274→  const startDrag = (e: React.PointerEvent) => {
   275→    if (linkDragging) return;
   276→    const rect = boardRef.current?.getBoundingClientRect();
   277→    if (!rect) return;
   278→    setDragging({
   279→      dx: (e.clientX - rect.left) / zoom,
   280→      dy: (e.clientY - rect.top) / zoom,
   281→    });
   282→  };
   283→  useEffect(() => {
   284→    if (!dragging) return;
   285→    const onMove = (e: PointerEvent) => {
   286→      const rect = boardRef.current?.getBoundingClientRect();
   287→      if (!rect) return;
   288→      setFlowPos(page.id, (e.clientX - rect.left) / zoom - dragging.dx, (e.clientY - rect.top) / zoom - dragging.dy);
   289→    };
   290→    const onUp = () => setDragging(null);
   291→    window.addEventListener('pointermove', onMove);
   292→    window.addEventListener('pointerup', onUp, { once: true });
   293→    return () => {
   294→      window.removeEventListener('pointermove', onMove);
   295→      window.removeEventListener('pointerup', onUp);
   296→    };
   297→  }, [dragging, page.id, setFlowPos, zoom]);
   298→
   299→  return (
   300→    <div
   301→      ref={boardRef}
   302→      className={`absolute select-none ${dragging ? 'z-30' : 'z-10'}`}
   303→      style={{ left: pos.x, top: pos.y, width: AB_W }}
   304→      onPointerDown={(e) => e.stopPropagation()}
   305→    >
   306→      {/* 连线起点（右侧圆点，拖到目标画板建立连接） */}
   307→      <button
   308→        aria-label={`从「${page.name}」拖出连线`}
   309→        title="按住拖到目标画板建立跳转连接"
   310→        className="absolute -right-3 top-[52%] z-40 flex size-6 -translate-y-1/2 cursor-crosshair items-center justify-center rounded-full border-2 border-white bg-violet-500 text-white shadow-md transition-all hover:scale-125 hover:bg-violet-600"
   311→        onPointerDown={(e) => {
   312→          e.stopPropagation();
   313→          e.preventDefault();
   314→          try { (e.currentTarget as HTMLElement).setPointerCapture(e.pointerId); } catch { /* noop */ }
   315→          onLinkStart(page.id, e);
   316→        }}
   317→      >
   318→        <Plus className="size-3.5" />
   319→      </button>
   320→
   321→      {/* chrome 标题条 */}
   322→      <div
   323→        className={`flex items-center gap-1 rounded-t-xl border border-b-0 bg-white px-2 ${dragging ? 'cursor-grabbing shadow-lg' : 'cursor-grab hover:bg-zinc-50'}`}
   324→        style={{ height: CHROME_H }}
   325→        onPointerDown={startDrag}
   326→        onDoubleClick={() => { setRenaming(true); setNameDraft(page.name); }}
   327→        title="拖动移动画板位置 · 双击重命名"
   328→      >
   329→        {page.isHome && <Home className="size-3 shrink-0 text-amber-500" />}
   330→        {renaming ? (
   331→          <input
   332→            value={nameDraft}
   333→            onChange={(e) => setNameDraft(e.target.value)}
   334→            onBlur={() => { setRenaming(false); if (nameDraft.trim() && nameDraft !== page.name) updatePage(page.id, { name: nameDraft.trim() }); }}
   335→            onKeyDown={(e) => { if (e.key === 'Enter') (e.target as HTMLInputElement).blur(); if (e.key === 'Escape') setRenaming(false); }}
   336→            className="min-w-0 flex-1 rounded border border-violet-300 px-1 text-[11px] font-bold outline-none"
   337→            autoFocus
   338→            onPointerDown={(e) => e.stopPropagation()}
   339→          />
   340→        ) : (
   341→          <span className="min-w-0 flex-1 truncate text-[11px] font-bold">{page.name}</span>
   342→        )}
   343→        <span className="shrink-0 rounded-full bg-zinc-100 px-1.5 text-[9px] font-semibold text-zinc-400">
   344→          {page.components.length}
   345→        </span>
   346→        <WidgetPickerPopover pageId={page.id}>
   347→          <button
   348→            className="flex size-5 shrink-0 items-center justify-center rounded bg-violet-500 text-white transition-transform hover:scale-110"
   349→            title="添加组件到该页"
   350→            onPointerDown={(e) => e.stopPropagation()}
   351→          >
   352→            <Plus className="size-3" />
   353→          </button>
   354→        </WidgetPickerPopover>
   355→        <button
   356→          className="flex size-5 shrink-0 items-center justify-center rounded border text-zinc-500 hover:bg-zinc-900 hover:text-white"
   357→          title="在编辑器中打开此页"
   358→          onPointerDown={(e) => e.stopPropagation()}
   359→          onClick={() => onOpenEditor(page.id)}
   360→        >
   361→          <Pencil className="size-2.5" />
   362→        </button>
   363→      </div>
   364→
   365→      {/* 机身（缩放渲染，内容可点击选中组件） */}
   366→      <div
   367→        className={`relative overflow-hidden rounded-b-xl border border-t-0 bg-white shadow-md transition-shadow ${
   368→          focused ? 'ring-[3px] ring-amber-400' : selWidgetId ? 'ring-2 ring-violet-300' : 'hover:shadow-lg'
   369→        }`}
   370→        style={{ width: AB_W, height: AB_H }}
   371→        onPointerDown={(e) => {
   372→          e.stopPropagation();
   373→          /* 命中机身空白（未被组件拦截）= 清除选择 */
   374→          if (e.target === e.currentTarget) onWidgetPointerDown(page.id, '');
   375→        }}
   376→      >
   377→        <div
   378→          className="absolute left-0 top-0 origin-top-left"
   379→          style={{ transform: `scale(${AB_SCALE})`, width: FRAME_W, height: FRAME_H }}
   380→        >
   381→          <PhoneFrame theme={theme} pageBg={page.background}>
   382→            <div className="flex h-full flex-col">
   383→              <div
   384→                className={`min-h-0 flex-1 overflow-y-auto overflow-x-hidden ${tabs.length > 0 ? '' : 'pb-2'}`}
   385→                onPointerDown={(e) => {
   386→                  /* 内容空白点击 = 清除选择（组件内部会 stopPropagation） */
   387→                  if (e.target === e.currentTarget || (e.target as HTMLElement).dataset?.canvasBlank) {
   388→                    onWidgetPointerDown(page.id, '');
   389→                  }
   390→                }}
   391→                data-canvas-blank="true"
   392→              >
   393→                {isFree ? (
   394→                  <div className="relative w-full" style={{ minHeight: 700 }}>
   395→                    {visible.map((w) => (
   396→                      <div
   397→                        key={w.id}
   398→                        className="absolute"
   399→                        style={{ left: w.x ?? 0, top: w.y ?? 0, width: w.w ?? 355 }}
   400→                        onPointerDown={(e) => { e.stopPropagation(); onWidgetPointerDown(page.id, w.id); }}
   401→                      >
   402→                        <div className={selWidgetId === w.id ? 'rounded outline outline-2 outline-offset-1 outline-violet-500' : 'rounded hover:outline hover:outline-1 hover:outline-violet-300'}>
   403→                          <WidgetRenderer w={w} />
   404→                        </div>
   405→                      </div>
   406→                    ))}
   407→                    {visible.length === 0 && (
   408→                      <div className="m-3 flex h-32 flex-col items-center justify-center gap-1 rounded-2xl border-2 border-dashed border-zinc-300 text-zinc-400">
   409→                        <p className="text-xs font-semibold">「{page.name}」还是空的</p>
   410→                        <p className="text-[10px]">点画板上方 + 添加组件</p>
   411→                      </div>
   412→                    )}
   413→                  </div>
   414→                ) : (
   415→                  <div className="relative pb-6">
   416→                    {visible.map((w) => (
   417→                      <div
   418→                        key={w.id}
   419→                        onPointerDown={(e) => { e.stopPropagation(); onWidgetPointerDown(page.id, w.id); }}
   420→                      >
   421→                        <div className={selWidgetId === w.id ? 'rounded outline outline-2 outline-offset-[-1px] outline-violet-500' : ''}>
   422→                          <WidgetRenderer w={w} />
   423→                        </div>
   424→                      </div>
   425→                    ))}
   426→                    {visible.length === 0 && (
   427→                      <div className="m-3 flex h-32 flex-col items-center justify-center gap-1 rounded-2xl border-2 border-dashed border-zinc-300 text-zinc-400">
   428→                        <p className="text-xs font-semibold">「{page.name}」还是空的</p>
   429→                        <p className="text-[10px]">点画板上方 + 添加组件</p>
   430→                      </div>
   431→                    )}
   432→                  </div>
   433→                )}
   434→              </div>
   435→              {/* App 级底部导航：点击 = 聚焦目标画板 */}
   436→              {tabs.length > 0 && (
   437→                <AppTabBar
   438→                  tabs={tabs}
   439→                  activePageId={page.id}
   440→                  onSelect={(tab) => onBodyPointerDown(tab.pageId)}
   441→                />
   442→              )}
   443→            </div>
   444→          </PhoneFrame>
   445→        </div>
   446→      </div>
   447→    </div>
   448→  );
   449→}

   451→/* ==================== 无限画布主组件 ==================== */
   452→export function InfiniteCanvas() {
   453→  /* 调试便捷入口（只读检查用） */
   454→  useEffect(() => {
