    34→/**

    40→ * - 流式布局下拖动时浮出提示，可一键切换自由布局
    41→ */

    46→/** 组件缺少自由布局坐标（旧数据） */
    47→function missingCoords(components: WidgetInstance[]): boolean {
    48→  return components.some((c) => typeof c.x !== 'number' || typeof c.y !== 'number');

    52→  const pages = useBuilder((s) => s.pages);
    53→  const currentPageId = useBuilder((s) => s.currentPageId);
    54→  const selectedWidgetId = useBuilder((s) => s.selectedWidgetId);
    55→  const theme = useBuilder((s) => s.project?.theme);
    56→  const select = useBuilder((s) => s.select);
    57→  const removeWidget = useBuilder((s) => s.removeWidget);
    58→  const duplicateWidget = useBuilder((s) => s.duplicateWidget);
    59→
    60→  const [zoom, setZoom] = useState(0.85);
    61→  const [indicator, setIndicator] = useState<{ index: number; y: number } | null>(null);
    62→  const [freePreview, setFreePreview] = useState<{ x: number; y: number; w: number } | null>(null);
    63→  const [guides, setGuides] = useState<{ v: number[]; h: number[] }>(EMPTY_GUIDES);
    64→  const [sizeBadge, setSizeBadge] = useState<string | null>(null);
    65→  const [posBadge, setPosBadge] = useState<string | null>(null);
    66→  const [freeHeight, setFreeHeight] = useState(700);
    67→  const [showFlowTip, setShowFlowTip] = useState(true);
    68→  const [flowDragHint, setFlowDragHint] = useState(false);
    69→  const flowHintTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    70→
    71→  const itemRefs = useRef<Map<string, HTMLDivElement>>(new Map());
    72→  const contentRef = useRef<HTMLDivElement>(null);
    73→  const suppressClickRef = useRef(false);
    74→  const dragMetaRef = useRef<{ offX: number; offY: number; batch: boolean }>({ offX: 0, offY: 0, batch: false });
    75→  const resizeRef = useRef<null | {
    76→    id: string; dir: string; startX: number; startY: number;
    77→    rect: { x: number; y: number; w: number; h: number };
    78→  }>(null);
    79→  const freePreviewRef = useRef<{ x: number; y: number; w: number } | null>(null);
    80→  const zoomRef = useRef(zoom);
    81→  const isFreeRef = useRef(false);
    82→
    83→  useEffect(() => { zoomRef.current = zoom; }, [zoom]);
    84→  useEffect(() => {
    85→    const b = useBuilder.getState();
    86→    const pg = b.pages.find((p) => p.id === b.currentPageId);
    87→    isFreeRef.current = pg?.layout === 'free' && !missingCoords(pg?.components ?? []);
    88→  }, [pages, currentPageId]);
    89→  useEffect(() => () => { useDnd.getState().end(); }, []);
    90→
    91→  const page = pages.find((p) => p.id === currentPageId);
    92→  const isFreeMode = page?.layout === 'free';
    93→  const isFree = isFreeMode && !missingCoords(page?.components ?? []);
    94→  const t = theme ?? { primary: '#f97316', radius: 'md' as const, dark: false };
    95→  const dndActive = useDnd((s) => s.active);
    96→
    97→  /* ---------- 坐标与落点计算 ---------- */
    98→
    99→  const toLocal = useCallback((clientX: number, clientY: number) => {
   100→    const el = contentRef.current;
   101→    const rect = el?.getBoundingClientRect();
   102→    if (!el || !rect) return { x: 0, y: 0 };
   103→    return {
   104→      x: (clientX - rect.left) / zoomRef.current,
   105→      y: (clientY - rect.top) / zoomRef.current + el.scrollTop,
   106→    };
   107→  }, []);
   108→
   109→  const insideContent = useCallback((clientX: number, clientY: number) => {
   110→    const rect = contentRef.current?.getBoundingClientRect();
   111→    if (!rect) return false;
   112→    const pad = 8;
   113→    return (
   114→      clientX >= rect.left - pad && clientX <= rect.right + pad &&
   115→      clientY >= rect.top - pad && clientY <= rect.bottom + pad
   116→    );
   117→  }, []);
   118→
   119→  const computeDrop = useCallback((clientY: number) => {
   120→    const b = useBuilder.getState();
   121→    const list = b.pages.find((p) => p.id === b.currentPageId)?.components ?? [];
   122→    const origin = contentRef.current?.getBoundingClientRect();
   123→    const scrollTop = contentRef.current?.scrollTop ?? 0;
   124→    for (let i = 0; i < list.length; i++) {
   125→      const rect = itemRefs.current.get(list[i].id)?.getBoundingClientRect();
   126→      if (!rect) continue;
   127→      if (clientY < rect.top + rect.height / 2) {
   128→        return { index: i, y: origin ? (rect.top - origin.top) / zoomRef.current + scrollTop : 0 };
   129→      }
   130→    }
   131→    const last = list.length > 0 ? itemRefs.current.get(list[list.length - 1].id)?.getBoundingClientRect() : null;
   132→    return { index: list.length, y: origin && last ? (last.bottom - origin.top) / zoomRef.current + scrollTop : 0 };
   133→  }, []);
   134→
   135→  /** 自由布局吸附：其它组件边缘/中心 + 屏幕边缘/水平中心 + 4px 网格 */
   136→  const snapPos = useCallback((x: number, y: number, movingId: string | null, w: number, h: number) => {
   137→    const b = useBuilder.getState();
   138→    const comps = b.pages.find((p) => p.id === b.currentPageId)?.components ?? [];
   139→    const T = 6;
   140→    const vCands: { p: number; line: number }[] = [
   141→      { p: 0, line: 0 },
   142→      { p: (SCREEN_W - w) / 2, line: SCREEN_W / 2 },
   143→      { p: SCREEN_W - w, line: SCREEN_W },
   144→    ];
   145→    const hCands: { p: number; line: number }[] = [{ p: 0, line: 0 }];
   146→    comps.forEach((c) => {
   147→      if (c.id === movingId) return;
   148→      const cx = c.x ?? 0, cy = c.y ?? 0, cw = c.w ?? 355, ch = c.h ?? 56;
   149→      vCands.push({ p: cx, line: cx }, { p: cx + cw / 2 - w / 2, line: cx + cw / 2 }, { p: cx + cw - w, line: cx + cw });
   150→      hCands.push({ p: cy, line: cy }, { p: cy + ch / 2 - h / 2, line: cy + ch / 2 }, { p: cy + ch - h, line: cy + ch }, { p: cy + ch, line: cy + ch });
   151→    });
   152→    let nx = x, ny = y;
   153→    const g: { v: number[]; h: number[] } = { v: [], h: [] };
   154→    for (const c of vCands) {
   155→      if (Math.abs(x - c.p) <= T) { nx = Math.max(0, Math.min(c.p, SCREEN_W - w)); g.v.push(c.line); break; }
   156→    }
   157→    for (const c of hCands) {
   158→      if (Math.abs(y - c.p) <= T) { ny = Math.max(0, c.p); g.h.push(c.line); break; }
   159→    }
   160→    if (g.v.length === 0) nx = Math.round(x / SNAP) * SNAP;
   161→    if (g.h.length === 0) ny = Math.round(y / SNAP) * SNAP;
   162→    return { x: Math.max(0, Math.min(nx, SCREEN_W - w)), y: Math.max(0, ny), guides: g };
   163→  }, []);
   164→
   165→  /* ---------- 拖拽过程 ---------- */
   166→
   167→  const handleDragMove = useCallback((clientX: number, clientY: number) => {
   168→    const st = useDnd.getState();
   169→
   170→    /* 缩放手柄拖拽中 */
   171→    if (resizeRef.current) {
   172→      const r = resizeRef.current;
   173→      setPosBadge(null);
   174→      const dx = (clientX - r.startX) / zoomRef.current;
   175→      const dy = (clientY - r.startY) / zoomRef.current;
   176→      let { x, y, w, h } = r.rect;
   177→      if (r.dir.includes('e')) w = r.rect.w + dx;
   178→      if (r.dir.includes('s')) h = r.rect.h + dy;
   179→      if (r.dir.includes('w')) { w = r.rect.w - dx; x = r.rect.x + dx; }
   180→      if (r.dir.includes('n')) { h = r.rect.h - dy; y = r.rect.y + dy; }
   181→      if (dirClampWest(r.dir) && x < 0) { w = r.rect.w + r.rect.x; x = 0; }
   182→      if (dirClampNorth(r.dir) && y < 0) { h = r.rect.h + r.rect.y; y = 0; }
   183→      w = Math.max(MIN_W, w);
   184→      h = Math.max(MIN_H, h);
   185→      if (x + w > SCREEN_W) w = SCREEN_W - x;
   186→      x = Math.round(x / SNAP) * SNAP;
   187→      y = Math.round(Math.max(0, y) / SNAP) * SNAP;
   188→      w = Math.round(w / SNAP) * SNAP;
   189→      h = Math.round(h / SNAP) * SNAP;
   190→      useBuilder.getState().setWidgetRect(r.id, { x, y, w, h }, true);
   191→      setSizeBadge(`${w} × ${h}`);
   192→      return;
   193→    }
   194→
   195→    if (!insideContent(clientX, clientY)) {
   196→      setIndicator(null);
   197→      setFreePreview(null);
   198→      freePreviewRef.current = null;
   199→      setGuides(EMPTY_GUIDES);
   200→      setPosBadge(null);
   201→      return;
   202→    }
   203→
   204→    if (isFreeRef.current) {
   205→      setIndicator(null);
   206→      setPosBadge(null);
   207→      if (st.kind === 'new') {
   208→        const def = getWidget(st.widgetType ?? '');
   209→        const w0 = def?.fullBleed ? SCREEN_W : 355;
   210→        const { x, y } = toLocal(clientX, clientY);
   211→        const snapped = snapPos(x - w0 / 2, y - 24, null, w0, 56);
   212→        const preview = { x: snapped.x, y: snapped.y, w: w0 };
   213→        freePreviewRef.current = preview;
   214→        setFreePreview(preview);
   215→        setGuides(snapped.guides);
   216→      } else if (st.kind === 'move') {
   217→        const id = st.moveId!;
   218→        const b = useBuilder.getState();
   219→        const comp = b.pages
   220→          .find((p) => p.id === b.currentPageId)
   221→          ?.components.find((c) => c.id === id);
   222→        if (!comp) return;
   223→        if (!dragMetaRef.current.batch) {
   224→          b.beginBatch();
   225→          dragMetaRef.current.batch = true;
   226→        }
   227→        const cw = comp.w ?? 355;
   228→        const ch = comp.h ?? 56;
   229→        const { x, y } = toLocal(clientX, clientY);
   230→        const snapped = snapPos(x - dragMetaRef.current.offX, y - dragMetaRef.current.offY, id, cw, ch);
   231→        b.setWidgetRect(id, { x: snapped.x, y: snapped.y }, true);
   232→        setPosBadge(`${Math.round(snapped.x)} · ${Math.round(snapped.y)}`);
   233→        setGuides(snapped.guides);
   234→        setFreePreview(null);
   235→        freePreviewRef.current = null;
   236→      }
   237→    } else {
   238→      setFreePreview(null);
   239→      freePreviewRef.current = null;
   240→      setGuides(EMPTY_GUIDES);
   241→      setPosBadge(null);
   242→      setIndicator(computeDrop(clientY));
   243→      /* 流式拖动：浮出提示（结束后再保留 6 秒，方便点击切换） */
   244→      if (st.started && st.kind === 'move') {
   245→        if (flowHintTimer.current) clearTimeout(flowHintTimer.current);
   246→        setFlowDragHint(true);
   247→      }
   248→    }
   249→  }, [computeDrop, insideContent, snapPos, toLocal]);
   250→
   251→  const handleUp = useCallback((clientX: number, clientY: number) => {
   252→    const st = useDnd.getState();
   253→    const b = useBuilder.getState();
   254→    const pageNow = b.pages.find((p) => p.id === b.currentPageId);
   255→    const freeNow = pageNow?.layout === 'free';
   256→
   257→    /* 缩放结束：合并历史 */
   258→    if (resizeRef.current) {
   259→      if (dragMetaRef.current.batch) b.endBatch();
   260→      resizeRef.current = null;
   261→      setSizeBadge(null);
   262→      dragMetaRef.current = { offX: 0, offY: 0, batch: false };
   263→      useDnd.getState().end();
   264→      return;
   265→    }
   266→
   267→    if (st.started) {
   268→      suppressClickRef.current = true;
   269→      useDnd.setState({ suppressNextClick: true });
   270→      setTimeout(() => { suppressClickRef.current = false; }, 90);
   271→    }
   272→
   273→    if (st.started && st.kind === 'new' && st.widgetType) {
   274→      if (insideContent(clientX, clientY)) {
   275→        if (freeNow) {
   276→          const { x, y } = toLocal(clientX, clientY);
   277→          const def = getWidget(st.widgetType);
   278→          const w0 = def?.fullBleed ? SCREEN_W : 355;
   279→          const fp = freePreviewRef.current;
   280→          const rx = fp?.x ?? Math.max(0, Math.round((x - w0 / 2) / SNAP) * SNAP);
   281→          const ry = fp?.y ?? Math.max(0, Math.round((y - 24) / SNAP) * SNAP);
   282→          b.addWidget(st.widgetType, undefined, { x: rx, y: ry, w: w0 });
   283→        } else {
   284→          b.addWidget(st.widgetType, computeDrop(clientY).index);
   285→        }
   286→      }
   287→    } else if (st.started && st.kind === 'move' && st.moveId) {
   288→      if (freeNow) {
   289→        if (dragMetaRef.current.batch) b.endBatch();
   290→      } else if (insideContent(clientX, clientY)) {
   291→        /* 仅画布内松手才排序；拖出画布松手视为取消 */
   292→        b.moveWidget(st.moveId, computeDrop(clientY).index);
   293→      }
   294→    }
   295→
   296→    dragMetaRef.current = { offX: 0, offY: 0, batch: false };
   297→    setIndicator(null);
   298→    setFreePreview(null);
   299→    freePreviewRef.current = null;
   300→    setGuides(EMPTY_GUIDES);
   301→    setSizeBadge(null);
   302→    setPosBadge(null);
   303→    /* 流式拖动结束后：提示再保留 6 秒，之后自动隐藏 */
   304→    if (flowHintTimer.current) clearTimeout(flowHintTimer.current);
   305→    flowHintTimer.current = setTimeout(() => setFlowDragHint(false), 6000);
   306→    useDnd.getState().end();
   307→  }, [computeDrop, insideContent, toLocal]);
   308→
   309→  /* 拖拽期间挂全局指针监听 */
   310→  useEffect(() => {
   311→    if (!dndActive) return;
   312→    const onMove = (e: PointerEvent) => {
   313→      useDnd.getState().move(e.clientX, e.clientY);
   314→      if (!useDnd.getState().started) return;
   315→      handleDragMove(e.clientX, e.clientY);
   316→    };
   317→    const onUp = (e: PointerEvent) => handleUp(e.clientX, e.clientY);
   318→    window.addEventListener('pointermove', onMove);
   319→    window.addEventListener('pointerup', onUp, true);
   320→    window.addEventListener('pointercancel', onUp, true);
   321→    return () => {
   322→      window.removeEventListener('pointermove', onMove);
   323→      window.removeEventListener('pointerup', onUp, true);
   324→      window.removeEventListener('pointercancel', onUp, true);
   325→    };
   326→  }, [dndActive, handleDragMove, handleUp]);
   327→
   328→  /* ---------- 布局切换 ---------- */
   329→
   330→  /** 启用自由布局：已有坐标直接切；缺坐标（旧数据）先按当前渲染堆叠生成坐标 */
   331→  const enableFree = useCallback(() => {
   332→    const b = useBuilder.getState();
   333→    const pg = b.pages.find((p) => p.id === b.currentPageId);
   334→    if (!pg || pg.layout === 'free') return;
   335→    if (!missingCoords(pg.components)) {
   336→      b.updatePage(pg.id, { layout: 'free' });
   337→      setFlowDragHint(false);
   338→      return;
   339→    }
   340→    const rects: Record<string, { x: number; y: number; w: number }> = {};
   341→    let y = 12;
   342→    pg.components.forEach((c) => {
   343→      const el = itemRefs.current.get(c.id);
   344→      const h = el ? el.offsetHeight : 64;
   345→      const full = getWidget(c.type)?.fullBleed;
   346→      rects[c.id] = { x: full ? 0 : 10, y, w: full ? SCREEN_W : 355 };
   347→      y += h + 12;
   348→    });
   349→    b.convertToFreeLayout(rects);
   350→    setFlowDragHint(false);
   351→    toast('已启用自由布局', { description: '拖动组件任意摆放，选中后拖手柄调整大小' });
   352→  }, []);
   353→
   354→  /** 拖拽进行中一键切换：先干净结束拖拽，避免状态错乱 */
   355→  const enableFreeNow = useCallback(() => {
   356→    useDnd.getState().end();
   357→    setIndicator(null);
   358→    setFreePreview(null);
   359→    freePreviewRef.current = null;
   360→    setGuides(EMPTY_GUIDES);
   361→    setPosBadge(null);
   362→    dragMetaRef.current = { offX: 0, offY: 0, batch: false };
   363→    resizeRef.current = null;
   364→    enableFree();
   365→  }, [enableFree]);
   366→
   367→  const setLayoutMode = useCallback(
   368→    (mode: 'flow' | 'free') => {
   369→      const b = useBuilder.getState();
   370→      const pg = b.pages.find((p) => p.id === b.currentPageId);
   371→      if (!pg) return;
   372→      if (mode === 'free') {
   373→        enableFree();
   374→      } else if (pg.layout !== 'flow') {
   375→        b.updatePage(pg.id, { layout: 'flow' });
   376→      }
   377→    },
   378→    [enableFree]
   379→  );
   380→
   381→  /* 旧页面自动升级为自由布局（显式选择流式的页面除外），视觉无感、不记历史
   382→     - 旧数据 layout 为空：直接采用自由布局（已有坐标则原样保留，还原之前的自由排版） */
   383→  useEffect(() => {
   384→    if (!page || page.layout === 'flow') return;
   385→    if (page.layout === 'free' && !missingCoords(page.components)) return;
   386→    const comps = page.components;
   387→    const b = useBuilder.getState();
   388→    if (comps.length === 0 || !missingCoords(comps)) {
   389→      b.migrateToFree(page.id);
   390→      return;
   391→    }
   392→    const rects: Record<string, { x: number; y: number; w: number }> = {};
   393→    let y = 12;
   394→    comps.forEach((c) => {
   395→      const el = itemRefs.current.get(c.id);
   396→      const h = el ? el.offsetHeight : 64;
   397→      const full = getWidget(c.type)?.fullBleed;
   398→      rects[c.id] = { x: full ? 0 : 10, y, w: full ? SCREEN_W : 355 };
   399→      y += h + 12;
   400→    });
   401→    b.migrateToFree(page.id, rects);
   402→    if (!freeMigrateToastShown) {
   403→      freeMigrateToastShown = true;
   404→      toast('已自动启用自由布局', { description: '页面上的组件现在可以任意拖放位置了' });
   405→    }
   406→  }, [page?.id, page?.layout, page?.components]);
   407→
   408→  /* 自由布局：方向键微调选中组件（Shift = 10px 步进） */
   409→  useEffect(() => {
   410→    if (!isFreeMode) return;
   411→    const onKey = (e: KeyboardEvent) => {
   412→      const el = document.activeElement as HTMLElement | null;
   413→      if (el && (el.tagName === 'INPUT' || el.tagName === 'TEXTAREA' || el.isContentEditable)) return;
   414→      const dv: Record<string, [number, number]> = {
   415→        ArrowLeft: [-1, 0], ArrowRight: [1, 0], ArrowUp: [0, -1], ArrowDown: [0, 1],
   416→      };
   417→      const d = dv[e.key];
   418→      if (!d) return;
   419→      const b = useBuilder.getState();
   420→      if (useDnd.getState().active || !b.selectedWidgetId) return;
   421→      const comp = b.pages
   422→        .find((p) => p.id === b.currentPageId)
