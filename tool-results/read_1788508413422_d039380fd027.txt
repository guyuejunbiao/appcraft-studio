     1→'use client';
     2→
     3→import { create } from 'zustand';
     4→import {
     5→  DEFAULT_THEME, deepClone, uid,
     6→  type ProjectData, type ProjectSummary, type PublishSummary,
     7→  type PageData, type ConnectionData, type WidgetInstance, type ThemeConfig, type AppSnapshot,
     8→} from './types';
     9→import { getWidget } from '@/components/widgets/registry';
    10→
    11→export type BuilderView = 'home' | 'editor' | 'flow' | 'preview';
    12→
    13→interface HistorySnap {
    14→  pages: PageData[];
    15→  connections: ConnectionData[];
    16→}
    17→
    18→/** 复制/粘贴剪贴板（会话级，跨页面也可粘贴） */
    19→let widgetClipboard: { type: string; props: Record<string, any>; width?: WidgetInstance['width']; align?: WidgetInstance['align'] } | null = null;
    20→
    21→interface BuilderState {
    22→  view: BuilderView;
    23→  /* 首页数据 */
    24→  projects: ProjectSummary[];
    25→  publishes: PublishSummary[];
    26→  homeLoading: boolean;
    27→  homeLoaded: boolean;
    28→  /* 当前项目 */
    29→  project: ProjectData | null;
    30→  pages: PageData[];
    31→  connections: ConnectionData[];
    32→  currentPageId: string | null;
    33→  selectedWidgetId: string | null;
    34→  /** 多选集合（含 selectedWidgetId；Shift 点选/框选/批量操作） */
    35→  selectedIds: string[];
    36→  dirty: boolean;
    37→  saving: boolean;
    38→  lastSavedAt: string | null;
    39→  /* 拖拽/批量编辑期间的临时快照（合并为一条历史） */
    40→  batchSnap: HistorySnap | null;
    41→  /* 上架 */
    42→  publishOpen: boolean;
    43→  previewSnapshot: (AppSnapshot & { version: number }) | null;
    44→  /* 历史 */
    45→  past: HistorySnap[];
    46→  future: HistorySnap[];
    47→  /* 动作 */
    48→  loadHome: () => Promise<void>;
    49→  createProject: (name: string, description: string, templateId: string) => Promise<void>;
    50→  deleteProject: (id: string) => Promise<void>;
    51→  renameProject: (id: string, name: string) => Promise<void>;
    52→  openProject: (id: string) => Promise<void>;
    53→  setView: (v: BuilderView) => void;
    54→  addWidget: (type: string, index?: number, rect?: { x: number; y: number; w: number }) => void;
    55→  moveWidget: (id: string, toIndex: number) => void;
    56→  removeWidget: (id: string) => void;
    57→  /** 批量删除（一条历史；联动清理相关连接） */
    58→  removeWidgets: (ids: string[]) => void;
    59→  duplicateWidget: (id: string) => void;
    60→  /** 批量复制：返回新 id 列表（一条历史，整体错开 16px） */
    61→  duplicateWidgets: (ids: string[]) => string[];
    62→  /** 复制组件到剪贴板（Ctrl+C） */
    63→  copyWidget: (id: string) => void;
    64→  /** 粘贴剪贴板组件（Ctrl+V）；自由布局错开 12px，流式追加到末尾 */
    65→  pasteWidget: () => void;
    66→  /** 图层置顶/置底/上移一层/下移一层（数组序即层叠序，流式下置顶置底等于移到末尾/开头） */
    67→  reorderWidget: (id: string, pos: 'front' | 'back' | 'up' | 'down') => void;
    68→  updateWidget: (id: string, patch: Partial<WidgetInstance>) => void;
    69→  updateWidgetProps: (id: string, props: Record<string, any>) => void;
    70→  /** 自由布局：更新位置/大小（transient=true 时不记录历史，用于拖拽过程） */
    71→  setWidgetRect: (id: string, rect: Partial<Pick<WidgetInstance, 'x' | 'y' | 'w' | 'h'>>, transient?: boolean) => void;
    72→  /** 多选批量更新矩形（组拖拽用，transient 同上） */
    73→  setWidgetsRects: (map: Record<string, Partial<Pick<WidgetInstance, 'x' | 'y' | 'w' | 'h'>>>, transient?: boolean) => void;
    74→  /** 多选对齐（选择集包围盒）：left/hcenter/right/top/vcenter/bottom */
    75→  alignWidgets: (mode: 'left' | 'hcenter' | 'right' | 'top' | 'vcenter' | 'bottom') => void;
    76→  /** 多选等间距分布（需要画布测量尺寸，sizes 缺省用存储 h/w） */
    77→  distributeWidgets: (axis: 'h' | 'v', sizes?: Record<string, { w?: number; h?: number }>) => void;
    78→  /** 批量编辑：拖拽/缩放开始时调用，结束时 endBatch 合并为一条历史 */
    79→  beginBatch: () => void;
    80→  endBatch: () => void;
    81→  /** 流式 → 自由布局迁移（rects 由画布测量后传入，记录历史） */
    82→  convertToFreeLayout: (rects: Record<string, { x: number; y: number; w: number }>) => void;
    83→  /** 旧数据自动升级为自由布局（不记历史，用户无感知迁移） */
    84→  migrateToFree: (pageId: string, rects?: Record<string, { x: number; y: number; w: number }>) => void;
    85→  select: (id: string | null, additive?: boolean) => void;
    86→  /** 框选/批量设置选中集合（同时更新 primary） */
    87→  selectMany: (ids: string[]) => void;
    88→  /** 清空多选（保留 primary 不变或一并清空） */
    89→  clearMulti: () => void;
    90→  setCurrentPage: (id: string) => void;
    91→  addPage: () => void;
    92→  duplicatePage: (id: string) => void;
    93→  updatePage: (id: string, patch: Partial<PageData>) => void;
    94→  removePage: (id: string) => void;
    95→  setFlowPos: (id: string, x: number, y: number) => void;
    96→  addConnection: (c: Omit<ConnectionData, 'id' | 'action'>) => void;
    97→  removeConnection: (id: string) => void;
    98→  updateTheme: (patch: Partial<ThemeConfig>) => void;
    99→  undo: () => void;
   100→  redo: () => void;
   101→  save: () => Promise<boolean>;
   102→  setPublishOpen: (b: boolean) => void;
   103→  publish: (name: string, description: string) => Promise<number | null>;
   104→  openPublishPreview: (publishId: string) => Promise<void>;
   105→  exitPublishPreview: () => void;
   106→  exportProject: () => void;
   107→  importProject: (payload: unknown) => Promise<boolean>;
   108→}
   109→
   110→const MAX_HISTORY = 60;
   111→
   112→export const useBuilder = create<BuilderState>((set, get) => {
   113→  /** 提交一次变更：记录历史 + 标脏 */
   114→  const commit = (fn: (s: { pages: PageData[]; connections: ConnectionData[] }) => Partial<BuilderState>) => {
   115→    const { pages, connections, past } = get();
   116→    const snap: HistorySnap = { pages: deepClone(pages), connections: deepClone(connections) };
   117→    set({
   118→      past: [...past.slice(-MAX_HISTORY), snap],
   119→      future: [],
   120→      dirty: true,
   121→      ...fn({ pages, connections }),
   122→    });
   123→  };
   124→
   125→  const currentPage = () => get().pages.find((p) => p.id === get().currentPageId);
   126→
   127→  return {
   128→    view: 'home',
   129→    projects: [],
   130→    publishes: [],
   131→    homeLoading: false,
   132→    homeLoaded: false,
   133→    project: null,
   134→    pages: [],
   135→    connections: [],
   136→    currentPageId: null,
   137→    selectedWidgetId: null,
   138→    selectedIds: [],
   139→    dirty: false,
   140→    saving: false,
   141→    lastSavedAt: null,
   142→    batchSnap: null,
   143→    publishOpen: false,
   144→    previewSnapshot: null,
   145→    past: [],
   146→    future: [],
   147→
   148→    async loadHome() {
   149→      set({ homeLoading: true });
   150→      try {
   151→        const [projects, publishes] = await Promise.all([
   152→          fetch('/api/projects').then((r) => r.json()),
   153→          fetch('/api/publishes').then((r) => r.json()),
   154→        ]);
   155→        set({ projects, publishes, homeLoaded: true });
   156→      } finally {
   157→        set({ homeLoading: false });
   158→      }
   159→    },
   160→
   161→    async createProject(name, description, templateId) {
   162→      const res = await fetch('/api/projects', {
   163→        method: 'POST',
   164→        headers: { 'Content-Type': 'application/json' },
   165→        body: JSON.stringify({ name, description, templateId }),
   166→      });
   167→      const { id } = await res.json();
   168→      await get().loadHome();
   169→      await get().openProject(id);
   170→    },
   171→
   172→    async deleteProject(id) {
   173→      await fetch(`/api/projects/${id}`, { method: 'DELETE' });
   174→      if (get().project?.id === id) set({ project: null, pages: [], connections: [], currentPageId: null });
   175→      await get().loadHome();
   176→    },
   177→
   178→    async renameProject(id, name) {
   179→      const { projects } = get();
   180→      set({ projects: projects.map((p) => (p.id === id ? { ...p, name } : p)) });
   181→      if (get().project?.id === id) {
   182→        set({ project: { ...get().project!, name } });
   183→        get().save();
   184→      }
   185→    },
   186→
   187→    async openProject(id) {
   188→      const res = await fetch(`/api/projects/${id}`);
   189→      if (!res.ok) return;
   190→      const data: ProjectData = await res.json();
   191→      const home = data.pages.find((p) => p.isHome);
   192→      set({
   193→        project: data,
   194→        pages: data.pages,
   195→        connections: data.connections,
   196→        currentPageId: home?.id || data.pages[0]?.id || null,
   197→        selectedWidgetId: null,
   198→        selectedIds: [],
   199→        dirty: false,
   200→        saving: false,
   201→        lastSavedAt: null,
   202→        past: [],
   203→        future: [],
   204→        previewSnapshot: null,
   205→        view: 'editor',
   206→      });
   207→    },
   208→
   209→    setView(v) {
   210→      set({ view: v });
   211→      if (v === 'home') {
   212→        get().loadHome();
   213→      }
   214→    },
   215→
   216→    addWidget(type, index, rect) {
   217→      const page = currentPage();
   218→      if (!page) return;
   219→      const def = getWidget(type);
   220→      if (!def) return;
   221→      const w: WidgetInstance = {
   222→        id: uid(),
   223→        type,
   224→        props: deepClone(def.defaultProps),
   225→        width: 'full',
   226→        align: 'left',
   227→        mt: 0,
   228→        mb: 8,
   229→      };
   230→      /* 自由布局：按落点/智能位置放置（不钳制，可摆到首屏以下，画布自动增高） */
   231→      if (page.layout === 'free') {
   232→        const fullW = def.fullBleed ? 375 : 355;
   233→        const maxBottom = page.components.reduce(
   234→          (m, c) => Math.max(m, (c.y ?? 0) + (c.h ?? 64)),
   235→          0
   236→        );
   237→        w.x = rect?.x ?? (def.fullBleed ? 0 : 10);
   238→        w.y = rect?.y ?? maxBottom + 12;
   239→        w.w = rect?.w ?? fullW;
   240→        w.x = Math.max(0, Math.min(w.x ?? 0, 375 - (w.w ?? fullW)));
   241→      }
   242→      commit(({ pages }) => ({
   243→        pages: pages.map((p) =>
   244→          p.id === page.id
   245→            ? {
   246→                ...p,
   247→                components: index === undefined
   248→                  ? [...p.components, w]
   249→                  : [...p.components.slice(0, index), w, ...p.components.slice(index)],
   250→              }
   251→            : p
   252→        ),
   253→      }));
   254→      set({ selectedWidgetId: w.id, selectedIds: [w.id] });
   255→    },
   256→
   257→    moveWidget(id, toIndex) {
   258→      const page = currentPage();
   259→      if (!page) return;
   260→      commit(({ pages }) => ({
   261→        pages: pages.map((p) => {
   262→          if (p.id !== page.id) return p;
   263→          const from = p.components.findIndex((c) => c.id === id);
   264→          if (from < 0) return p;
   265→          const list = [...p.components];
   266→          const [item] = list.splice(from, 1);
   267→          const target = from < toIndex ? toIndex - 1 : toIndex;
   268→          list.splice(Math.max(0, Math.min(target, list.length)), 0, item);
   269→          return { ...p, components: list };
   270→        }),
   271→      }));
   272→    },
   273→
   274→    removeWidget(id) {
   275→      const page = currentPage();
   276→      if (!page) return;
   277→      commit(({ pages, connections }) => ({
   278→        pages: pages.map((p) => (p.id === page.id ? { ...p, components: p.components.filter((c) => c.id !== id) } : p)),
   279→        connections: connections.filter((c) => !(c.fromPageId === page.id && c.fromWidgetId === id)),
   280→      }));
   281→      if (get().selectedWidgetId === id) set({ selectedWidgetId: null });
   282→      set({ selectedIds: get().selectedIds.filter((x) => x !== id) });
   283→    },
   284→
   285→    removeWidgets(ids) {
   286→      const page = currentPage();
   287→      if (!page) return;
   288→      const set2 = new Set(ids);
   289→      commit(({ pages, connections }) => ({
   290→        pages: pages.map((p) => (p.id === page.id ? { ...p, components: p.components.filter((c) => !set2.has(c.id) || c.locked) } : p)),
   291→        connections: connections.filter((c) => !(c.fromPageId === page.id && set2.has(c.fromWidgetId))),
   292→      }));
   293→      const remaining = get().pages.find((p) => p.id === page.id)?.components.map((c) => c.id) ?? [];
   294→      set({
   295→        selectedIds: get().selectedIds.filter((x) => remaining.includes(x)),
   296→        selectedWidgetId: remaining.includes(get().selectedWidgetId ?? '') ? get().selectedWidgetId : null,
   297→      });
   298→    },
   299→
   300→    duplicateWidget(id) {
   301→      const [copyId] = get().duplicateWidgets([id]);
   302→      if (copyId) set({ selectedWidgetId: copyId, selectedIds: [copyId] });
   303→    },
   304→
   305→    duplicateWidgets(ids) {
   306→      const page = currentPage();
   307→      if (!page || ids.length === 0) return [];
   308→      const OFFSET = 16;
   309→      const srcList = page.components;
   310→      const pairs = ids
   311→        .map((id) => srcList.find((c) => c.id === id))
   312→        .filter((c): c is WidgetInstance => !!c && !c.locked)
   313→        .map((src) => {
   314→          const copy: WidgetInstance = { ...deepClone(src), id: uid() };
   315→          if (page.layout === 'free') {
   316→            const cw = copy.w ?? 355;
   317→            copy.x = Math.max(0, Math.min((copy.x ?? 0) + OFFSET, Math.max(0, 375 - cw)));
   318→            copy.y = (copy.y ?? 0) + OFFSET;
   319→          }
   320→          return { srcId: src.id, clone: copy };
   321→        });
   322→      if (pairs.length === 0) return [];
   323→      const created = pairs.map((p) => p.clone.id);
   324→      commit(({ pages }) => ({
   325→        pages: pages.map((p) => {
   326→          if (p.id !== page.id) return p;
   327→          const list = [...p.components];
   328→          /* 倒序插入：插在各自源组件后面，保持图层相对顺序 */
   329→          for (let i = pairs.length - 1; i >= 0; i--) {
   330→            const idx = list.findIndex((c) => c.id === pairs[i].srcId);
   331→            list.splice(idx >= 0 ? idx + 1 : list.length, 0, pairs[i].clone);
   332→          }
   333→          return { ...p, components: list };
   334→        }),
   335→      }));
   336→      set({ selectedWidgetId: created[created.length - 1], selectedIds: created });
   337→      return created;
   338→    },
   339→
   340→    copyWidget(id) {
   341→      const page = currentPage();
   342→      if (!page) return;
   343→      const src = page.components.find((c) => c.id === id);
   344→      if (!src) return;
   345→      widgetClipboard = { type: src.type, props: deepClone(src.props), width: src.width, align: src.align };
   346→    },
   347→
   348→    pasteWidget() {
   349→      if (!widgetClipboard) return;
   350→      const page = currentPage();
   351→      if (!page) return;
   352→      const def = getWidget(widgetClipboard.type);
   353→      if (!def) return;
   354→      const w: WidgetInstance = {
   355→        id: uid(),
   356→        type: widgetClipboard.type,
   357→        props: deepClone(widgetClipboard.props),
   358→        width: widgetClipboard.width ?? 'full',
   359→        align: widgetClipboard.align ?? 'left',
   360→        mt: 0,
   361→        mb: 8,
   362→      };
   363→      if (page.layout === 'free') {
   364→        const fullW = def.fullBleed ? 375 : 355;
   365→        const maxBottom = page.components.reduce(
   366→          (m, c) => Math.max(m, (c.y ?? 0) + (c.h ?? 64)),
   367→          0
   368→        );
   369→        w.x = def.fullBleed ? 0 : 10;
   370→        w.y = maxBottom + 12;
   371→        w.w = fullW;
   372→      }
   373→      commit(({ pages }) => ({
   374→        pages: pages.map((p) => (p.id === page.id ? { ...p, components: [...p.components, w] } : p)),
   375→      }));
   376→      set({ selectedWidgetId: w.id, selectedIds: [w.id] });
   377→    },
   378→
   379→    reorderWidget(id, pos) {
   380→      const page = currentPage();
   381→      if (!page) return;
   382→      commit(({ pages }) => ({
   383→        pages: pages.map((p) => {
   384→          if (p.id !== page.id) return p;
   385→          const from = p.components.findIndex((c) => c.id === id);
   386→          if (from < 0) return p;
   387→          const list = [...p.components];
   388→          const [item] = list.splice(from, 1);
   389→          if (pos === 'front') list.push(item);
   390→          else if (pos === 'back') list.unshift(item);
   391→          else if (pos === 'up') list.splice(Math.min(from + 1, list.length), 0, item);
   392→          else list.splice(Math.max(from - 1, 0), 0, item);
   393→          return { ...p, components: list };
   394→        }),
   395→      }));
   396→    },
   397→
   398→    updateWidget(id, patch) {
   399→      const page = currentPage();
   400→      if (!page) return;
   401→      commit(({ pages }) => ({
   402→        pages: pages.map((p) =>
   403→          p.id === page.id
   404→            ? { ...p, components: p.components.map((c) => (c.id === id ? { ...c, ...patch } : c)) }
   405→            : p
   406→        ),
   407→      }));
   408→    },
   409→
   410→    updateWidgetProps(id, props) {
   411→      const page = currentPage();
   412→      if (!page) return;
   413→      commit(({ pages }) => ({
   414→        pages: pages.map((p) =>
   415→          p.id === page.id
   416→            ? { ...p, components: p.components.map((c) => (c.id === id ? { ...c, props: { ...c.props, ...props } } : c)) }
   417→            : p
   418→        ),
   419→      }));
   420→    },
   421→
   422→    setWidgetRect(id, rect, transient = false) {
   423→      const page = currentPage();
   424→      if (!page) return;
   425→      const patchPages = (pages: PageData[]) =>
   426→        pages.map((p) =>
   427→          p.id === page.id
   428→            ? { ...p, components: p.components.map((c) => (c.id === id ? { ...c, ...rect } : c)) }
   429→            : p
   430→        );
   431→      if (transient) {
   432→        set({ pages: patchPages(get().pages), dirty: true });
   433→      } else {
   434→        commit(({ pages }) => ({ pages: patchPages(pages) }));
   435→      }
   436→    },
   437→
   438→    setWidgetsRects(map, transient = false) {
   439→      const page = currentPage();
   440→      if (!page) return;
   441→      const patchPages = (pages: PageData[]) =>
   442→        pages.map((p) =>
   443→          p.id === page.id
   444→            ? { ...p, components: p.components.map((c) => (map[c.id] ? { ...c, ...map[c.id] } : c)) }
   445→            : p
   446→        );
   447→      if (transient) {
   448→        set({ pages: patchPages(get().pages), dirty: true });
   449→      } else {
   450→        commit(({ pages }) => ({ pages: patchPages(pages) }));
   451→      }
   452→    },
   453→
   454→    alignWidgets(mode) {
   455→      const page = currentPage();
   456→      if (!page || page.layout !== 'free') return;
   457→      const { selectedIds } = get();
   458→      if (selectedIds.length < 2) return;
   459→      const comps = page.components.filter((c) => selectedIds.includes(c.id) && !c.locked);
   460→      if (comps.length < 2) return;
   461→      const rects = comps.map((c) => ({
   462→        id: c.id,
   463→        x: c.x ?? 0,
   464→        y: c.y ?? 0,
   465→        w: c.w ?? 355,
   466→        h: c.h ?? 56,
   467→      }));
   468→      const minX = Math.min(...rects.map((r) => r.x));
   469→      const maxX = Math.max(...rects.map((r) => r.x + r.w));
   470→      const minY = Math.min(...rects.map((r) => r.y));
   471→      const maxY = Math.max(...rects.map((r) => r.y + r.h));
   472→      const map: Record<string, Partial<Pick<WidgetInstance, 'x' | 'y'>>> = {};
   473→      for (const r of rects) {
   474→        if (mode === 'left') map[r.id] = { x: minX };
   475→        else if (mode === 'right') map[r.id] = { x: maxX - r.w };
   476→        else if (mode === 'hcenter') map[r.id] = { x: Math.round((minX + maxX) / 2 - r.w / 2) };
   477→        else if (mode === 'top') map[r.id] = { y: minY };
   478→        else if (mode === 'bottom') map[r.id] = { y: maxY - r.h };
   479→        else if (mode === 'vcenter') map[r.id] = { y: Math.round((minY + maxY) / 2 - r.h / 2) };
   480→      }
   481→      get().setWidgetsRects(map);
   482→    },
   483→
   484→    distributeWidgets(axis, sizes) {
   485→      const page = currentPage();
   486→      if (!page || page.layout !== 'free') return;
   487→      const { selectedIds } = get();
   488→      if (selectedIds.length < 3) return;
   489→      const comps = page.components.filter((c) => selectedIds.includes(c.id) && !c.locked);
   490→      if (comps.length < 3) return;
   491→      const items = comps
   492→        .map((c) => ({
   493→          id: c.id,
   494→          x: c.x ?? 0,
   495→          y: c.y ?? 0,
   496→          w: sizes?.[c.id]?.w ?? c.w ?? 355,
   497→          h: sizes?.[c.id]?.h ?? c.h ?? 56,
   498→        }))
   499→        .sort((a, b) => (axis === 'h' ? a.x - b.x : a.y - b.y));
   500→      const first = items[0];
   501→      const last = items[items.length - 1];
   502→      if (axis === 'h') {
   503→        const span = last.x + last.w - first.x;
   504→        const totalW = items.reduce((s, r) => s + r.w, 0);
   505→        const gap = (span - totalW) / (items.length - 1);
   506→        if (!isFinite(gap) || gap < 0) return;
   507→        const map: Record<string, Partial<Pick<WidgetInstance, 'x' | 'y'>>> = {};
   508→        let cursor = first.x;
   509→        for (const r of items) {
   510→          if (r.id !== first.id && r.id !== last.id) map[r.id] = { x: Math.round(cursor) };
   511→          cursor += r.w + gap;
   512→        }
   513→        get().setWidgetsRects(map);
   514→      } else {
   515→        const span = last.y + last.h - first.y;
   516→        const totalH = items.reduce((s, r) => s + r.h, 0);
   517→        const gap = (span - totalH) / (items.length - 1);
   518→        if (!isFinite(gap) || gap < 0) return;
   519→        const map: Record<string, Partial<Pick<WidgetInstance, 'x' | 'y'>>> = {};
   520→        let cursor = first.y;
   521→        for (const r of items) {
   522→          if (r.id !== first.id && r.id !== last.id) map[r.id] = { y: Math.round(cursor) };
   523→          cursor += r.h + gap;
   524→        }
   525→        get().setWidgetsRects(map);
   526→      }
   527→    },

   849→/** 自动保存：脏状态 1.2s 后保存 */
   850→export function bindAutosave() {
   851→  let timer: ReturnType<typeof setTimeout> | null = null;
   852→  let stopped = false;
   853→  const unsub = useBuilder.subscribe((state) => {
   854→    if (stopped) return;
   855→    if (state.dirty && state.view !== 'home') {
   856→      if (timer) clearTimeout(timer);
   857→      timer = setTimeout(() => {
   858→        useBuilder.getState().save();
   859→      }, 1200);
   860→    }
   861→  });
   862→  return () => {
   863→    stopped = true;
   864→    if (timer) clearTimeout(timer);
   865→    unsub();
   866→  };
   867→}
