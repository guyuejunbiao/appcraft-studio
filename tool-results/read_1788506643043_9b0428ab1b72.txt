   126→   126→  const computeDrop = useCallback((clientY: number) => {

   142→   142→  /** 自由布局吸附：其它组件边缘/中心 + 屏幕边缘/水平中心 + 4px 网格 */
   143→   143→  const snapPos = useCallback((x: number, y: number, movingId: string | null, w: number, h: number) => {
   144→   144→    const b = useBuilder.getState();
   145→   145→    const comps = b.pages.find((p) => p.id === b.currentPageId)?.components ?? [];
   146→   146→    const T = 6;
   147→   147→    const vCands: { p: number; line: number }[] = [
   148→   148→      { p: 0, line: 0 },
   149→   149→      { p: (SCREEN_W - w) / 2, line: SCREEN_W / 2 },
   150→   150→      { p: SCREEN_W - w, line: SCREEN_W },
   151→   151→    ];
   152→   152→    const hCands: { p: number; line: number }[] = [{ p: 0, line: 0 }];
   153→   153→    comps.forEach((c) => {
   154→   154→      if (c.id === movingId) return;
   155→   155→      const cx = c.x ?? 0, cy = c.y ?? 0, cw = c.w ?? 355, ch = c.h ?? 56;
   156→   156→      vCands.push({ p: cx, line: cx }, { p: cx + cw / 2 - w / 2, line: cx + cw / 2 }, { p: cx + cw - w, line: cx + cw });
   157→   157→      hCands.push({ p: cy, line: cy }, { p: cy + ch / 2 - h / 2, line: cy + ch / 2 }, { p: cy + ch - h, line: cy + ch }, { p: cy + ch, line: cy + ch });
   158→   158→    });
   159→   159→    let nx = x, ny = y;
   160→   160→    const g: { v: number[]; h: number[] } = { v: [], h: [] };
   161→   161→    for (const c of vCands) {
   162→   162→      if (Math.abs(x - c.p) <= T) { nx = Math.max(0, Math.min(c.p, SCREEN_W - w)); g.v.push(c.line); break; }
   163→   163→    }
   164→   164→    for (const c of hCands) {
   165→   165→      if (Math.abs(y - c.p) <= T) { ny = Math.max(0, c.p); g.h.push(c.line); break; }
   166→   166→    }
   167→   167→    if (g.v.length === 0) nx = Math.round(x / SNAP) * SNAP;
   168→   168→    if (g.h.length === 0) ny = Math.round(y / SNAP) * SNAP;
   169→   169→    return { x: Math.max(0, Math.min(nx, SCREEN_W - w)), y: Math.max(0, ny), guides: g };
   170→   170→  }, []);

   184→   184→      if (r.dir.includes('e')) w = r.rect.w + dx;
   185→   185→      if (r.dir.includes('s')) h = r.rect.h + dy;
   186→   186→      if (r.dir.includes('w')) { w = r.rect.w - dx; x = r.rect.x + dx; }
   187→   187→      if (r.dir.includes('n')) { h = r.rect.h - dy; y = r.rect.y + dy; }
   188→   188→      if (dirClampWest(r.dir) && x < 0) { w = r.rect.w + r.rect.x; x = 0; }

   190→   190→      w = Math.max(MIN_W, w);
   191→   191→      h = Math.max(MIN_H, h);
   192→   192→      if (x + w > SCREEN_W) w = SCREEN_W - x;
   193→   193→      x = Math.round(x / SNAP) * SNAP;
   194→   194→      y = Math.round(Math.max(0, y) / SNAP) * SNAP;

   200→   200→    }
   201→   201→
   202→   202→    if (!insideContent(clientX, clientY)) {
   203→   203→      setIndicator(null);
   204→   204→      setFreePreview(null);
   205→   205→      freePreviewRef.current = null;
   206→   206→      setGuides(EMPTY_GUIDES);
   207→   207→      setPosBadge(null);
   208→   208→      return;
   209→   209→    }
   210→   210→

   212→   212→      setIndicator(null);
   213→   213→      setPosBadge(null);

   216→   216→        const { x, y } = toLocal(clientX, clientY);
   217→   217→        const rect = { x0: marqueeStartRef.current.x, y0: marqueeStartRef.current.y, x1: x, y1: y };
   218→   218→        setMarquee(rect);
   219→   219→        const bx0 = Math.min(rect.x0, rect.x1), bx1 = Math.max(rect.x0, rect.x1);
   220→   220→        const by0 = Math.min(rect.y0, rect.y1), by1 = Math.max(rect.y0, rect.y1);
   221→   221→        const b = useBuilder.getState();
   222→   222→        const hits: string[] = [];
   223→   223→        b.pages
   224→   224→          .find((p) => p.id === b.currentPageId)

   237→   237→      }

   255→   255→          b.beginBatch();

   257→   257→        }

   259→   259→        const gs = groupStartRef.current;
   260→   260→        if (gs && Object.keys(gs.map).length > 1) {

   279→   279→          freePreviewRef.current = null;

   282→   282→        const cw = comp.w ?? 355;
   283→   283→        const ch = comp.h ?? 56;
   284→   284→        const snapped = snapPos(x - dragMetaRef.current.offX, y - dragMetaRef.current.offY, id, cw, ch);

   286→   286→        setPosBadge(`${Math.round(snapped.x)} · ${Math.round(snapped.y)}`);
   287→   287→        setGuides(snapped.guides);
   288→   288→        setFreePreview(null);
   289→   289→        freePreviewRef.current = null;
   290→   290→      }
   291→   291→    } else {
   292→   292→      setFreePreview(null);
   293→   293→      freePreviewRef.current = null;
   294→   294→      setGuides(EMPTY_GUIDES);
   295→   295→      setPosBadge(null);
   296→   296→      setIndicator(computeDrop(clientY));
   297→   297→      /* 流式拖动：浮出提示（结束后再保留 6 秒，方便点击切换） */
   298→   298→      if (st.started && st.kind === 'move') {
   299→   299→        if (flowHintTimer.current) clearTimeout(flowHintTimer.current);
   300→   300→        setFlowDragHint(true);
   301→   301→      }
   302→   302→    }
   303→   303→  }, [computeDrop, insideContent, snapPos, toLocal]);
   304→   304→
   305→   305→  const handleUp = useCallback((clientX: number, clientY: number) => {
   306→   306→    const st = useDnd.getState();
   307→   307→    const b = useBuilder.getState();
   308→   308→    const pageNow = b.pages.find((p) => p.id === b.currentPageId);
   309→   309→    const freeNow = pageNow?.layout === 'free';
   310→   310→
   311→   311→    /* 框选结束：命中集已在移动中实时写入 selectedIds；全部落空则清空。
   312→   312→       同时抑制紧随的 click，避免容器 onClick 把框选结果清空 */
   313→   313→    if (st.kind === 'marquee') {
   314→   314→      if (st.started) {
   315→   315→        suppressClickRef.current = true;
   316→   316→        useDnd.setState({ suppressNextClick: true });
   317→   317→        setTimeout(() => { suppressClickRef.current = false; }, 90);
   318→   318→      }
   319→   319→      if (marqueeHitRef.current.length === 0) b.select(null);
   320→   320→      setMarquee(null);
   321→   321→      marqueeHitRef.current = [];
   322→   322→      useDnd.getState().end();
   323→   323→      return;
   324→   324→    }
   325→   325→
   326→   326→    /* 缩放结束：合并历史 */
   327→   327→    if (resizeRef.current) {
   328→   328→      if (dragMetaRef.current.batch) b.endBatch();
   329→   329→      resizeRef.current = null;
   330→   330→      setSizeBadge(null);
   331→   331→      dragMetaRef.current = { offX: 0, offY: 0, batch: false };
   332→   332→      useDnd.getState().end();
   333→   333→      return;
   334→   334→    }
   335→   335→
   336→   336→    if (st.started) {
   337→   337→      suppressClickRef.current = true;
   338→   338→      useDnd.setState({ suppressNextClick: true });
   339→   339→      setTimeout(() => { suppressClickRef.current = false; }, 90);
   340→   340→    }
   341→   341→
   342→   342→    if (st.started && st.kind === 'new' && st.widgetType) {
   343→   343→      if (insideContent(clientX, clientY)) {
   344→   344→        if (freeNow) {
   345→   345→          const { x, y } = toLocal(clientX, clientY);
   346→   346→          const def = getWidget(st.widgetType);
   347→   347→          const w0 = def?.fullBleed ? SCREEN_W : 355;
   348→   348→          const fp = freePreviewRef.current;
   349→   349→          const rx = fp?.x ?? Math.max(0, Math.round((x - w0 / 2) / SNAP) * SNAP);
   350→   350→          const ry = fp?.y ?? Math.max(0, Math.round((y - 24) / SNAP) * SNAP);
