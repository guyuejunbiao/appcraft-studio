   118→   118→const MAX_HISTORY = 60;
   119→   119→
   120→   120→export const useBuilder = create<BuilderState>((set, get) => {
   121→   121→  /** 提交一次变更：记录历史 + 标脏 */
   122→   122→  const commit = (fn: (s: { pages: PageData[]; connections: ConnectionData[] }) => Partial<BuilderState>) => {
   123→   123→    const { pages, connections, past } = get();
   124→   124→    const snap: HistorySnap = { pages: deepClone(pages), connections: deepClone(connections) };
   125→   125→    set({
   126→   126→      past: [...past.slice(-MAX_HISTORY), snap],
   127→   127→      future: [],
   128→   128→      dirty: true,
   129→   129→      ...fn({ pages, connections }),
   130→   130→    });
   131→   131→  };
   132→   132→
   133→   133→  const currentPage = () => get().pages.find((p) => p.id === get().currentPageId);
   134→   134→
   135→   135→  return {
   136→   136→    view: 'home',
   137→   137→    projects: [],
   138→   138→    publishes: [],
   139→   139→    homeLoading: false,
   140→   140→    homeLoaded: false,
   141→   141→    project: null,
   142→   142→    pages: [],
   143→   143→    connections: [],
   144→   144→    currentPageId: null,
   145→   145→    selectedWidgetId: null,
   146→   146→    selectedIds: [],
   147→   147→    dirty: false,
   148→   148→    saving: false,
   149→   149→    lastSavedAt: null,
   150→   150→    batchSnap: null,
   151→   151→    publishOpen: false,
   152→   152→    previewSnapshot: null,
   153→   153→    past: [],
   154→   154→    future: [],
   155→   155→
   156→   156→    async loadHome() {
   157→   157→      set({ homeLoading: true });
   158→   158→      try {
   159→   159→        const [projects, publishes] = await Promise.all([
   160→   160→          fetch('/api/projects').then((r) => r.json()),
   161→   161→          fetch('/api/publishes').then((r) => r.json()),
   162→   162→        ]);
   163→   163→        set({ projects, publishes, homeLoaded: true });
   164→   164→      } finally {
   165→   165→        set({ homeLoading: false });
   166→   166→      }
   167→   167→    },
   168→   168→
   169→   169→    async createProject(name, description, templateId) {
   170→   170→      const res = await fetch('/api/projects', {
   171→   171→        method: 'POST',
   172→   172→        headers: { 'Content-Type': 'application/json' },
   173→   173→        body: JSON.stringify({ name, description, templateId }),
   174→   174→      });
   175→   175→      const { id } = await res.json();
   176→   176→      await get().loadHome();
   177→   177→      await get().openProject(id);
   178→   178→    },
   179→   179→
   180→   180→    async deleteProject(id) {
   181→   181→      await fetch(`/api/projects/${id}`, { method: 'DELETE' });
   182→   182→      if (get().project?.id === id) set({ project: null, pages: [], connections: [], currentPageId: null });
   183→   183→      await get().loadHome();
   184→   184→    },
   185→   185→
   186→   186→    async renameProject(id, name) {
   187→   187→      const { projects } = get();
   188→   188→      set({ projects: projects.map((p) => (p.id === id ? { ...p, name } : p)) });
   189→   189→      if (get().project?.id === id) {
   190→   190→        set({ project: { ...get().project!, name } });
   191→   191→        get().save();
   192→   192→      }
   193→   193→    },
   194→   194→
   195→   195→    async openProject(id) {
   196→   196→      const res = await fetch(`/api/projects/${id}`);
   197→   197→      if (!res.ok) return;
   198→   198→      const data: ProjectData = await res.json();
   199→   199→      const home = data.pages.find((p) => p.isHome);
   200→   200→      set({
   201→   201→        project: data,
   202→   202→        pages: data.pages,
   203→   203→        connections: data.connections,
   204→   204→        currentPageId: home?.id || data.pages[0]?.id || null,
   205→   205→        selectedWidgetId: null,
   206→   206→        selectedIds: [],
   207→   207→        dirty: false,
   208→   208→        saving: false,
   209→   209→        lastSavedAt: null,
   210→   210→        past: [],
   211→   211→        future: [],
   212→   212→        previewSnapshot: null,
   213→   213→        view: 'editor',
   214→   214→      });
   215→   215→    },
   216→   216→
   217→   217→    setView(v) {
   218→   218→      set({ view: v });
   219→   219→      if (v === 'home') {
   220→   220→        get().loadHome();
   221→   221→      }
   222→   222→    },
   223→   223→
   224→   224→    addWidget(type, index, rect) {
   225→   225→      const page = currentPage();
   226→   226→      if (!page) return;
   227→   227→      const def = getWidget(type);
   228→   228→      if (!def) return;
   229→   229→      const w: WidgetInstance = {
   230→   230→        id: uid(),
   231→   231→        type,
   232→   232→        props: deepClone(def.defaultProps),
   233→   233→        width: 'full',
   234→   234→        align: 'left',
   235→   235→        mt: 0,
   236→   236→        mb: 8,
   237→   237→      };
   238→   238→      /* 自由布局：按落点/智能位置放置（不钳制，可摆到首屏以下，画布自动增高） */
   239→   239→      if (page.layout === 'free') {
   240→   240→        const fullW = def.fullBleed ? 375 : 355;
   241→   241→        const maxBottom = page.components.reduce(
   242→   242→          (m, c) => Math.max(m, (c.y ?? 0) + (c.h ?? 64)),
   243→   243→          0
   244→   244→        );
   245→   245→        w.x = rect?.x ?? (def.fullBleed ? 0 : 10);
   246→   246→        w.y = rect?.y ?? maxBottom + 12;
   247→   247→        w.w = rect?.w ?? fullW;
   248→   248→        w.x = Math.max(0, Math.min(w.x ?? 0, 375 - (w.w ?? fullW)));
   249→   249→      }
   250→   250→      commit(({ pages }) => ({
   251→   251→        pages: pages.map((p) =>
   252→   252→          p.id === page.id
   253→   253→            ? {
   254→   254→                ...p,
   255→   255→                components: index === undefined
   256→   256→                  ? [...p.components, w]
   257→   257→                  : [...p.components.slice(0, index), w, ...p.components.slice(index)],
   258→   258→              }
   259→   259→            : p
   260→   260→        ),
   261→   261→      }));
   262→   262→      set({ selectedWidgetId: w.id, selectedIds: [w.id] });
   263→   263→    },
   264→   264→
   265→   265→    moveWidget(id, toIndex) {
   266→   266→      const page = currentPage();
   267→   267→      if (!page) return;
   268→   268→      commit(({ pages }) => ({
   269→   269→        pages: pages.map((p) => {
   270→   270→          if (p.id !== page.id) return p;
   271→   271→          const from = p.components.findIndex((c) => c.id === id);
   272→   272→          if (from < 0) return p;
   273→   273→          const list = [...p.components];
   274→   274→          const [item] = list.splice(from, 1);
   275→   275→          const target = from < toIndex ? toIndex - 1 : toIndex;
   276→   276→          list.splice(Math.max(0, Math.min(target, list.length)), 0, item);
   277→   277→          return { ...p, components: list };
   278→   278→        }),
   279→   279→      }));
   280→   280→    },
   281→   281→
   282→   282→    removeWidget(id) {
   283→   283→      const page = currentPage();
   284→   284→      if (!page) return;
   285→   285→      commit(({ pages, connections }) => ({
   286→   286→        pages: pages.map((p) => (p.id === page.id ? { ...p, components: p.components.filter((c) => c.id !== id) } : p)),
   287→   287→        connections: connections.filter((c) => !(c.fromPageId === page.id && c.fromWidgetId === id)),
   288→   288→      }));
   289→   289→      if (get().selectedWidgetId === id) set({ selectedWidgetId: null });
   290→   290→      set({ selectedIds: get().selectedIds.filter((x) => x !== id) });
   291→   291→    },
   292→   292→
   293→   293→    removeWidgets(ids) {
   294→   294→      const page = currentPage();
   295→   295→      if (!page) return;
   296→   296→      const set2 = new Set(ids);
   297→   297→      commit(({ pages, connections }) => ({
   298→   298→        pages: pages.map((p) => (p.id === page.id ? { ...p, components: p.components.filter((c) => !set2.has(c.id) || c.locked) } : p)),
   299→   299→        connections: connections.filter((c) => !(c.fromPageId === page.id && set2.has(c.fromWidgetId))),
   300→   300→      }));
   301→   301→      const remaining = get().pages.find((p) => p.id === page.id)?.components.map((c) => c.id) ?? [];
   302→   302→      set({
   303→   303→        selectedIds: get().selectedIds.filter((x) => remaining.includes(x)),
   304→   304→        selectedWidgetId: remaining.includes(get().selectedWidgetId ?? '') ? get().selectedWidgetId : null,
   305→   305→      });
   306→   306→    },
   307→   307→
   308→   308→    duplicateWidget(id) {
   309→   309→      const [copyId] = get().duplicateWidgets([id]);
   310→   310→      if (copyId) set({ selectedWidgetId: copyId, selectedIds: [copyId] });
   311→   311→    },
   312→   312→
   313→   313→    duplicateWidgets(ids) {
   314→   314→      const page = currentPage();
   315→   315→      if (!page || ids.length === 0) return [];
   316→   316→      const OFFSET = 16;
   317→   317→      const srcList = page.components;
   318→   318→      const pairs = ids
   319→   319→        .map((id) => srcList.find((c) => c.id === id))
   320→   320→        .filter((c): c is WidgetInstance => !!c && !c.locked)
   321→   321→        .map((src) => {
   322→   322→          const copy: WidgetInstance = { ...deepClone(src), id: uid() };
   323→   323→          if (page.layout === 'free') {
   324→   324→            const cw = copy.w ?? 355;
   325→   325→            copy.x = Math.max(0, Math.min((copy.x ?? 0) + OFFSET, Math.max(0, 375 - cw)));
   326→   326→            copy.y = (copy.y ?? 0) + OFFSET;
   327→   327→          }
   328→   328→          return { srcId: src.id, clone: copy };
   329→   329→        });
   330→   330→      if (pairs.length === 0) return [];
   331→   331→      const created = pairs.map((p) => p.clone.id);
   332→   332→      commit(({ pages }) => ({
   333→   333→        pages: pages.map((p) => {
   334→   334→          if (p.id !== page.id) return p;
   335→   335→          const list = [...p.components];
   336→   336→          /* 倒序插入：插在各自源组件后面，保持图层相对顺序 */
   337→   337→          for (let i = pairs.length - 1; i >= 0; i--) {
   338→   338→            const idx = list.findIndex((c) => c.id === pairs[i].srcId);
   339→   339→            list.splice(idx >= 0 ? idx + 1 : list.length, 0, pairs[i].clone);
   340→   340→          }
   341→   341→          return { ...p, components: list };
   342→   342→        }),
   343→   343→      }));
   344→   344→      set({ selectedWidgetId: created[created.length - 1], selectedIds: created });
   345→   345→      return created;
   346→   346→    },
   347→   347→
   348→   348→    copyWidget(id) {
   349→   349→      const page = currentPage();
   350→   350→      if (!page) return;
   351→   351→      const src = page.components.find((c) => c.id === id);
   352→   352→      if (!src) return;
   353→   353→      widgetClipboard = { type: src.type, props: deepClone(src.props), width: src.width, align: src.align };
   354→   354→    },
   355→   355→
   356→   356→    pasteWidget() {
   357→   357→      if (!widgetClipboard) return false;
   358→   358→      const page = currentPage();
   359→   359→      if (!page) return false;
   360→   360→      const def = getWidget(widgetClipboard.type);
   361→   361→      if (!def) return false;
   362→   362→      const w: WidgetInstance = {
   363→   363→        id: uid(),
   364→   364→        type: widgetClipboard.type,
   365→   365→        props: deepClone(widgetClipboard.props),
   366→   366→        width: widgetClipboard.width ?? 'full',
   367→   367→        align: widgetClipboard.align ?? 'left',
   368→   368→        mt: 0,
   369→   369→        mb: 8,
   370→   370→      };
   371→   371→      if (page.layout === 'free') {
   372→   372→        const fullW = def.fullBleed ? 375 : 355;
   373→   373→        const maxBottom = page.components.reduce(
   374→   374→          (m, c) => Math.max(m, (c.y ?? 0) + (c.h ?? 64)),
   375→   375→          0
   376→   376→        );
   377→   377→        w.x = def.fullBleed ? 0 : 10;
   378→   378→        w.y = maxBottom + 12;
   379→   379→        w.w = fullW;
   380→   380→      }
   381→   381→      commit(({ pages }) => ({
   382→   382→        pages: pages.map((p) => (p.id === page.id ? { ...p, components: [...p.components, w] } : p)),
   383→   383→      }));
   384→   384→      set({ selectedWidgetId: w.id, selectedIds: [w.id] });
   385→   385→      return true;
   386→   386→    },
   387→   387→
   388→   388→    reorderWidget(id, pos) {
   389→   389→      const page = currentPage();
   390→   390→      if (!page) return;
   391→   391→      commit(({ pages }) => ({
   392→   392→        pages: pages.map((p) => {
   393→   393→          if (p.id !== page.id) return p;
   394→   394→          const from = p.components.findIndex((c) => c.id === id);
   395→   395→          if (from < 0) return p;
   396→   396→          const list = [...p.components];
   397→   397→          const [item] = list.splice(from, 1);
   398→   398→          if (pos === 'front') list.push(item);
   399→   399→          else if (pos === 'back') list.unshift(item);
   400→   400→          else if (pos === 'up') list.splice(Math.min(from + 1, list.length), 0, item);
   401→   401→          else list.splice(Math.max(from - 1, 0), 0, item);
   402→   402→          return { ...p, components: list };
   403→   403→        }),
   404→   404→      }));
   405→   405→    },
   406→   406→
   407→   407→    updateWidget(id, patch) {
   408→   408→      const page = currentPage();
   409→   409→      if (!page) return;

   920→   920→/** 自动保存：脏状态 1.2s 后保存 */
   921→   921→export function bindAutosave() {
   922→   922→  let timer: ReturnType<typeof setTimeout> | null = null;
   923→   923→  let stopped = false;
   924→   924→  const unsub = useBuilder.subscribe((state) => {
   925→   925→    if (stopped) return;
   926→   926→    if (state.dirty && state.view !== 'home') {
   927→   927→      if (timer) clearTimeout(timer);
   928→   928→      timer = setTimeout(() => {
   929→   929→        useBuilder.getState().save();
   930→   930→      }, 1200);
   931→   931→    }
   932→   932→  });
   933→   933→  return () => {
   934→   934→    stopped = true;
   935→   935→    if (timer) clearTimeout(timer);
   936→   936→    unsub();
   937→   937→  };
   938→   938→}
