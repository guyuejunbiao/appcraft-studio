   161→   161→export const useBuilder = create<BuilderState>((set, get) => {
   162→   162→  /** 提交一次变更：记录历史 + 标脏 */
   163→   163→  const commit = (fn: (s: { pages: PageData[]; connections: ConnectionData[] }) => Partial<BuilderState>) => {
   164→   164→    const { pages, connections, past } = get();
   165→   165→    const snap: HistorySnap = { pages: deepClone(pages), connections: deepClone(connections) };
   166→   166→    set({
   167→   167→      past: [...past.slice(-MAX_HISTORY), snap],
   168→   168→      future: [],
   169→   169→      dirty: true,
   170→   170→      ...fn({ pages, connections }),
   171→   171→    });
   172→   172→  };
   173→   173→
   174→   174→  const currentPage = () => get().pages.find((p) => p.id === get().currentPageId);
   175→   175→
   176→   176→  return {
   177→   177→    view: 'home',
   178→   178→    projects: [],
   179→   179→    publishes: [],
   180→   180→    homeLoading: false,
   181→   181→    homeLoaded: false,
   182→   182→    project: null,
   183→   183→    pages: [],
   184→   184→    connections: [],
   185→   185→    currentPageId: null,
   186→   186→    selectedWidgetId: null,
   187→   187→    selectedIds: [],
   188→   188→    dirty: false,
   189→   189→    saving: false,
   190→   190→    lastSavedAt: null,
   191→   191→    batchSnap: null,
   192→   192→    publishOpen: false,
   193→   193→    previewSnapshot: null,
   194→   194→    presets: [],
   195→   195→    presetsLoaded: false,
   196→   196→    past: [],
   197→   197→    future: [],
   198→   198→
   199→   199→    async loadHome() {
   200→   200→      set({ homeLoading: true });
   201→   201→      try {
   202→   202→        const [projects, publishes] = await Promise.all([
   203→   203→          fetch('/api/projects').then((r) => r.json()),
   204→   204→          fetch('/api/publishes').then((r) => r.json()),
   205→   205→        ]);
   206→   206→        set({ projects, publishes, homeLoaded: true });
   207→   207→      } finally {
   208→   208→        set({ homeLoading: false });
   209→   209→      }
   210→   210→    },
   211→   211→
   212→   212→    async createProject(name, description, templateId) {
   213→   213→      const res = await fetch('/api/projects', {
   214→   214→        method: 'POST',
   215→   215→        headers: { 'Content-Type': 'application/json' },
   216→   216→        body: JSON.stringify({ name, description, templateId }),
   217→   217→      });
   218→   218→      const { id } = await res.json();
   219→   219→      await get().loadHome();
   220→   220→      await get().openProject(id);
   221→   221→    },
   222→   222→
   223→   223→    async deleteProject(id) {
   224→   224→      await fetch(`/api/projects/${id}`, { method: 'DELETE' });
   225→   225→      if (get().project?.id === id) set({ project: null, pages: [], connections: [], currentPageId: null });
   226→   226→      await get().loadHome();
   227→   227→    },
   228→   228→
   229→   229→    async renameProject(id, name) {
   230→   230→      const { projects } = get();
   231→   231→      set({ projects: projects.map((p) => (p.id === id ? { ...p, name } : p)) });
   232→   232→      if (get().project?.id === id) {
   233→   233→        set({ project: { ...get().project!, name } });
   234→   234→        get().save();
   235→   235→      }
   236→   236→    },
   237→   237→
   238→   238→    async openProject(id) {
   239→   239→      const res = await fetch(`/api/projects/${id}`);
   240→   240→      if (!res.ok) return;
   241→   241→      const data: ProjectData = await res.json();
   242→   242→      const home = data.pages.find((p) => p.isHome);
   243→   243→      set({
   244→   244→        project: data,
   245→   245→        pages: data.pages,
   246→   246→        connections: data.connections,
   247→   247→        currentPageId: home?.id || data.pages[0]?.id || null,
   248→   248→        selectedWidgetId: null,
   249→   249→        selectedIds: [],
   250→   250→        dirty: false,
   251→   251→        saving: false,
   252→   252→        lastSavedAt: null,
   253→   253→        past: [],
   254→   254→        future: [],
   255→   255→        previewSnapshot: null,
   256→   256→        view: 'editor',
   257→   257→      });
   258→   258→    },
   259→   259→
   260→   260→    setView(v) {
   261→   261→      set({ view: v });
   262→   262→      if (v === 'home') {
   263→   263→        get().loadHome();
   264→   264→      }
   265→   265→    },
   266→   266→
   267→   267→    addWidget(type, index, rect) {
   268→   268→      const page = currentPage();
   269→   269→      if (!page) return;
   270→   270→      const def = getWidget(type);
   271→   271→      if (!def) return;
   272→   272→      const w: WidgetInstance = {
   273→   273→        id: uid(),
   274→   274→        type,
   275→   275→        props: deepClone(def.defaultProps),
   276→   276→        width: 'full',
   277→   277→        align: 'left',
   278→   278→        mt: 0,
   279→   279→        mb: 8,
   280→   280→      };
   281→   281→      /* 自由布局：按落点/智能位置放置（不钳制，可摆到首屏以下，画布自动增高） */
   282→   282→      if (page.layout === 'free') {
   283→   283→        const fullW = def.fullBleed ? 375 : 355;
   284→   284→        const maxBottom = page.components.reduce(
   285→   285→          (m, c) => Math.max(m, (c.y ?? 0) + (c.h ?? 64)),
   286→   286→          0
   287→   287→        );
   288→   288→        w.x = rect?.x ?? (def.fullBleed ? 0 : 10);
   289→   289→        w.y = rect?.y ?? maxBottom + 12;
   290→   290→        w.w = rect?.w ?? fullW;
   291→   291→        w.x = Math.max(0, Math.min(w.x ?? 0, 375 - (w.w ?? fullW)));
   292→   292→      }
   293→   293→      commit(({ pages }) => ({
   294→   294→        pages: pages.map((p) =>
   295→   295→          p.id === page.id
   296→   296→            ? {
   297→   297→                ...p,
   298→   298→                components: index === undefined
   299→   299→                  ? [...p.components, w]
   300→   300→                  : [...p.components.slice(0, index), w, ...p.components.slice(index)],
   301→   301→              }
   302→   302→            : p
   303→   303→        ),
   304→   304→      }));
   305→   305→      set({ selectedWidgetId: w.id, selectedIds: [w.id] });
   306→   306→    },
   307→   307→
   308→   308→    moveWidget(id, toIndex) {
   309→   309→      const page = currentPage();
   310→   310→      if (!page) return;
   311→   311→      commit(({ pages }) => ({
   312→   312→        pages: pages.map((p) => {
   313→   313→          if (p.id !== page.id) return p;
   314→   314→          const from = p.components.findIndex((c) => c.id === id);
   315→   315→          if (from < 0) return p;
   316→   316→          const list = [...p.components];
   317→   317→          const [item] = list.splice(from, 1);
   318→   318→          const target = from < toIndex ? toIndex - 1 : toIndex;
   319→   319→          list.splice(Math.max(0, Math.min(target, list.length)), 0, item);
   320→   320→          return { ...p, components: list };
   321→   321→        }),
   322→   322→      }));
   323→   323→    },
   324→   324→
   325→   325→    removeWidget(id) {
   326→   326→      const page = currentPage();
   327→   327→      if (!page) return;
   328→   328→      commit(({ pages, connections }) => ({
   329→   329→        pages: pages.map((p) => (p.id === page.id ? { ...p, components: p.components.filter((c) => c.id !== id) } : p)),
   330→   330→        connections: connections.filter((c) => !(c.fromPageId === page.id && c.fromWidgetId === id)),
   331→   331→      }));
   332→   332→      if (get().selectedWidgetId === id) set({ selectedWidgetId: null });
   333→   333→      set({ selectedIds: get().selectedIds.filter((x) => x !== id) });
   334→   334→    },
   335→   335→
   336→   336→    removeWidgets(ids) {
   337→   337→      const page = currentPage();
   338→   338→      if (!page) return;
   339→   339→      const set2 = new Set(ids);
   340→   340→      commit(({ pages, connections }) => ({
   341→   341→        pages: pages.map((p) => (p.id === page.id ? { ...p, components: p.components.filter((c) => !set2.has(c.id) || c.locked) } : p)),
   342→   342→        connections: connections.filter((c) => !(c.fromPageId === page.id && set2.has(c.fromWidgetId))),
   343→   343→      }));
   344→   344→      const remaining = get().pages.find((p) => p.id === page.id)?.components.map((c) => c.id) ?? [];
   345→   345→      set({
   346→   346→        selectedIds: get().selectedIds.filter((x) => remaining.includes(x)),
   347→   347→        selectedWidgetId: remaining.includes(get().selectedWidgetId ?? '') ? get().selectedWidgetId : null,
   348→   348→      });
   349→   349→    },
   350→   350→
   351→   351→    duplicateWidget(id) {
   352→   352→      const [copyId] = get().duplicateWidgets([id]);
   353→   353→      if (copyId) set({ selectedWidgetId: copyId, selectedIds: [copyId] });
   354→   354→    },
   355→   355→
   356→   356→    duplicateWidgets(ids) {
   357→   357→      const page = currentPage();
   358→   358→      if (!page || ids.length === 0) return [];
   359→   359→      const OFFSET = 16;
   360→   360→      const srcList = page.components;
   361→   361→      const pairs = ids
   362→   362→        .map((id) => srcList.find((c) => c.id === id))
   363→   363→        .filter((c): c is WidgetInstance => !!c && !c.locked)
   364→   364→        .map((src) => {
   365→   365→          const copy: WidgetInstance = { ...deepClone(src), id: uid() };
   366→   366→          if (page.layout === 'free') {
   367→   367→            const cw = copy.w ?? 355;
   368→   368→            copy.x = Math.max(0, Math.min((copy.x ?? 0) + OFFSET, Math.max(0, 375 - cw)));
   369→   369→            copy.y = (copy.y ?? 0) + OFFSET;
   370→   370→          }
   371→   371→          return { srcId: src.id, clone: copy };
   372→   372→        });
   373→   373→      if (pairs.length === 0) return [];
   374→   374→      const created = pairs.map((p) => p.clone.id);

   571→   571→        const map: Record<string, Partial<Pick<WidgetInstance, 'x' | 'y'>>> = {};
   572→   572→        let cursor = first.y;
   573→   573→        for (const r of items) {

   581→   581→    /* 一键整理（对标 m3e-canvas Tidy）：编组聚成单元 → 按行分组 → 通栏贴边、
   582→   582→       其余从顶部边距起按 16px 行距归位堆叠；一条 commit 一条历史可撤销 */
   583→   583→    tidyPage(sizes) {
   584→   584→      const page = currentPage();
   585→   585→      if (!page || page.layout !== 'free') return;
   586→   586→      const movable = page.components.filter(
   587→   587→        (c) => !c.locked && !c.hidden && typeof c.x === 'number' && typeof c.y === 'number'
   588→   588→      );
   589→   589→      if (movable.length === 0) return;
   590→   590→      const dim = (c: WidgetInstance) => ({
   591→   591→        w: sizes?.[c.id]?.w ?? c.w ?? 355,
   592→   592→        h: sizes?.[c.id]?.h ?? c.h ?? 56,
   593→   593→      });
   594→   594→
   595→   595→      /* 编组聚合成单元（成员保持相对偏移整组移动），散件各自成单元 */
   596→   596→      const groupsMap = new Map<string, WidgetInstance[]>();
   597→   597→      const singles: WidgetInstance[] = [];
   598→   598→      for (const c of movable) {
   599→   599→        if (c.group) {
   600→   600→          const arr = groupsMap.get(c.group) ?? [];
   601→   601→          arr.push(c);
   602→   602→          groupsMap.set(c.group, arr);
   603→   603→        } else singles.push(c);
   604→   604→      }
   605→   605→      type Unit = { x: number; y: number; w: number; h: number; members: { id: string; dx: number; dy: number }[] };
   606→   606→      const units: Unit[] = singles.map((c) => {
   607→   607→        const d = dim(c);
   608→   608→        return { x: c.x ?? 0, y: c.y ?? 0, w: d.w, h: d.h, members: [{ id: c.id, dx: 0, dy: 0 }] };
   609→   609→      });
   610→   610→      for (const members of groupsMap.values()) {
   611→   611→        if (members.length === 0) continue;
   612→   612→        const dims = members.map((m) => ({ m, d: dim(m) }));
   613→   613→        const minX = Math.min(...dims.map(({ m }) => m.x ?? 0));
   614→   614→        const minY = Math.min(...dims.map(({ m }) => m.y ?? 0));
   615→   615→        const maxR = Math.max(...dims.map(({ m, d }) => (m.x ?? 0) + d.w));
   616→   616→        const maxB = Math.max(...dims.map(({ m, d }) => (m.y ?? 0) + d.h));
   617→   617→        units.push({
   618→   618→          x: minX,
   619→   619→          y: minY,
   620→   620→          w: maxR - minX,
   621→   621→          h: maxB - minY,
   622→   622→          members: dims.map(({ m }) => ({ id: m.id, dx: (m.x ?? 0) - minX, dy: (m.y ?? 0) - minY })),
   623→   623→        });
   624→   624→      }
   625→   625→
   626→   626→      /* 按 y（再按 x）排序 → 行分组：与当前行首单元 y 差 < 28px 视为同一视觉行 */
   627→   627→      units.sort((a, b) => a.y - b.y || a.x - b.x);
   628→   628→      const ROW_TOL = 28;
   629→   629→      const rows: Unit[][] = [];
   630→   630→      for (const u of units) {
   631→   631→        const row = rows[rows.length - 1];
   632→   632→        if (row && Math.abs(u.y - row[0].y) < ROW_TOL) row.push(u);
   633→   633→        else rows.push([u]);
   634→   634→      }
   635→   635→
   636→   636→      /* 重排：行内从左到右依次排（列距 8），行距 16，起始边距 16；
   637→   637→         行内总宽超出屏幕（375 - 2×10 边距）则拆为逐行堆叠（通栏自然贴边） */
   638→   638→      const M = 10;
   639→   639→      const COL_G = 8;
   640→   640→      const ROW_G = 16;
   641→   641→      const SCREEN_W = 375;
   642→   642→      const map: Record<string, Partial<WidgetInstance>> = {};
   643→   643→      let cursorY = M + 6;
   644→   644→      for (const row of rows) {
   645→   645→        row.sort((a, b) => a.x - b.x);
   646→   646→        const totalW = row.reduce((acc, u) => acc + u.w, 0) + COL_G * (row.length - 1);
   647→   647→        if (totalW <= SCREEN_W - M * 2) {
   648→   648→          const rowH = Math.max(...row.map((u) => u.h));
   649→   649→          let cx = M;
   650→   650→          for (const u of row) {
   651→   651→            for (const m of u.members) map[m.id] = { x: Math.round(cx + m.dx), y: Math.round(cursorY + m.dy) };
   652→   652→            cx += u.w + COL_G;
   653→   653→          }
   654→   654→          cursorY += rowH + ROW_G;
   655→   655→        } else {
   656→   656→          for (const u of row) {
   657→   657→            for (const m of u.members) map[m.id] = { x: Math.round(M + m.dx), y: Math.round(cursorY + m.dy) };
   658→   658→            cursorY += u.h + ROW_G;
   659→   659→          }
   660→   660→        }
   661→   661→      }
   662→   662→      get().patchWidgets(map);
   663→   663→    },
